## 使用指南

### 导入枚举类型

```typescript
// 从lowcode-shared统一导入
import {
  DatabaseType,
  RelationType,
  LayoutType,
  FormControlType,
  SortDirection
} from '@smartabp/lowcode-shared'

// 使用枚举
const dbType: DatabaseType = DatabaseType.PostgreSQL
const relation: RelationType = RelationType.OneToMany
```

### 使用国际化错误消息

```typescript
import {
  setCurrentLocale,
  formatValidationError,
  ErrorMessageContext
} from '@smartabp/lowcode-shared'

// 设置语言
setCurrentLocale('zh-CN')

// 单个错误格式化
const msg = formatValidationError('name', 'required')
// 输出: "名称是必填项"

// 批量错误格式化
const context = new ErrorMessageContext('zh-CN')
const messages = context.formatBatch([
  { field: 'name', key: 'required' },
  { field: 'email', key: 'invalid_email' }
])
// 输出: ["名称是必填项", "邮箱不是有效的邮箱地址"]
```

### 使用枚举工具函数

```typescript
import {
  enumToOptions,
  isValidEnumValue,
  DatabaseType
} from '@smartabp/lowcode-shared'

// 转换为UI选项
const options = enumToOptions(DatabaseType, {
  'PostgreSQL': 'PostgreSQL数据库',
  'MySQL': 'MySQL数据库'
})

// 验证枚举值
const isValid = isValidEnumValue(DatabaseType, 'PostgreSQL') // true
```
