public class ElasticsearchAuditLogStore : IAuditLogStore
{
    private readonly IElasticClient _elasticClient;
    private readonly ILogger<ElasticsearchAuditLogStore> _logger;

    public async Task SaveAuditLogAsync(PermissionAuditLog auditLog)
    {
        try
        {
            // 增强审计日志信息
            var enrichedLog = await EnrichAuditLogAsync(auditLog);

            // 计算风险等级
            enrichedLog.RiskLevel = await CalculateRiskLevelAsync(enrichedLog);

            // 获取地理位置信息
            enrichedLog.GeoLocation = await GetGeoLocationAsync(enrichedLog.ClientIP);

            // 保存到Elasticsearch
            var indexName = $"permission-audit-{DateTime.UtcNow:yyyy-MM}";
            await _elasticClient.IndexAsync(enrichedLog, idx => idx.Index(indexName));

            // 实时风险分析
            await ProcessRealTimeRiskAnalysis(enrichedLog);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "审计日志保存失败: {AuditLog}", auditLog);
            throw;
        }
    }

    private async Task<PermissionAuditLog> EnrichAuditLogAsync(PermissionAuditLog auditLog)
    {
        // 用户信息增强
        auditLog.UserInfo = await _userService.GetUserInfoAsync(auditLog.UserId);

        // 会话信息增强
        auditLog.SessionInfo = await _sessionService.GetSessionInfoAsync(auditLog.SessionId);

        // 权限上下文增强
        auditLog.PermissionContext = await _permissionService.GetPermissionContextAsync(
            auditLog.UserId, auditLog.Permission, auditLog.Resource);

        return auditLog;
    }

    private async Task<RiskLevel> CalculateRiskLevelAsync(PermissionAuditLog auditLog)
    {
        var riskScore = 0;

        // 时间风险因子
        if (IsOutsideBusinessHours(auditLog.Timestamp))
            riskScore += 20;

        // 地理位置风险因子
        if (await IsUnusualLocationAsync(auditLog.UserId, auditLog.ClientIP))
            riskScore += 30;

        // 权限敏感性风险因子
        if (auditLog.Permission.IsSensitive)
            riskScore += 25;

        // 频率风险因子
        if (await IsHighFrequencyAccessAsync(auditLog.UserId, auditLog.Permission))
            riskScore += 15;

        // 失败次数风险因子
        if (auditLog.Result == AuditResult.Failed)
        {
            var recentFailures = await GetRecentFailuresAsync(auditLog.UserId);
            riskScore += Math.Min(recentFailures * 10, 50);
        }

        return riskScore switch
        {
            < 30 => RiskLevel.Low,
            < 60 => RiskLevel.Medium,
            < 80 => RiskLevel.High,
            _ => RiskLevel.Critical
        };
    }
}

// 实时风险告警服务
public class RealTimeRiskAlertService
{
    private readonly IEventBus _eventBus;
    private readonly INotificationService _notificationService;

    public async Task ProcessRiskAlertAsync(PermissionAuditLog auditLog)
    {
        if (auditLog.RiskLevel >= RiskLevel.High)
        {
            var alert = new SecurityAlert
            {
                Id = Guid.NewGuid(),
                Type = SecurityAlertType.HighRiskPermissionAccess,
                Severity = auditLog.RiskLevel,
                UserId = auditLog.UserId,
                Description = $"高风险权限访问: {auditLog.Permission.Name}",
                Timestamp = DateTime.UtcNow,
                Context = auditLog
            };

            // 发送实时告警
            await _notificationService.SendSecurityAlertAsync(alert);

            // 发布安全事件
            await _eventBus.PublishAsync(new SecurityIncidentEvent(alert));

            // 如果是关键风险，立即通知安全团队
            if (auditLog.RiskLevel == RiskLevel.Critical)
            {
                await _notificationService.SendUrgentSecurityAlertAsync(alert);
            }
        }
    }
}

// SOX合规报告生成器
public class SOXComplianceReportGenerator : IComplianceReportGenerator
{
    public async Task<SOXComplianceReport> GenerateSOXReportAsync(
        DateTime startDate,
        DateTime endDate,
        Guid? tenantId = null)
    {
        var query = BuildSOXQueryAsync(startDate, endDate, tenantId);
        var auditData = await _elasticClient.SearchAsync<PermissionAuditLog>(query);

        var report = new SOXComplianceReport
        {
            ReportPeriod = new DateRange(startDate, endDate),
            TenantId = tenantId,
            GeneratedAt = DateTime.UtcNow,

            // 关键控制点分析
            AccessControlChanges = await AnalyzeAccessControlChangesAsync(auditData.Documents),
            PrivilegedUserAccess = await AnalyzePrivilegedUserAccessAsync(auditData.Documents),
            DataAccessPatterns = await AnalyzeDataAccessPatternsAsync(auditData.Documents),
            SecurityIncidents = await AnalyzeSecurityIncidentsAsync(auditData.Documents),

            // 合规状态评估
            ComplianceStatus = await EvaluateSOXComplianceAsync(auditData.Documents),
            Recommendations = await GenerateSOXRecommendationsAsync(auditData.Documents)
        };

        return report;
    }

    private async Task<List<AccessControlChange>> AnalyzeAccessControlChangesAsync(
        IEnumerable<PermissionAuditLog> auditLogs)
    {
        return auditLogs
            .Where(log => log.Action.StartsWith("Permission.") || log.Action.StartsWith("Role."))
            .Select(log => new AccessControlChange
            {
                Timestamp = log.Timestamp,
                UserId = log.UserId,
                Action = log.Action,
                Target = log.Resource,
                Approver = log.ApprovalInfo?.ApproverId,
                BusinessJustification = log.ApprovalInfo?.BusinessJustification,
                RiskLevel = log.RiskLevel
            })
            .OrderByDescending(change => change.Timestamp)
            .ToList();
    }
}

      }
