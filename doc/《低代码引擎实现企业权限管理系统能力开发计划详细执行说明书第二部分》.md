```csharp
    }
```

```csharp
public class OptimizedPermissionInheritanceEngine
{
    private readonly IMemoryCache _computationCache;

    public async Task<List<EffectivePermission>> CalculateEffectivePermissionsAsync(
        Guid userId,
        List<Role> userRoles)
    {
        var cacheKey = $"effective_permissions:{userId}:{string.Join(",", userRoles.Select(r => r.Id))}";

        return await _computationCache.GetOrCreateAsync(cacheKey, async factory =>
        {
            factory.SetAbsoluteExpiration(TimeSpan.FromMinutes(15));

            // 并行计算各种权限来源
            var tasks = new[]
            {
                CalculateDirectPermissionsAsync(userId),
                CalculateRolePermissionsAsync(userRoles),
                CalculateInheritedPermissionsAsync(userRoles),
                CalculateOrganizationPermissionsAsync(userId)
            };

            var results = await Task.WhenAll(tasks);

            // 权限合并和冲突解决
            return MergeAndResolvePermissions(results);
        });
    }

    private List<EffectivePermission> MergeAndResolvePermissions(IEnumerable<Permission>[] permissionSources)
    {
        var permissionMap = new Dictionary<string, EffectivePermission>();

        // 优先级: 直接权限 > 角色权限 > 继承权限 > 组织权限
        var priorityOrder = new[] { PermissionSource.Direct, PermissionSource.Role, PermissionSource.Inherited, PermissionSource.Organization };

        foreach (var source in priorityOrder)
        {
            var sourceIndex = Array.IndexOf(priorityOrder, source);
            foreach (var permission in permissionSources[sourceIndex])
            {
                var key = $"{permission.Name}:{permission.Resource}";
                if (!permissionMap.ContainsKey(key))
                {
                    permissionMap[key] = new EffectivePermission
                    {
                        Name = permission.Name,
                        Resource = permission.Resource,
                        IsGranted = permission.IsGranted,
                        Source = source,
                        Priority = priorityOrder.Length - sourceIndex
                    };
                }
            }
        }

        return permissionMap.Values.ToList();
    }
}
```

```typescript
      p95ResponseTime: checkMetric?.getPercentile(95) || 0,
      cacheHitRate: this.calculateCacheHitRate(hitMetric, missMetric),
```

```vue
const layoutRoles = (roles: Role[], relations: InheritanceRelation[]) => {
  // 简化的层次布局算法
  const levels: Role[][] = []
  const processed = new Set<string>()

  // 找到根角色（没有父角色的角色）
  const rootRoles = roles.filter(role =>
    !relations.some(rel => rel.childId === role.id)
  )

  levels[0] = rootRoles
  rootRoles.forEach(role => processed.add(role.id))

  // 逐层计算子角色位置
  let currentLevel = 0
  while (levels[currentLevel]?.length) {
    const nextLevel: Role[] = []

    levels[currentLevel].forEach(parentRole => {
      const children = relations
        .filter(rel => rel.parentId === parentRole.id)
        .map(rel => roles.find(r => r.id === rel.childId))
        .filter(child => child && !processed.has(child.id))

      children.forEach(child => {
        if (child) {
          nextLevel.push(child)
          processed.add(child.id)
        }
      })
    })

    if (nextLevel.length) {
      levels[++currentLevel] = nextLevel
    } else {
      break
    }
  }

  // 计算节点坐标
  const result: RoleNode[] = []
  levels.forEach((levelRoles, levelIndex) => {
    const y = 100 + levelIndex * 100
    const levelWidth = canvasWidth.value - 200
    const nodeSpacing = levelWidth / (levelRoles.length + 1)

    levelRoles.forEach((role, index) => {
      result.push({
        ...role,
        x: 100 + (index + 1) * nodeSpacing - nodeWidth.value / 2,
        y: y
      })
    })
  })

  return result
}
```

```csharp
public class TenantShardingStrategy : IShardingStrategy
{
    private readonly IConfiguration _configuration;
    private readonly ILogger<TenantShardingStrategy> _logger;

    public string GetConnectionString(Guid? tenantId)
    {
        if (!tenantId.HasValue)
            return _configuration.GetConnectionString("Default");

        var tenantInfo = GetTenantInfo(tenantId.Value);

        // 大租户独立数据库
        if (tenantInfo.UserCount > 100000)
        {
            return _configuration.GetConnectionString($"Tenant_{tenantId:N}");
        }

        // 中等租户分片数据库
        if (tenantInfo.UserCount > 10000)
        {
            var shardIndex = tenantId.Value.GetHashCode() % 10;
            return _configuration.GetConnectionString($"Shard_{shardIndex:D2}");
        }

        // 小租户共享数据库
        return _configuration.GetConnectionString("Shared");
    }

    public string GetTablePrefix(Guid? tenantId)
    {
        if (!tenantId.HasValue) return "";

        var tenantInfo = GetTenantInfo(tenantId.Value);

        // 大租户独立表前缀
        if (tenantInfo.UserCount > 100000)
            return $"T_{tenantId:N}_";

        return "";
    }
```
