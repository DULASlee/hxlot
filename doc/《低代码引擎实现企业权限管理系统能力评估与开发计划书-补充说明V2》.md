### 2.1 表单设计器核心架构

#### 2.1.1 表单设计器插件实现

**🎯 目标**: 基于现有EntityDesigner，扩展支持拖拽的表单定制功能

**表单设计器插件架构**:
```typescript
// 表单设计器插件 - 基于现有微内核架构
class FormDesignerPlugin implements LowCodePlugin {
  metadata = {
    name: "FormDesignerPlugin",
    version: "1.0.0",
    capabilities: ["form-designer", "drag-drop", "dynamic-validation", "conditional-display"],
    dependencies: ["vue3", "element-plus", "vuedraggable", "form-render"]
  }

  async canHandle(schema: any): Promise<boolean> {
    return schema.type === "form" &&
           schema.designMode === "visual" &&
           schema.features?.includes("drag-drop")
  }

  async generate(schema: FormSchema, config: any, context: PluginContext): Promise<GeneratedCode> {
    const formDesigner = this.generateFormDesignerComponent(schema)
    const formRenderer = this.generateFormRendererComponent(schema)
    const formValidators = this.generateFormValidators(schema)
    const formStore = this.generateFormStore(schema)

    return {
      components: [formDesigner, formRenderer],
      validators: [formValidators],
      stores: [formStore],
      types: this.generateFormTypes(schema),
      utils: this.generateFormUtils(schema)
    }
  }

  private generateFormDesignerComponent(schema: FormSchema): VueComponent {
    return {
      template: this.buildFormDesignerTemplate(schema),
      script: this.buildFormDesignerScript(schema),
      style: this.buildFormDesignerStyle(schema)
    }
  }
}
```

```vue
const filterComponents = (components: any[]) => {
  if (!componentSearchKeyword.value) return components

  const keyword = componentSearchKeyword.value.toLowerCase()
  return components.filter(component =>
    component.label.toLowerCase().includes(keyword) ||
    component.type.toLowerCase().includes(keyword)
  )
}

const getFieldComponent = (fieldType: string) => {
  const componentMap = {

    textarea: "el-input",
    number: "el-input-number",

    radio: "el-radio-group",
    checkbox: "el-checkbox-group",
    switch: "el-switch",

    time: "el-time-picker",
```
