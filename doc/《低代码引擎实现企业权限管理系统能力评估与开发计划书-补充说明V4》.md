## 5. 高性能缓存架构实现

### 5.1 三级缓存系统核心实现

#### 5.1.1 权限缓存引擎

**🎯 目标**: 实现内存→Redis→数据库的三级缓存架构，支持预热、失效、一致性保证

**权限缓存引擎核心实现**:
```csharp
// 权限缓存引擎 - 三级缓存架构
public class PermissionCacheEngine : IPermissionCacheEngine
{
    private readonly IMemoryCache _memoryCache;
    private readonly IDistributedCache _distributedCache;
    private readonly IPermissionRepository _permissionRepository;
    private readonly ILogger<PermissionCacheEngine> _logger;
    private readonly PermissionCacheOptions _options;
    private readonly IEventBus _eventBus;

    // 分布式锁，防止缓存击穿
    private readonly IDistributedLock _distributedLock;

    // 布隆过滤器，防止缓存穿透
    private readonly IBloomFilter _bloomFilter;

    // 性能计数器
    private readonly PerformanceCounter _hitRateCounter;
    private readonly PerformanceCounter _loadTimeCounter;

    public PermissionCacheEngine(
        IMemoryCache memoryCache,
        IDistributedCache distributedCache,
        IPermissionRepository permissionRepository,
        ILogger<PermissionCacheEngine> logger,
        IOptions<PermissionCacheOptions> options,
        IEventBus eventBus,
        IDistributedLock distributedLock,
        IBloomFilter bloomFilter)
    {
        _memoryCache = memoryCache;
        _distributedCache = distributedCache;
        _permissionRepository = permissionRepository;
        _logger = logger;
        _options = options.Value;
        _eventBus = eventBus;
        _distributedLock = distributedLock;
        _bloomFilter = bloomFilter;

        // 初始化性能计数器
        _hitRateCounter = new PerformanceCounter("PermissionCache", "HitRate");
        _loadTimeCounter = new PerformanceCounter("PermissionCache", "LoadTime");
    }

    public async Task<UserPermissionSet> GetUserPermissionsAsync(
        Guid? tenantId,
        Guid userId,
        CancellationToken cancellationToken = default)
    {
        var stopwatch = Stopwatch.StartNew();
        var cacheKey = GeneratePermissionCacheKey(tenantId, userId);

        try
        {
            // L1缓存：内存缓存
            if (_memoryCache.TryGetValue(cacheKey, out UserPermissionSet cachedPermissions))
            {
                _hitRateCounter.Increment("L1");
                _logger.LogDebug("L1缓存命中: {CacheKey}", cacheKey);
                return cachedPermissions;
            }

            // L2缓存：Redis分布式缓存
            var serializedPermissions = await _distributedCache.GetStringAsync(cacheKey, cancellationToken);
            if (!string.IsNullOrEmpty(serializedPermissions))
            {
                var permissions = JsonSerializer.Deserialize<UserPermissionSet>(serializedPermissions);

                // 回写到L1缓存
                var memoryOptions = new MemoryCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(_options.L1CacheExpirationMinutes),
                    Priority = CacheItemPriority.High,
                    Size = EstimatePermissionSetSize(permissions)
                };

                _memoryCache.Set(cacheKey, permissions, memoryOptions);

                _hitRateCounter.Increment("L2");
                _logger.LogDebug("L2缓存命中: {CacheKey}", cacheKey);
                return permissions;
            }

            // 布隆过滤器检查，防止缓存穿透
            if (!_bloomFilter.Contains(cacheKey))
            {
                _logger.LogWarning("布隆过滤器拦截可能不存在的权限查询: {CacheKey}", cacheKey);
                return UserPermissionSet.Empty;
            }

            // L3缓存：数据库 + 分布式锁防止缓存击穿
            return await LoadPermissionsWithLockAsync(tenantId, userId, cacheKey, cancellationToken);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "获取用户权限失败: TenantId={TenantId}, UserId={UserId}", tenantId, userId);

            // 降级策略：返回最小权限集
            return await GetFallbackPermissionsAsync(tenantId, userId, cancellationToken);
        }
        finally
        {
            stopwatch.Stop();
            _loadTimeCounter.Record(stopwatch.ElapsedMilliseconds);

            // 记录性能指标
            if (stopwatch.ElapsedMilliseconds > _options.SlowQueryThresholdMs)
            {
                _logger.LogWarning("权限查询耗时过长: {ElapsedMs}ms, CacheKey={CacheKey}",
                    stopwatch.ElapsedMilliseconds, cacheKey);
            }
        }
    }

    private async Task<UserPermissionSet> LoadPermissionsWithLockAsync(
        Guid? tenantId,
        Guid userId,
        string cacheKey,
        CancellationToken cancellationToken)
    {
        var lockKey = $"permission_lock:{cacheKey}";
        var lockExpiry = TimeSpan.FromMinutes(2);

        using var lockHandle = await _distributedLock.AcquireAsync(lockKey, lockExpiry, cancellationToken);

        if (lockHandle == null)
        {
            _logger.LogWarning("获取分布式锁失败，等待其他实例加载权限: {CacheKey}", cacheKey);

            // 等待一段时间后重试缓存
            await Task.Delay(100, cancellationToken);

            var retryResult = await _distributedCache.GetStringAsync(cacheKey, cancellationToken);
            if (!string.IsNullOrEmpty(retryResult))
            {
                return JsonSerializer.Deserialize<UserPermissionSet>(retryResult);
            }

            // 如果还是没有，直接从数据库加载（不缓存）
            return await _permissionRepository.GetUserPermissionsAsync(tenantId, userId, cancellationToken);
        }

        _logger.LogDebug("获取分布式锁成功，开始加载权限: {CacheKey}", cacheKey);

        // 双重检查，防止其他实例已经加载
        var doubleCheckResult = await _distributedCache.GetStringAsync(cacheKey, cancellationToken);
        if (!string.IsNullOrEmpty(doubleCheckResult))
        {
            return JsonSerializer.Deserialize<UserPermissionSet>(doubleCheckResult);
        }

        // 从数据库加载权限
        var permissions = await LoadPermissionsFromDatabaseAsync(tenantId, userId, cancellationToken);

        // 缓存到各级缓存
        await SetPermissionsCacheAsync(cacheKey, permissions, cancellationToken);

        // 添加到布隆过滤器
        _bloomFilter.Add(cacheKey);

        _hitRateCounter.Increment("L3");
        _logger.LogDebug("从数据库加载权限并缓存: {CacheKey}", cacheKey);

        return permissions;
    }

    private async Task<UserPermissionSet> LoadPermissionsFromDatabaseAsync(
        Guid? tenantId,
        Guid userId,
        CancellationToken cancellationToken)
    {
        var stopwatch = Stopwatch.StartNew();

        try
        {
            // 并行加载用户的各种权限信息
            var tasks = new[]
            {
                _permissionRepository.GetUserDirectPermissionsAsync(tenantId, userId, cancellationToken),
                _permissionRepository.GetUserRolePermissionsAsync(tenantId, userId, cancellationToken),
                _permissionRepository.GetUserOrganizationPermissionsAsync(tenantId, userId, cancellationToken),
                _permissionRepository.GetUserDataPermissionsAsync(tenantId, userId, cancellationToken)
            };

            var results = await Task.WhenAll(tasks);

            var directPermissions = results[0];
            var rolePermissions = results[1];
            var organizationPermissions = results[2];
            var dataPermissions = results[3];

            // 计算有效权限（考虑优先级和冲突解决）
            var effectivePermissions = CalculateEffectivePermissions(
                directPermissions,
                rolePermissions,
                organizationPermissions);

            var permissionSet = new UserPermissionSet
            {
                UserId = userId,
                TenantId = tenantId,
                DirectPermissions = directPermissions,
                RolePermissions = rolePermissions,
                OrganizationPermissions = organizationPermissions,
                EffectivePermissions = effectivePermissions,
                DataPermissions = dataPermissions,
                LoadTime = DateTime.UtcNow,
                Version = GeneratePermissionVersion()
            };

            return permissionSet;
        }
        finally
        {
            stopwatch.Stop();
            _logger.LogDebug("数据库权限加载耗时: {ElapsedMs}ms", stopwatch.ElapsedMilliseconds);
        }
    }

    private async Task SetPermissionsCacheAsync(
        string cacheKey,
        UserPermissionSet permissions,
        CancellationToken cancellationToken)
    {
        var serializedPermissions = JsonSerializer.Serialize(permissions, new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
        });

        // L2缓存：Redis分布式缓存
        var distributedOptions = new DistributedCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(_options.L2CacheExpirationMinutes),
            SlidingExpiration = TimeSpan.FromMinutes(_options.L2SlidingExpirationMinutes)
        };

        await _distributedCache.SetStringAsync(cacheKey, serializedPermissions, distributedOptions, cancellationToken);

        // L1缓存：内存缓存
        var memoryOptions = new MemoryCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(_options.L1CacheExpirationMinutes),
            Priority = CacheItemPriority.High,
            Size = EstimatePermissionSetSize(permissions),
            PostEvictionCallbacks =
            {
                new PostEvictionCallbackRegistration
                {
                    EvictionCallback = OnPermissionEvicted,
                    State = cacheKey
                }
            }
        };

        _memoryCache.Set(cacheKey, permissions, memoryOptions);
    }

    private void OnPermissionEvicted(object key, object value, EvictionReason reason, object state)
    {
        _logger.LogDebug("权限缓存被逐出: Key={Key}, Reason={Reason}", key, reason);

        // 如果是因为容量不足被逐出，记录警告
        if (reason == EvictionReason.Capacity)
        {
            _logger.LogWarning("内存缓存容量不足，权限被逐出: {Key}", key);
        }
    }

    public async Task InvalidateUserPermissionsAsync(
        Guid? tenantId,
        Guid userId,
        InvalidationReason reason = InvalidationReason.Manual,
        CancellationToken cancellationToken = default)
    {
        var cacheKey = GeneratePermissionCacheKey(tenantId, userId);

        _logger.LogInformation("开始失效用户权限缓存: TenantId={TenantId}, UserId={UserId}, Reason={Reason}",
            tenantId, userId, reason);

        // L1缓存失效
        _memoryCache.Remove(cacheKey);

        // L2缓存失效
        await _distributedCache.RemoveAsync(cacheKey, cancellationToken);

        // 发布缓存失效事件，通知其他实例
        await _eventBus.PublishAsync(new PermissionCacheInvalidatedEvent
        {
            TenantId = tenantId,
            UserId = userId,
            CacheKey = cacheKey,
            Reason = reason,
            Timestamp = DateTime.UtcNow
        }, cancellationToken);

        _logger.LogInformation("用户权限缓存失效完成: {CacheKey}", cacheKey);
    }

    public async Task InvalidateRolePermissionsAsync(
        Guid? tenantId,
        Guid roleId,
        CancellationToken cancellationToken = default)
    {
        // 获取该角色下的所有用户
        var userIds = await _permissionRepository.GetUsersInRoleAsync(tenantId, roleId, cancellationToken);

        _logger.LogInformation("开始失效角色相关权限缓存: TenantId={TenantId}, RoleId={RoleId}, UserCount={UserCount}",
            tenantId, roleId, userIds.Count);

        // 并行失效所有相关用户的权限缓存
        var tasks = userIds.Select(userId =>
            InvalidateUserPermissionsAsync(tenantId, userId, InvalidationReason.RoleChanged, cancellationToken));

        await Task.WhenAll(tasks);

        _logger.LogInformation("角色相关权限缓存失效完成: RoleId={RoleId}", roleId);
    }

    public async Task PrewarmPermissionCacheAsync(
        Guid? tenantId,
        IEnumerable<Guid> userIds,
        CancellationToken cancellationToken = default)
    {
        _logger.LogInformation("开始预热权限缓存: TenantId={TenantId}, UserCount={UserCount}",
            tenantId, userIds.Count());

        var semaphore = new SemaphoreSlim(_options.PrewarmConcurrency, _options.PrewarmConcurrency);
        var tasks = userIds.Select(async userId =>
        {
            await semaphore.WaitAsync(cancellationToken);
            try
            {
                await GetUserPermissionsAsync(tenantId, userId, cancellationToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "预热用户权限缓存失败: UserId={UserId}", userId);
            }
            finally
            {
                semaphore.Release();
            }
        });

        await Task.WhenAll(tasks);

        _logger.LogInformation("权限缓存预热完成: TenantId={TenantId}", tenantId);
    }

    // 权限计算核心算法
    private List<EffectivePermission> CalculateEffectivePermissions(
        List<DirectPermission> directPermissions,
        List<RolePermission> rolePermissions,
        List<OrganizationPermission> organizationPermissions)
    {
        var effectivePermissions = new Dictionary<string, EffectivePermission>();

        // 优先级：直接权限 > 角色权限 > 组织权限

        // 1. 组织权限（最低优先级）
        foreach (var orgPerm in organizationPermissions)
        {
            var key = $"{orgPerm.PermissionName}:{orgPerm.Resource}";
            effectivePermissions[key] = new EffectivePermission
            {
                PermissionName = orgPerm.PermissionName,
                Resource = orgPerm.Resource,
                IsGranted = orgPerm.IsGranted,
                Source = PermissionSource.Organization,
                SourceId = orgPerm.OrganizationId.ToString(),
                Priority = 1,
                ExpirationTime = orgPerm.ExpirationTime
            };
        }

        // 2. 角色权限（中等优先级）
        foreach (var rolePerm in rolePermissions)
        {
            var key = $"{rolePerm.PermissionName}:{rolePerm.Resource}";

            // 如果已存在更高优先级的权限，跳过
            if (effectivePermissions.TryGetValue(key, out var existing) && existing.Priority >= 2)
                continue;

            effectivePermissions[key] = new EffectivePermission
            {
                PermissionName = rolePerm.PermissionName,
                Resource = rolePerm.Resource,
                IsGranted = rolePerm.IsGranted,
                Source = PermissionSource.Role,
                SourceId = rolePerm.RoleId.ToString(),
                Priority = 2,
                ExpirationTime = rolePerm.ExpirationTime
            };
        }

        // 3. 直接权限（最高优先级）
        foreach (var directPerm in directPermissions)
        {
            var key = $"{directPerm.PermissionName}:{directPerm.Resource}";
            effectivePermissions[key] = new EffectivePermission
            {
                PermissionName = directPerm.PermissionName,
                Resource = directPerm.Resource,
                IsGranted = directPerm.IsGranted,
                Source = PermissionSource.Direct,
                SourceId = directPerm.UserId.ToString(),
                Priority = 3,
                ExpirationTime = directPerm.ExpirationTime
            };
        }

        // 过滤已过期的权限
        var now = DateTime.UtcNow;
        return effectivePermissions.Values
            .Where(p => !p.ExpirationTime.HasValue || p.ExpirationTime > now)
            .ToList();
    }

    // 辅助方法
    private string GeneratePermissionCacheKey(Guid? tenantId, Guid userId)
    {
        return $"permissions:{tenantId?.ToString() ?? "host"}:{userId}";
    }

    private int EstimatePermissionSetSize(UserPermissionSet permissions)
    {
        // 估算权限集合的内存大小（字节）
        var baseSize = 1024; // 基础对象大小
        var permissionSize = (permissions.EffectivePermissions?.Count ?? 0) * 256; // 每个权限约256字节
        return baseSize + permissionSize;
    }

    private string GeneratePermissionVersion()
    {
        return DateTime.UtcNow.Ticks.ToString("X");
    }

    private async Task<UserPermissionSet> GetFallbackPermissionsAsync(
        Guid? tenantId,
        Guid userId,
        CancellationToken cancellationToken)
    {
        // 降级策略：返回最基本的权限集合
        try
        {
            return await _permissionRepository.GetBasicPermissionsAsync(tenantId, userId, cancellationToken);
        }
        catch
        {
            return UserPermissionSet.Empty;
        }
    }
}
```

### 5.2 缓存预热与失效策略

#### 5.2.1 智能缓存预热服务

**智能缓存预热实现**:
```csharp
// 智能缓存预热服务
public class PermissionCachePrewarmService : BackgroundService
{
    private readonly IPermissionCacheEngine _cacheEngine;
    private readonly IPermissionRepository _repository;
    private readonly ILogger<PermissionCachePrewarmService> _logger;
    private readonly PermissionCacheOptions _options;
    private readonly IEventBus _eventBus;

    public PermissionCachePrewarmService(
        IPermissionCacheEngine cacheEngine,
        IPermissionRepository repository,
        ILogger<PermissionCachePrewarmService> logger,
        IOptions<PermissionCacheOptions> options,
        IEventBus eventBus)
    {
        _cacheEngine = cacheEngine;
        _repository = repository;
        _logger = logger;
        _options = options.Value;
        _eventBus = eventBus;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("权限缓存预热服务启动");

        // 启动时立即执行一次预热
        await PerformPrewarmAsync(stoppingToken);

        // 定期预热
        using var timer = new PeriodicTimer(TimeSpan.FromMinutes(_options.PrewarmIntervalMinutes));

        while (!stoppingToken.IsCancellationRequested && await timer.WaitForNextTickAsync(stoppingToken))
        {
            try
            {
                await PerformPrewarmAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "定期权限缓存预热失败");
            }
        }
    }

    private async Task PerformPrewarmAsync(CancellationToken cancellationToken)
    {
        var stopwatch = Stopwatch.StartNew();

        try
        {
            _logger.LogInformation("开始权限缓存预热");

            // 获取需要预热的用户策略
            var prewarmStrategy = await DeterminePrewarmStrategyAsync(cancellationToken);

            await ExecutePrewarmStrategyAsync(prewarmStrategy, cancellationToken);

            stopwatch.Stop();
            _logger.LogInformation("权限缓存预热完成，耗时: {ElapsedMs}ms, 用户数: {UserCount}",
                stopwatch.ElapsedMilliseconds, prewarmStrategy.TotalUsers);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "权限缓存预热失败");
        }
    }

    private async Task<PrewarmStrategy> DeterminePrewarmStrategyAsync(CancellationToken cancellationToken)
    {
        var strategy = new PrewarmStrategy();

        // 1. 活跃用户优先
        var activeUsers = await _repository.GetActiveUsersAsync(
            TimeSpan.FromDays(_options.ActiveUserDays),
            cancellationToken);
        strategy.ActiveUsers = activeUsers.Take(_options.MaxActiveUsersToPrewarm).ToList();

        // 2. VIP用户
        var vipUsers = await _repository.GetVipUsersAsync(cancellationToken);
        strategy.VipUsers = vipUsers.ToList();

        // 3. 系统用户
        var systemUsers = await _repository.GetSystemUsersAsync(cancellationToken);
        strategy.SystemUsers = systemUsers.ToList();

        // 4. 基于缓存命中率选择用户
        var lowHitRateUsers = await GetLowHitRateUsersAsync(cancellationToken);
        strategy.LowHitRateUsers = lowHitRateUsers.Take(_options.MaxLowHitRateUsersToPrewarm).ToList();

        strategy.TotalUsers = strategy.ActiveUsers.Count +
                             strategy.VipUsers.Count +
                             strategy.SystemUsers.Count +
                             strategy.LowHitRateUsers.Count;

        _logger.LogDebug("预热策略确定: 活跃用户={Active}, VIP用户={Vip}, 系统用户={System}, 低命中率用户={LowHitRate}",
            strategy.ActiveUsers.Count, strategy.VipUsers.Count,
            strategy.SystemUsers.Count, strategy.LowHitRateUsers.Count);

        return strategy;
    }

    private async Task ExecutePrewarmStrategyAsync(PrewarmStrategy strategy, CancellationToken cancellationToken)
    {
        // 优先级队列执行预热
        var prewarmTasks = new List<Task>();

        // 1. 最高优先级：VIP用户
        prewarmTasks.AddRange(strategy.VipUsers.Select(user =>
            PrewarmUserWithPriorityAsync(user.TenantId, user.Id, Priority.Critical, cancellationToken)));

        // 2. 高优先级：系统用户
        prewarmTasks.AddRange(strategy.SystemUsers.Select(user =>
            PrewarmUserWithPriorityAsync(user.TenantId, user.Id, Priority.High, cancellationToken)));

        // 3. 中等优先级：活跃用户
        prewarmTasks.AddRange(strategy.ActiveUsers.Select(user =>
            PrewarmUserWithPriorityAsync(user.TenantId, user.Id, Priority.Normal, cancellationToken)));

        // 4. 低优先级：低命中率用户
        prewarmTasks.AddRange(strategy.LowHitRateUsers.Select(user =>
            PrewarmUserWithPriorityAsync(user.TenantId, user.Id, Priority.Low, cancellationToken)));

        // 分批执行，避免系统过载
        var batchSize = _options.PrewarmBatchSize;
        for (int i = 0; i < prewarmTasks.Count; i += batchSize)
        {
            var batch = prewarmTasks.Skip(i).Take(batchSize);
            await Task.WhenAll(batch);

            // 批次间休息，避免过度压力
            if (i + batchSize < prewarmTasks.Count)
            {
                await Task.Delay(_options.PrewarmBatchDelayMs, cancellationToken);
            }
        }
    }

    private async Task PrewarmUserWithPriorityAsync(
        Guid? tenantId,
        Guid userId,
        Priority priority,
        CancellationToken cancellationToken)
    {
        var delay = priority switch
        {
            Priority.Critical => 0,
            Priority.High => 50,
            Priority.Normal => 100,
            Priority.Low => 200,
            _ => 100
        };

        if (delay > 0)
        {
            await Task.Delay(delay, cancellationToken);
        }

        try
        {
            await _cacheEngine.GetUserPermissionsAsync(tenantId, userId, cancellationToken);
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "预热用户权限失败: TenantId={TenantId}, UserId={UserId}, Priority={Priority}",
                tenantId, userId, priority);
        }
    }

    private async Task<List<UserInfo>> GetLowHitRateUsersAsync(CancellationToken cancellationToken)
    {
        // 从缓存统计数据中找出命中率低的用户
        // 这里可以集成APM系统或自定义统计
        var cacheStatistics = await _repository.GetCacheStatisticsAsync(cancellationToken);

        return cacheStatistics
            .Where(stat => stat.HitRate < _options.LowHitRateThreshold)
            .OrderBy(stat => stat.HitRate)
            .Select(stat => new UserInfo { TenantId = stat.TenantId, Id = stat.UserId })
            .ToList();
    }

    public enum Priority
    {
        Critical = 1,
        High = 2,
        Normal = 3,
        Low = 4
    }

    private class PrewarmStrategy
    {
        public List<UserInfo> ActiveUsers { get; set; } = new();
        public List<UserInfo> VipUsers { get; set; } = new();
        public List<UserInfo> SystemUsers { get; set; } = new();
        public List<UserInfo> LowHitRateUsers { get; set; } = new();
        public int TotalUsers { get; set; }
    }
}
```

### 5.3 权限计算引擎优化

#### 5.3.1 权限继承算法实现

**高性能权限继承计算**:
```csharp
// 权限继承计算引擎
public class PermissionInheritanceEngine : IPermissionInheritanceEngine
{
    private readonly ILogger<PermissionInheritanceEngine> _logger;
    private readonly IMemoryCache _memoryCache;
    private readonly PermissionOptions _options;

    // 图结构缓存，用于快速计算继承关系
    private readonly ConcurrentDictionary<string, RoleHierarchyGraph> _hierarchyGraphs = new();

    public PermissionInheritanceEngine(
        ILogger<PermissionInheritanceEngine> logger,
        IMemoryCache memoryCache,
        IOptions<PermissionOptions> options)
    {
        _logger = logger;
        _memoryCache = memoryCache;
        _options = options.Value;
    }

    public async Task<List<InheritedPermission>> CalculateInheritedPermissionsAsync(
        Guid? tenantId,
        List<UserRole> userRoles,
        CancellationToken cancellationToken = default)
    {
        var stopwatch = Stopwatch.StartNew();

        try
        {
            // 获取角色层次图
            var hierarchyGraph = await GetRoleHierarchyGraphAsync(tenantId, cancellationToken);

            // 计算所有继承的角色
            var allInheritedRoles = CalculateAllInheritedRoles(userRoles, hierarchyGraph);

            // 获取所有角色的权限
            var rolePermissions = await GetRolePermissionsBatchAsync(
                tenantId,
                allInheritedRoles,
                cancellationToken);

            // 解决权限冲突并计算最终继承权限
            var inheritedPermissions = ResolvePermissionConflicts(
                userRoles,
                allInheritedRoles,
                rolePermissions,
                hierarchyGraph);

            stopwatch.Stop();
            _logger.LogDebug("权限继承计算完成，耗时: {ElapsedMs}ms, 角色数: {RoleCount}, 继承权限数: {PermissionCount}",
                stopwatch.ElapsedMilliseconds, userRoles.Count, inheritedPermissions.Count);

            return inheritedPermissions;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "权限继承计算失败: TenantId={TenantId}", tenantId);
            throw;
        }
    }

    private async Task<RoleHierarchyGraph> GetRoleHierarchyGraphAsync(
        Guid? tenantId,
        CancellationToken cancellationToken)
    {
        var cacheKey = $"role_hierarchy:{tenantId?.ToString() ?? "host"}";

        if (_hierarchyGraphs.TryGetValue(cacheKey, out var cachedGraph))
        {
            // 检查图的有效性
            if (cachedGraph.LastUpdated > DateTime.UtcNow.AddMinutes(-_options.HierarchyCacheExpirationMinutes))
            {
                return cachedGraph;
            }
        }

        // 重新构建角色层次图
        var graph = await BuildRoleHierarchyGraphAsync(tenantId, cancellationToken);
        _hierarchyGraphs[cacheKey] = graph;

        return graph;
    }

    private async Task<RoleHierarchyGraph> BuildRoleHierarchyGraphAsync(
        Guid? tenantId,
        CancellationToken cancellationToken)
    {
        var graph = new RoleHierarchyGraph(tenantId);

        // 获取所有角色和继承关系
        var roles = await GetAllRolesAsync(tenantId, cancellationToken);
        var inheritanceRelations = await GetRoleInheritanceRelationsAsync(tenantId, cancellationToken);

        // 构建图结构
        foreach (var role in roles)
        {
            graph.AddRole(role);
        }

        foreach (var relation in inheritanceRelations)
        {
            graph.AddInheritanceRelation(relation.ParentRoleId, relation.ChildRoleId);
        }

        // 检测循环依赖
        var cycles = graph.DetectCycles();
        if (cycles.Any())
        {
            _logger.LogWarning("检测到角色继承循环依赖: {Cycles}",
                string.Join(", ", cycles.Select(c => string.Join(" -> ", c))));

            // 根据配置决定是否抛出异常
            if (_options.ThrowOnCyclicInheritance)
            {
                throw new InvalidOperationException($"角色继承存在循环依赖: {string.Join(", ", cycles.First())}");
            }
        }

        // 预计算传递闭包，提高查询性能
        graph.BuildTransitiveClosure();

        graph.LastUpdated = DateTime.UtcNow;

        _logger.LogDebug("角色层次图构建完成: TenantId={TenantId}, 角色数={RoleCount}, 继承关系数={RelationCount}",
            tenantId, roles.Count, inheritanceRelations.Count);

        return graph;
    }

    private List<InheritedRole> CalculateAllInheritedRoles(
        List<UserRole> userRoles,
        RoleHierarchyGraph hierarchyGraph)
    {
        var allInheritedRoles = new List<InheritedRole>();
        var processedRoles = new HashSet<Guid>();

        foreach (var userRole in userRoles)
        {
            // 使用广度优先搜索遍历继承树
            var queue = new Queue<(Guid roleId, int depth, List<Guid> path)>();
            queue.Enqueue((userRole.RoleId, 0, new List<Guid> { userRole.RoleId }));

            var visitedInCurrentTree = new HashSet<Guid>();

            while (queue.Count > 0)
            {
                var (currentRoleId, depth, path) = queue.Dequeue();

                if (visitedInCurrentTree.Contains(currentRoleId))
                    continue;

                visitedInCurrentTree.Add(currentRoleId);

                // 添加当前角色（如果还没处理过）
                if (!processedRoles.Contains(currentRoleId))
                {
                    allInheritedRoles.Add(new InheritedRole
                    {
                        RoleId = currentRoleId,
                        SourceUserRole = userRole,
                        InheritanceDepth = depth,
                        InheritancePath = new List<Guid>(path),
                        IsDirectRole = depth == 0
                    });

                    processedRoles.Add(currentRoleId);
                }

                // 获取父角色并继续遍历
                if (depth < _options.MaxInheritanceDepth)
                {
                    var parentRoles = hierarchyGraph.GetParentRoles(currentRoleId);
                    foreach (var parentRoleId in parentRoles)
                    {
                        if (!visitedInCurrentTree.Contains(parentRoleId))
                        {
                            var newPath = new List<Guid>(path) { parentRoleId };
                            queue.Enqueue((parentRoleId, depth + 1, newPath));
                        }
                    }
                }
            }
        }

        return allInheritedRoles;
    }

    private List<InheritedPermission> ResolvePermissionConflicts(
        List<UserRole> userRoles,
        List<InheritedRole> allInheritedRoles,
        Dictionary<Guid, List<RolePermission>> rolePermissions,
        RoleHierarchyGraph hierarchyGraph)
    {
        var permissionMap = new Dictionary<string, List<PermissionCandidate>>();

        // 收集所有权限候选
        foreach (var inheritedRole in allInheritedRoles)
        {
            if (!rolePermissions.TryGetValue(inheritedRole.RoleId, out var permissions))
                continue;

            foreach (var permission in permissions)
            {
                var key = $"{permission.PermissionName}:{permission.Resource ?? "*"}";

                if (!permissionMap.ContainsKey(key))
                {
                    permissionMap[key] = new List<PermissionCandidate>();
                }

                permissionMap[key].Add(new PermissionCandidate
                {
                    Permission = permission,
                    InheritedRole = inheritedRole,
                    Priority = CalculatePermissionPriority(inheritedRole, hierarchyGraph)
                });
            }
        }

        // 解决冲突并选择最终权限
        var finalPermissions = new List<InheritedPermission>();

        foreach (var (permissionKey, candidates) in permissionMap)
        {
            var resolvedPermission = ResolvePermissionConflict(permissionKey, candidates);
            if (resolvedPermission != null)
            {
                finalPermissions.Add(resolvedPermission);
            }
        }

        return finalPermissions;
    }

    private InheritedPermission ResolvePermissionConflict(
        string permissionKey,
        List<PermissionCandidate> candidates)
    {
        // 权限冲突解决策略
        switch (_options.ConflictResolutionStrategy)
        {
            case ConflictResolutionStrategy.HighestPriority:
                return ResolveByHighestPriority(candidates);

            case ConflictResolutionStrategy.MostSpecific:
                return ResolveByMostSpecific(candidates);

            case ConflictResolutionStrategy.GrantWins:
                return ResolveByGrantWins(candidates);

            case ConflictResolutionStrategy.DenyWins:
                return ResolveByDenyWins(candidates);

            default:
                return ResolveByHighestPriority(candidates);
        }
    }

    private InheritedPermission ResolveByHighestPriority(List<PermissionCandidate> candidates)
    {
        var winner = candidates.OrderByDescending(c => c.Priority).First();

        return new InheritedPermission
        {
            PermissionName = winner.Permission.PermissionName,
            Resource = winner.Permission.Resource,
            IsGranted = winner.Permission.IsGranted,
            SourceRoleId = winner.InheritedRole.RoleId,
            InheritanceDepth = winner.InheritedRole.InheritanceDepth,
            InheritancePath = winner.InheritedRole.InheritancePath,
            ConflictResolutionMethod = ConflictResolutionStrategy.HighestPriority,
            CompetingCandidates = candidates.Count
        };
    }

    private InheritedPermission ResolveByMostSpecific(List<PermissionCandidate> candidates)
    {
        // 优先选择最具体的资源权限（非通配符）
        var specificCandidates = candidates.Where(c => c.Permission.Resource != "*" && !string.IsNullOrEmpty(c.Permission.Resource)).ToList();

        if (specificCandidates.Any())
        {
            var winner = specificCandidates.OrderByDescending(c => c.Priority).First();
            return CreateInheritedPermission(winner, ConflictResolutionStrategy.MostSpecific, candidates.Count);
        }

        // 如果没有具体资源权限，选择优先级最高的
        var winner2 = candidates.OrderByDescending(c => c.Priority).First();
        return CreateInheritedPermission(winner2, ConflictResolutionStrategy.MostSpecific, candidates.Count);
    }

    private InheritedPermission ResolveByGrantWins(List<PermissionCandidate> candidates)
    {
        // 如果有授权权限，优先选择授权
        var grantCandidates = candidates.Where(c => c.Permission.IsGranted).ToList();

        if (grantCandidates.Any())
        {
            var winner = grantCandidates.OrderByDescending(c => c.Priority).First();
            return CreateInheritedPermission(winner, ConflictResolutionStrategy.GrantWins, candidates.Count);
        }

        // 如果都是拒绝，选择优先级最高的
        var winner2 = candidates.OrderByDescending(c => c.Priority).First();
        return CreateInheritedPermission(winner2, ConflictResolutionStrategy.GrantWins, candidates.Count);
    }

    private InheritedPermission ResolveByDenyWins(List<PermissionCandidate> candidates)
    {
        // 如果有拒绝权限，优先选择拒绝
        var denyCandidates = candidates.Where(c => !c.Permission.IsGranted).ToList();

        if (denyCandidates.Any())
        {
            var winner = denyCandidates.OrderByDescending(c => c.Priority).First();
            return CreateInheritedPermission(winner, ConflictResolutionStrategy.DenyWins, candidates.Count);
        }

        // 如果都是授权，选择优先级最高的
        var winner2 = candidates.OrderByDescending(c => c.Priority).First();
        return CreateInheritedPermission(winner2, ConflictResolutionStrategy.DenyWins, candidates.Count);
    }

    private InheritedPermission CreateInheritedPermission(
        PermissionCandidate winner,
        ConflictResolutionStrategy method,
        int competingCount)
    {
        return new InheritedPermission
        {
            PermissionName = winner.Permission.PermissionName,
            Resource = winner.Permission.Resource,
            IsGranted = winner.Permission.IsGranted,
            SourceRoleId = winner.InheritedRole.RoleId,
            InheritanceDepth = winner.InheritedRole.InheritanceDepth,
            InheritancePath = winner.InheritedRole.InheritancePath,
            ConflictResolutionMethod = method,
            CompetingCandidates = competingCount
        };
    }

    private int CalculatePermissionPriority(InheritedRole inheritedRole, RoleHierarchyGraph hierarchyGraph)
    {
        // 优先级计算：直接角色 > 继承深度浅 > 角色优先级高
        var basePriority = inheritedRole.IsDirectRole ? 10000 : 0;
        var depthPenalty = inheritedRole.InheritanceDepth * 100;
        var rolePriority = hierarchyGraph.GetRolePriority(inheritedRole.RoleId);

        return basePriority - depthPenalty + rolePriority;
    }
```
