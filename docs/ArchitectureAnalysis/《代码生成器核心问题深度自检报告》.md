```csharp

{

    {
```

**诊断结果**: 🔴 **高危技术债务**

```csharp

// ✅ 修复后的安全生成流水线
public class SafeGenerationPipeline
{
    public async Task<GenerationResult> GenerateWithRecoveryAsync(ModuleMetadataDto metadata)
    {
        var result = new GenerationResult();
        var backup = await CreateBackupAsync();
        
        try
        {
            // 阶段1: 后端生成（可独立失败）
            var backendResult = await TryGenerateBackendAsync(metadata);
            result.BackendFiles = backendResult.Files;
            result.BackendSuccess = backendResult.Success;
            
            // 阶段2: 前端生成（可独立失败）
            var frontendResult = await TryGenerateFrontendAsync(metadata);
            result.FrontendFiles = frontendResult.Files;
            result.FrontendSuccess = frontendResult.Success;
            
            // 阶段3: 集成代码（可独立失败）
            var integrationResult = await TryGenerateIntegrationAsync(metadata);
            result.IntegrationFiles = integrationResult.Files;
            result.IntegrationSuccess = integrationResult.Success;
            
            // 至少一个阶段成功即为成功
            result.OverallSuccess = result.BackendSuccess || result.FrontendSuccess;
            
            if (!result.OverallSuccess)
            {
                await RestoreBackupAsync(backup);
            }
            
            return result;
        }
        catch (Exception ex)
        {
            await RestoreBackupAsync(backup);
            return GenerationResult.CompleteFailure(ex);
        }
    }

    private async Task<StageResult> TryGenerateBackendAsync(ModuleMetadataDto metadata)
    {
        try
        {
            var files = new Dictionary<string, string>();
            
            foreach (var entity in metadata.Entities)
            {
                // 原子文件操作
                var tempFile = Path.GetTempFileName();
                var entityCode = await GenerateEntityCodeAsync(entity);
                
                await File.WriteAllTextAsync(tempFile, entityCode);
                var targetPath = GetEntityFilePath(entity);
                
                // 原子移动操作
                if (File.Exists(targetPath))
                {
                    var backupPath = $"{targetPath}.backup.{DateTime.Now:yyyyMMddHHmmss}";
                    File.Move(targetPath, backupPath);
                }
                
                File.Move(tempFile, targetPath);
                files[targetPath] = entityCode;
            }
            
            return StageResult.Success(files);
        }
        catch (Exception ex)
        {
            return StageResult.Failed(ex.Message);
        }
    }
}
```
