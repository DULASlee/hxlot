// scripts/quality/check-page-completeness.ts
/**
 * 页面完整性检查脚本
 * 在Git提交前自动执行
 */
export async function checkPageCompleteness(filePath: string): Promise<CheckResult> {
  const fileContent = await fs.readFile(filePath, 'utf-8')
  
  const checks = [
    {
      name: '路由定义',
      check: () => checkRouteRegistration(filePath),
      error: '未在路由文件中注册！必须在src/router/modules/中定义'
    },
    {
      name: 'Layout使用',
      check: () => fileContent.includes('<MainLayout') || fileContent.includes('layout: MainLayout'),
      error: '未使用MainLayout！必须包裹在统一布局中'
    },
    {
      name: '菜单注册',
      check: () => checkMenuRegistration(filePath),
      error: '未在菜单配置中添加！必须在src/config/menu.ts中定义'
    },
    {
      name: '加载状态',
      check: () => fileContent.includes('loading') && fileContent.includes('v-loading'),
      error: '缺少loading处理！必须有加载状态显示'
    },
    {
      name: '错误处理',
      check: () => fileContent.includes('try') && fileContent.includes('catch'),
      error: '缺少错误处理！必须有try-catch或error边界'
    },
    {
      name: '空状态显示',
      check: () => fileContent.includes('empty') || fileContent.includes('v-if="data.length === 0"'),
      error: '缺少空状态显示！必须有empty提示'
    }
  ]
  
  const failures = checks.filter(c => !c.check())
  
  if (failures.length > 0) {
    return {
      success: false,
      errors: failures.map(f => f.error),
      message: '页面完整性检查失败！禁止提交！'
    }
  }
  
  return { success: true }
}
```

// scripts/quality/check-component-completeness.ts
export function checkComponentCompleteness(fileContent: string): CheckResult {
  const errors: string[] = []
  
  // 检查1: 下拉选择必须绑定change事件
  if (fileContent.includes('<el-select') && !fileContent.includes('@change')) {
    errors.push('❌ el-select缺少@change事件绑定！')
  }
  
  // 检查2: 禁止硬编码数据
  const hasHardcodedOptions = /<el-option\s+label="[^"]+"\s+value="[^"]+"\s*\/>/.test(fileContent)
  if (hasHardcodedOptions && !fileContent.includes(':options=')) {
    errors.push('❌ el-select使用硬编码数据！必须从API获取或使用:options绑定')
  }
  
  // 检查3: v-model必须绑定
  if (fileContent.includes('<el-input') && !fileContent.includes('v-model')) {
    errors.push('❌ el-input缺少v-model绑定！')
  }
  
  // 检查4: 表单必须有验证规则
  if (fileContent.includes('<el-form') && !fileContent.includes(':rules=')) {
    errors.push('❌ el-form缺少验证规则！')
  }
  
  return {
    success: errors.length === 0,
    errors
  }
}
```

// scripts/quality/check-api-implementation.ts
export function checkAPIImplementation(fileContent: string, filePath: string): CheckResult {
  const errors: string[] = []
  
  // 检查1: 禁止假API
  if (fileContent.includes('Promise.resolve({') && !fileContent.includes('// @mock')) {
    errors.push(`❌ ${filePath} 使用假数据！禁止Promise.resolve，必须调用http`)
  }
  
  // 检查2: 必须有http调用
  const hasHttpCall = /http\.(get|post|put|delete|patch)/.test(fileContent)
  if (fileContent.includes('export const') && !hasHttpCall && !fileContent.includes('// @mock')) {
    errors.push(`❌ ${filePath} API未实现！必须调用http`)
  }
  
  // 检查3: 必须有类型定义
  if (fileContent.includes('export const') && !fileContent.includes(': Promise<')) {
    errors.push(`❌ ${filePath} 缺少返回类型定义！`)
  }
  
  // 检查4: 必须有错误处理
  if (hasHttpCall && !fileContent.includes('try') && !fileContent.includes('.catch(')) {
    errors.push(`❌ ${filePath} 缺少错误处理！必须有try-catch或.catch()`)
  }
  
  return {
    success: errors.length === 0,
    errors
  }
}
```

/// <summary>
/// 后端API实现完整性检查

public class BackendImplementationChecker
{
    public CheckResult CheckAppService(string filePath, string fileContent)
    {
        var errors = new List<string>();
        
        // 检查1: 必须注入Repository
        if (!fileContent.Contains("IRepository<") && !fileContent.Contains("Repository"))
        {

            }

// scripts/quality/check-dto-consistency.ts
export async function checkDTOConsistency(): Promise<CheckResult> {
  const errors: string[] = []
  
  // 1. 读取UnifiedSchema
  const schemaFiles = await glob('src/schemas/**/*.json')
  
  for (const schemaFile of schemaFiles) {
    const schema = JSON.parse(await fs.readFile(schemaFile, 'utf-8'))
    const entityName = schema.name
    
    // 2. 检查前端DTO
    const frontendDtoPath = `src/types/${entityName}Dto.ts`
    if (!await fs.pathExists(frontendDtoPath)) {
      errors.push(`❌ 前端DTO缺失: ${frontendDtoPath}`)
      continue
    }
    
    const frontendDto = await fs.readFile(frontendDtoPath, 'utf-8')
    
    // 3. 检查后端DTO
    const backendDtoPath = `src/SmartAbp.Application.Contracts/${entityName}/${entityName}Dto.cs`
    if (!await fs.pathExists(backendDtoPath)) {
      errors.push(`❌ 后端DTO缺失: ${backendDtoPath}`)
      continue
    }
    
    const backendDto = await fs.readFile(backendDtoPath, 'utf-8')
    
    // 4. 字段一致性检查
    for (const field of schema.fields) {
      const frontendFieldName = toCamelCase(field.name)
      const backendFieldName = toPascalCase(field.name)
      
      if (!frontendDto.includes(frontendFieldName)) {
        errors.push(`❌ 前端DTO缺少字段: ${entityName}.${frontendFieldName}`)
      }
      
      if (!backendDto.includes(backendFieldName)) {
        errors.push(`❌ 后端DTO缺少字段: ${entityName}.${backendFieldName}`)
      }
    }
    
    // 5. 检查AutoMapper配置
    const mapperPath = `src/SmartAbp.Application/${entityName}/${entityName}AutoMapperProfile.cs`
    if (!await fs.pathExists(mapperPath)) {
      errors.push(`❌ AutoMapper配置缺失: ${mapperPath}`)
    }
  }
  
  return {
    success: errors.length === 0,
    errors
  }
}
```

// scripts/quality/check-code-duplication.ts
export async function checkCodeDuplication(): Promise<CheckResult> {
  const errors: string[] = []
  
  // 检查1: 重复组件
  const vueFiles = await glob('src/**/*.vue')
  const componentNames = vueFiles.map(f => path.basename(f))
  const duplicateComponents = findDuplicates(componentNames)
  
  if (duplicateComponents.length > 0) {
    errors.push(`❌ 发现重复组件: ${duplicateComponents.join(', ')}`)
  }
  
  // 检查2: 重复函数
  const tsFiles = await glob('src/**/*.ts')
  for (const file of tsFiles) {
    const content = await fs.readFile(file, 'utf-8')
    const functionNames = extractFunctionNames(content)
    
    // 检查是否存在相同功能的函数
    for (const funcName of functionNames) {
      const similarFuncs = await searchSimilarFunctions(funcName, tsFiles)
      if (similarFuncs.length > 1) {
        errors.push(`❌ 发现重复函数: ${funcName} 在 ${similarFuncs.length} 个文件中`)
      }
    }
  }
  
  // 检查3: CRUD未使用模板
  const crudFiles = await glob('src/**/*{List,Form,Detail}.vue')
  for (const file of crudFiles) {
    const content = await fs.readFile(file, 'utf-8')
    if (!content.includes('// @generated') && !content.includes('// @template')) {
      errors.push(`❌ CRUD页面未使用模板生成: ${file}，必须使用模板！`)
    }
  }
  
  return {
    success: errors.length === 0,
    errors
  }
}
