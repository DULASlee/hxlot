```csharp
// 基础告警服务抽象
public abstract class AlertServiceBase : DomainService
{
    protected readonly IAlertManager AlertManager;
    
    protected AlertServiceBase(IAlertManager alertManager)
    {
        AlertManager = alertManager;
    }
    
    protected virtual async Task<Alert> SendAlertAsync(
        AlertType type, 
        string message, 
        AlertLevel level,
        Dictionary<string, object> metadata = null)
    {
        var alertDto = new CreateAlertDto
        {
            Type = type,
            Message = message,
            Level = level,
            Source = GetType().Name,
            Metadata = metadata ?? new Dictionary<string, object>()
        };
        
        return await AlertManager.CreateAlertAsync(alertDto);
    }
}

// 权限告警服务 - 特化实现
public class PermissionAlertService : AlertServiceBase, IPermissionAlertService

    {
        var metadata = new Dictionary<string, object>
```

```csharp
// 统一缓存管理器
public interface ICacheManager
{
    Task<T> GetAsync<T>(string key, Func<Task<T>> factory = null);
    Task SetAsync<T>(string key, T value, TimeSpan? expiration = null);
    Task RemoveAsync(string key);
    Task RemoveByPatternAsync(string pattern);
}

// 权限缓存服务特化
public class PermissionCacheService : ICacheManager
{
    private readonly IDistributedCache _distributedCache;
    private readonly IMemoryCache _memoryCache;
    
    // L1+L2混合缓存实现
    public async Task<T> GetAsync<T>(string key, Func<Task<T>> factory = null)
    {
        // L1缓存检查
        if (_memoryCache.TryGetValue(key, out T cachedValue))
        {
            return cachedValue;
        }
        
        // L2缓存检查
        var distributedValue = await _distributedCache.GetStringAsync(key);
        if (!string.IsNullOrEmpty(distributedValue))
        {
            var deserializedValue = JsonSerializer.Deserialize<T>(distributedValue);
            _memoryCache.Set(key, deserializedValue, TimeSpan.FromMinutes(5));
            return deserializedValue;
        }
        
        // 缓存未命中，使用工厂方法生成
        if (factory != null)
        {
            var newValue = await factory();
            await SetAsync(key, newValue);
            return newValue;
        }
        
        return default(T);
    }
}
```

```csharp
// Tools/DtoRefactoringTool.cs
public class DtoRefactoringTool
{
    public async Task SplitDtosFileAsync(string sourceFilePath, string targetDirectory)
    {
        var sourceCode = await File.ReadAllTextAsync(sourceFilePath);
        var syntaxTree = CSharpSyntaxTree.ParseText(sourceCode);
        var root = syntaxTree.GetRoot();
        
        var classDeclarations = root.DescendantNodes()
            .OfType<ClassDeclarationSyntax>()
            .ToList();
        
        var groupedClasses = GroupClassesByDomain(classDeclarations);
        
        foreach (var group in groupedClasses)
        {
            await GenerateNewFileAsync(group.Key, group.Value, targetDirectory);
        }
    }
    
    private Dictionary<string, List<ClassDeclarationSyntax>> GroupClassesByDomain(
        List<ClassDeclarationSyntax> classes)
    {
        var groups = new Dictionary<string, List<ClassDeclarationSyntax>>();
        
        foreach (var cls in classes)
        {
            var domain = DetermineDomain(cls.Identifier.Text);
            if (!groups.ContainsKey(domain))
            {
                groups[domain] = new List<ClassDeclarationSyntax>();
            }
            groups[domain].Add(cls);
        }
        
        return groups;
    }
    
    private string DetermineDomain(string className)
    {
        // 智能分类逻辑
        if (className.Contains("Entity") || className.Contains("Property"))
            return "Entity";
        if (className.Contains("Module") || className.Contains("Metadata"))
            return "Module";
        if (className.Contains("Database") || className.Contains("Connection"))
            return "Database";
        if (className.Contains("Template") || className.Contains("Generation"))
            return "Template";
        if (className.Contains("Ddd") || className.Contains("Domain"))
            return "Architecture";
        if (className.Contains("Validation") || className.Contains("Issue"))
            return "Validation";
            
        return "Common";
    }
}
```

### 任务3.2：重复代码检测工具
```csharp
// Tools/DuplicateCodeDetector.cs
public class DuplicateCodeDetector
{
    public List<CodeDuplicate> DetectDuplicates(string projectPath)
    {
        var duplicates = new List<CodeDuplicate>();
        
        // 加载所有C#文件
        var csharpFiles = Directory.GetFiles(projectPath, "*.cs", SearchOption.AllDirectories);
        
        foreach (var file in csharpFiles)
        {
            var syntaxTree = CSharpSyntaxTree.ParseText(File.ReadAllText(file));
            var root = syntaxTree.GetRoot();
            
            // 检测重复的类定义
            var classes = root.DescendantNodes().OfType<ClassDeclarationSyntax>();
            duplicates.AddRange(FindDuplicateClasses(classes));
            
            // 检测重复的方法签名
            var methods = root.DescendantNodes().OfType<MethodDeclarationSyntax>();
            duplicates.AddRange(FindDuplicateMethods(methods));
        }
        
        return duplicates;
    }
    
    private List<CodeDuplicate> FindDuplicateClasses(
        IEnumerable<ClassDeclarationSyntax> classes)
    {
        var duplicates = new List<CodeDuplicate>();
        var classGroups = classes
            .GroupBy(c => CalculateClassSimilarity(c))
            .Where(g => g.Count() > 1);
            
        foreach (var group in classGroups)
        {
            duplicates.Add(new CodeDuplicate
            {
                Type = DuplicateType.Class,
                Items = group.Select(c => c.Identifier.Text).ToList(),
                SimilarityScore = group.Key
            });
        }
        
        return duplicates;
    }
}
```
