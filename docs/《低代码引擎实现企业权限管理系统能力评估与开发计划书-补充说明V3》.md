### 3.1 角色层次图生成器

#### 3.1.1 角色架构可视化插件

**🎯 目标**: 生成支持拖拽、编辑、继承关系展示的角色层次架构图

**角色层次可视化插件架构**:
```typescript
// 角色层次可视化插件 - 基于现有微内核架构
class RoleHierarchyVisualizerPlugin implements LowCodePlugin {
  metadata = {
    name: "RoleHierarchyVisualizerPlugin",
    version: "1.0.0",
    capabilities: ["role-hierarchy", "inheritance-visualization", "drag-drop-editing"],
    dependencies: ["vue3", "element-plus", "d3", "@antv/x6", "lodash-es"]
  }

  async canHandle(schema: any): Promise<boolean> {
    return schema.type === "role-hierarchy" &&
           schema.features?.includes("visualization") &&
           schema.displayMode === "interactive"
  }

  async generate(schema: RoleHierarchySchema, config: any, context: PluginContext): Promise<GeneratedCode> {
    const hierarchyViewer = this.generateRoleHierarchyComponent(schema)
    const roleEditor = this.generateRoleEditorComponent(schema)
    const inheritanceCalculator = this.generateInheritanceCalculator(schema)
    const hierarchyStore = this.generateHierarchyStore(schema)

    return {
      components: [hierarchyViewer, roleEditor],
      utils: [inheritanceCalculator],
      stores: [hierarchyStore],
      types: this.generateHierarchyTypes(schema),
      algorithms: this.generateLayoutAlgorithms(schema)
    }
  }
}
```

```vue
// 角色节点生成
const generateRoleNodes = (hierarchyTree: RoleHierarchyNode[]): RoleHierarchyNode[] => {
  const nodes: RoleHierarchyNode[] = []

  const calculateLayout = (node: RoleHierarchyNode, level: number, x: number, y: number) => {
    const nodeWithPosition = {
      ...node,
      x: x,
      y: y,
      level: level,
      isRoot: level === 0,
      isLeaf: !node.children || node.children.length === 0
    }

    nodes.push(nodeWithPosition)

    if (node.children) {
      const childWidth = canvasSize.value.width / node.children.length
      node.children.forEach((child, index) => {
        const childX = (index + 0.5) * childWidth
        const childY = y + 120
        calculateLayout(child, level + 1, childX, childY)
      })
    }
  }

  if (layoutAlgorithm.value === "hierarchical") {
    hierarchyTree.forEach((root, index) => {
      const rootX = (index + 0.5) * (canvasSize.value.width / hierarchyTree.length)
      calculateLayout(root, 0, rootX, 50)
    })
  }

  return nodes
}

// 连接线生成
const generateRoleConnections = (nodes: RoleHierarchyNode[], relations: any[]): RoleConnection[] => {
  const connections: RoleConnection[] = []

  relations.forEach(relation => {
    const sourceNode = nodes.find(n => n.id === relation.parentId)
    const targetNode = nodes.find(n => n.id === relation.childId)

    if (sourceNode && targetNode) {
      connections.push({
        source: relation.parentId,
        target: relation.childId,
        type: relation.type,
        path: generateConnectionPath(sourceNode, targetNode)
      })
    }
  })

  return connections
}

// 连接路径生成
const generateConnectionPath = (source: RoleHierarchyNode, target: RoleHierarchyNode): string => {
  const sourceX = source.x + nodeSize.value.width / 2
  const sourceY = source.y + nodeSize.value.height
  const targetX = target.x + nodeSize.value.width / 2
  const targetY = target.y

  const midY = (sourceY + targetY) / 2

  return `M ${sourceX} ${sourceY} C ${sourceX} ${midY} ${targetX} ${midY} ${targetX} ${targetY}`
}

// 角色操作方法

const getRoleNodeClass = (role: Role) => {
  const classes = []

  if (role.isActive) {
    classes.push("node-active")
  } else {
    classes.push("node-inactive")
  }

  switch (role.type) {
    case "system":
      classes.push("node-system")
      break
    case "business":
      classes.push("node-business")
      break
    case "custom":
      classes.push("node-custom")
      break
  }

  return classes
}

// 权限计算方法

const getPermissionSourceText = (permission: any): string => {
  switch (permission.source) {
    case "direct":
      return "直接授权"
    case "inherited":
      return `继承自: ${permission.inheritedFrom}`
    default:
      return "未知来源"
  }
}

const getPermissionSourceIcon = (permission: any): string => {
  switch (permission.source) {
    case "direct":
      return "🔐"
    case "inherited":
      return "⬇️"
    default:
      return "❓"
  }
}

// 角色路径计算
const getRolePathsToRoot = (role: Role): Role[][] => {
  const paths: Role[][] = []

  const findPaths = (currentRole: Role, currentPath: Role[]) => {
    const newPath = [currentRole, ...currentPath]

    if (!currentRole.parentRoles || currentRole.parentRoles.length === 0) {
      paths.push(newPath)
      return
    }

    currentRole.parentRoles.forEach(parent => {
      findPaths(parent, newPath)
    })
  }

  findPaths(role, [])
  return paths
}

// 可用父角色计算
const getAvailableParentRoles = (role: Role | null): Role[] => {
  if (!role) return roles.value

  // 排除自己和自己的子角色，防止循环继承
  const excludeIds = new Set([role.id])

  const addDescendants = (roleId: string) => {
    const descendants = roles.value.filter(r =>
      r.parentRoles?.some(p => p.id === roleId)
    )
    descendants.forEach(descendant => {
      excludeIds.add(descendant.id)
      addDescendants(descendant.id)
    })
  }

  addDescendants(role.id)

  return roles.value.filter(r => !excludeIds.has(r.id))
}
```

## 4. 数据权限拦截器实现方案

### 4.1 EF Core全局过滤器实现

#### 4.1.1 数据权限拦截器核心代码

**🎯 目标**: 实现动态SQL生成的数据权限拦截器，支持多租户、组织层次、角色权限

**数据权限拦截器实现**:
```csharp
// 数据权限拦截器 - 基于EF Core全局过滤器
public class DataPermissionInterceptor : DbCommandInterceptor
{
    private readonly ICurrentUser _currentUser;
    private readonly IDataPermissionProvider _dataPermissionProvider;
    private readonly ILogger<DataPermissionInterceptor> _logger;
    private readonly IMemoryCache _cache;
    private readonly DataPermissionOptions _options;

    public DataPermissionInterceptor(
        ICurrentUser currentUser,
        IDataPermissionProvider dataPermissionProvider,
        ILogger<DataPermissionInterceptor> logger,
        IMemoryCache cache,
        IOptions<DataPermissionOptions> options)
    {
        _currentUser = currentUser;
        _dataPermissionProvider = dataPermissionProvider;
        _logger = logger;
        _cache = cache;
        _options = options.Value;
    }

    public override InterceptionResult<DbDataReader> ReaderExecuting(
        DbCommand command,
        CommandEventData eventData,
        InterceptionResult<DbDataReader> result)
    {
        return ProcessCommand(command, eventData, result);
    }

    public override ValueTask<InterceptionResult<DbDataReader>> ReaderExecutingAsync(
        DbCommand command,
        CommandEventData eventData,
        InterceptionResult<DbDataReader> result,
        CancellationToken cancellationToken = default)
    {
        return new ValueTask<InterceptionResult<DbDataReader>>(
            ProcessCommand(command, eventData, result));
    }

    private InterceptionResult<DbDataReader> ProcessCommand<T>(
        DbCommand command,
        CommandEventData eventData,
        InterceptionResult<T> result)
    {
        try
        {
            // 跳过系统查询和管理员用户
            if (ShouldSkipInterception(command, eventData))
            {
                return (InterceptionResult<DbDataReader>)(object)result;
            }

            // 获取当前用户的数据权限
            var dataPermissions = GetUserDataPermissions();
            if (dataPermissions == null || !dataPermissions.Any())
            {
                return (InterceptionResult<DbDataReader>)(object)result;
            }

            // 解析原始SQL
            var sqlContext = ParseSqlCommand(command);
            if (sqlContext == null)
            {
                return (InterceptionResult<DbDataReader>)(object)result;
            }

            // 应用数据权限过滤
            var modifiedSql = ApplyDataPermissionFilters(sqlContext, dataPermissions);

            if (!string.IsNullOrEmpty(modifiedSql) && modifiedSql != command.CommandText)
            {
                _logger.LogDebug("原始SQL: {OriginalSql}", command.CommandText);
                _logger.LogDebug("修改后SQL: {ModifiedSql}", modifiedSql);

                command.CommandText = modifiedSql;
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "数据权限拦截处理失败");

            // 如果配置为严格模式，则抛出异常；否则记录错误并继续
            if (_options.StrictMode)
            {
                throw new DataPermissionException("数据权限验证失败", ex);
            }
        }

        return (InterceptionResult<DbDataReader>)(object)result;
    }

    private bool ShouldSkipInterception(DbCommand command, CommandEventData eventData)
    {
        // 跳过条件检查
        if (_currentUser?.Id == null)
            return true;

        // 跳过超级管理员
        if (_currentUser.IsInRole("SuperAdmin"))
            return true;

        // 跳过系统内部查询
        if (command.CommandText.Contains("__EFMigrationsHistory") ||
            command.CommandText.Contains("AbpSettings") ||
            command.CommandText.Contains("AbpUsers") && command.CommandText.Contains("UserName"))
            return true;

        // 跳过数据权限相关表的查询（避免循环）
        if (command.CommandText.Contains("AbpDataPermissions") ||
            command.CommandText.Contains("AbpOrganizationUnits"))
            return true;

        return false;
    }

    private List<DataPermissionRule> GetUserDataPermissions()
    {
        // 缓存键
        var cacheKey = $"DataPermissions:{_currentUser.TenantId}:{_currentUser.Id}";

        return _cache.GetOrCreate(cacheKey, factory =>
        {
            factory.SetAbsoluteExpiration(TimeSpan.FromMinutes(_options.CacheExpirationMinutes));

            return _dataPermissionProvider.GetUserDataPermissions(
                _currentUser.TenantId,
                _currentUser.Id);
        });
    }

    private SqlContext ParseSqlCommand(DbCommand command)
    {
        try
        {
            // 使用SQL解析器解析命令
            var parser = new SqlParser();
            return parser.Parse(command.CommandText);
        }
        catch (Exception ex)
        {
            _logger.LogWarning(ex, "SQL解析失败: {Sql}", command.CommandText);
            return null;
        }
    }

    private string ApplyDataPermissionFilters(SqlContext sqlContext, List<DataPermissionRule> permissions)
    {
        var sqlBuilder = new DataPermissionSqlBuilder(_logger);

        foreach (var table in sqlContext.Tables)
        {
            var entityType = GetEntityTypeFromTableName(table.Name);
            if (entityType == null) continue;

            var applicableRules = permissions.Where(p =>
                p.EntityType == entityType || p.EntityType == "*").ToList();

            if (!applicableRules.Any()) continue;

            foreach (var rule in applicableRules)
            {
                sqlBuilder.ApplyRule(table, rule, _currentUser);
            }
        }

        return sqlBuilder.BuildSql(sqlContext);
    }

    private Type GetEntityTypeFromTableName(string tableName)
    {
        // 根据表名获取实体类型
        var entityTypeMapping = _options.EntityTableMappings;
        return entityTypeMapping.TryGetValue(tableName, out var entityType) ? entityType : null;
    }
}
```

#### 4.1.2 数据权限SQL构建器

**动态SQL生成核心实现**:
```csharp
// 数据权限SQL构建器 - 动态生成WHERE条件
public class DataPermissionSqlBuilder
{
    private readonly ILogger _logger;
    private readonly List<DataPermissionCondition> _conditions;

    public DataPermissionSqlBuilder(ILogger logger)
    {
        _logger = logger;
        _conditions = new List<DataPermissionCondition>();
    }

    public void ApplyRule(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        switch (rule.ScopeType)
        {
            case DataScopeType.All:
                // 无需添加条件
                break;

            case DataScopeType.Self:
                ApplySelfDataScope(table, rule, currentUser);
                break;

            case DataScopeType.Department:
                ApplyDepartmentDataScope(table, rule, currentUser);
                break;

            case DataScopeType.Organization:
                ApplyOrganizationDataScope(table, rule, currentUser);
                break;

            case DataScopeType.Custom:
                ApplyCustomDataScope(table, rule, currentUser);
                break;

            case DataScopeType.Tenant:
                ApplyTenantDataScope(table, rule, currentUser);
                break;
        }
    }

    private void ApplySelfDataScope(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        // 仅能访问自己创建的数据
        var condition = new DataPermissionCondition
        {
            TableAlias = table.Alias,
            FieldName = rule.UserIdField ?? "CreatorId",
            Operator = "=",
            Value = currentUser.Id.ToString(),
            LogicalOperator = "AND"
        };

        _conditions.Add(condition);

        _logger.LogDebug("应用自身数据权限: {Table}.{Field} = {UserId}",
            table.Name, condition.FieldName, currentUser.Id);
    }

    private void ApplyDepartmentDataScope(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        // 可以访问本部门的数据
        var userDepartments = GetUserDepartments(currentUser.Id);
        if (!userDepartments.Any()) return;

        var condition = new DataPermissionCondition
        {
            TableAlias = table.Alias,
            FieldName = rule.DepartmentIdField ?? "DepartmentId",
            Operator = "IN",
            Value = $"({string.Join(",", userDepartments.Select(d => $"'{d}'"))})",
            LogicalOperator = "AND"
        };

        _conditions.Add(condition);

        _logger.LogDebug("应用部门数据权限: {Table}.{Field} IN ({Departments})",
            table.Name, condition.FieldName, string.Join(",", userDepartments));
    }

    private void ApplyOrganizationDataScope(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        // 可以访问本机构及子机构的数据
        var userOrganizations = GetUserOrganizationsWithChildren(currentUser.Id);
        if (!userOrganizations.Any()) return;

        // 生成层级查询条件
        var hierarchyConditions = new List<string>();

        foreach (var orgId in userOrganizations)
        {
            var hierarchyPath = GetOrganizationHierarchyPath(orgId);
            if (!string.IsNullOrEmpty(hierarchyPath))
            {
                hierarchyConditions.Add($"{table.Alias}.{rule.OrganizationIdField ?? "OrganizationId"} = '{orgId}'");
                hierarchyConditions.Add($"{table.Alias}.{rule.HierarchyPathField ?? "HierarchyPath"} LIKE '{hierarchyPath}%'");
            }
        }

        if (hierarchyConditions.Any())
        {
            var condition = new DataPermissionCondition
            {
                TableAlias = table.Alias,
                FieldName = "",
                Operator = "",
                Value = $"({string.Join(" OR ", hierarchyConditions)})",
                LogicalOperator = "AND",
                IsComplexCondition = true
            };

            _conditions.Add(condition);

            _logger.LogDebug("应用组织数据权限: {Condition}", condition.Value);
        }
    }

    private void ApplyCustomDataScope(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        // 自定义数据权限范围
        var customRules = GetUserCustomDataRules(currentUser.Id, rule.EntityType);

        foreach (var customRule in customRules)
        {
            var condition = BuildCustomCondition(table, customRule);
            if (condition != null)
            {
                _conditions.Add(condition);
            }
        }

        _logger.LogDebug("应用自定义数据权限: {RuleCount} 条规则", customRules.Count);
    }

    private void ApplyTenantDataScope(TableContext table, DataPermissionRule rule, ICurrentUser currentUser)
    {
        // 多租户数据隔离
        if (currentUser.TenantId.HasValue)
        {
            var condition = new DataPermissionCondition
            {
                TableAlias = table.Alias,
                FieldName = rule.TenantIdField ?? "TenantId",
                Operator = "=",
                Value = currentUser.TenantId.Value.ToString(),
                LogicalOperator = "AND"
            };

            _conditions.Add(condition);

            _logger.LogDebug("应用租户数据权限: {Table}.{Field} = {TenantId}",
                table.Name, condition.FieldName, currentUser.TenantId);
        }
    }

    private DataPermissionCondition BuildCustomCondition(TableContext table, CustomDataRule customRule)
    {
        // 根据自定义规则构建条件
        switch (customRule.RuleType)
        {
            case CustomRuleType.FieldEquals:
                return new DataPermissionCondition
                {
                    TableAlias = table.Alias,
                    FieldName = customRule.FieldName,
                    Operator = "=",
                    Value = $"'{customRule.Value}'",
                    LogicalOperator = "AND"
                };

            case CustomRuleType.FieldIn:
                var values = customRule.Values.Select(v => $"'{v}'");
                return new DataPermissionCondition
                {
                    TableAlias = table.Alias,
                    FieldName = customRule.FieldName,
                    Operator = "IN",
                    Value = $"({string.Join(",", values)})",
                    LogicalOperator = "AND"
                };

            case CustomRuleType.DateRange:
                return new DataPermissionCondition
                {
                    TableAlias = table.Alias,
                    FieldName = customRule.FieldName,
                    Operator = "BETWEEN",
                    Value = $"'{customRule.StartDate:yyyy-MM-dd}' AND '{customRule.EndDate:yyyy-MM-dd}'",
                    LogicalOperator = "AND"
                };

            case CustomRuleType.Expression:
                return new DataPermissionCondition
                {
                    TableAlias = table.Alias,
                    FieldName = "",
                    Operator = "",
                    Value = customRule.Expression,
                    LogicalOperator = "AND",
                    IsComplexCondition = true
                };

            default:
                return null;
        }
    }

    public string BuildSql(SqlContext sqlContext)
    {
        if (!_conditions.Any())
        {
            return sqlContext.OriginalSql;
        }

        // 构建WHERE条件
        var whereClause = BuildWhereClause();

        // 将条件插入到原始SQL中
        return InsertWhereClause(sqlContext.OriginalSql, whereClause, sqlContext);
    }

    private string BuildWhereClause()
    {
        var conditionStrings = new List<string>();

        foreach (var condition in _conditions)
        {
            string conditionSql;

            if (condition.IsComplexCondition)
            {
                conditionSql = condition.Value;
            }
            else
            {
                var fieldRef = string.IsNullOrEmpty(condition.TableAlias)
                    ? condition.FieldName
                    : $"{condition.TableAlias}.{condition.FieldName}";

                conditionSql = condition.Operator.ToUpper() switch
                {
                    "IN" => $"{fieldRef} IN {condition.Value}",
                    "BETWEEN" => $"{fieldRef} BETWEEN {condition.Value}",
                    "LIKE" => $"{fieldRef} LIKE {condition.Value}",
                    "IS NULL" => $"{fieldRef} IS NULL",
                    "IS NOT NULL" => $"{fieldRef} IS NOT NULL",
                    _ => $"{fieldRef} {condition.Operator} {condition.Value}"
                };
            }

            conditionStrings.Add(conditionSql);
        }

        return string.Join(" AND ", conditionStrings);
    }

    private string InsertWhereClause(string originalSql, string whereClause, SqlContext sqlContext)
    {
        // 使用正则表达式或SQL解析器插入WHERE条件
        var sqlUpper = originalSql.ToUpper();

        // 查找WHERE子句的位置
        var whereIndex = sqlUpper.IndexOf(" WHERE ");

        if (whereIndex > 0)
        {
            // 已存在WHERE子句，添加AND条件
            var insertPoint = whereIndex + " WHERE ".Length;
            return originalSql.Insert(insertPoint, $"({whereClause}) AND ");
        }
        else
        {
            // 不存在WHERE子句，添加新的WHERE
            var orderByIndex = sqlUpper.IndexOf(" ORDER BY ");
            var groupByIndex = sqlUpper.IndexOf(" GROUP BY ");
            var havingIndex = sqlUpper.IndexOf(" HAVING ");

            // 找到最早出现的子句位置
            var insertPoint = originalSql.Length;

            if (orderByIndex > 0) insertPoint = Math.Min(insertPoint, orderByIndex);
            if (groupByIndex > 0) insertPoint = Math.Min(insertPoint, groupByIndex);
            if (havingIndex > 0) insertPoint = Math.Min(insertPoint, havingIndex);

            return originalSql.Insert(insertPoint, $" WHERE {whereClause}");
        }
    }

    // 辅助方法
    private List<Guid> GetUserDepartments(Guid userId)
    {
        // 实现获取用户所属部门的逻辑
        // 这里可以通过依赖注入的服务来获取
        return new List<Guid>();
    }

    private List<Guid> GetUserOrganizationsWithChildren(Guid userId)
    {
        // 实现获取用户所属机构及其子机构的逻辑
        return new List<Guid>();
    }

    private string GetOrganizationHierarchyPath(Guid organizationId)
    {
        // 实现获取机构层级路径的逻辑
        return "";
    }

    private List<CustomDataRule> GetUserCustomDataRules(Guid userId, string entityType)
    {
        // 实现获取用户自定义数据规则的逻辑
        return new List<CustomDataRule>();
    }
}
```
