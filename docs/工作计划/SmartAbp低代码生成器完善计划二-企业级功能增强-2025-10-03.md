     generateWorkflowEngine(workflow: WorkflowDefinition): string {
       return `
   // 自动生成的工作流引擎
   public class ${workflow.name}WorkflowEngine : WorkflowEngine<${workflow.entityType}>
   {
     public override async Task<WorkflowResult> ExecuteAsync(
       ${workflow.entityType} entity, 
       string action, 
       ClaimsPrincipal user)
     {
       var currentState = entity.WorkflowState;
       var targetState = await GetTargetStateAsync(currentState, action);
       
       // 执行状态转换前的业务规则
       var preRuleResult = await ExecutePreTransitionRules(entity, currentState, targetState, user);
       if (!preRuleResult.Success)
       {
         return WorkflowResult.Failed(preRuleResult.ErrorMessage);
       }
       
       // 执行状态转换
       entity.WorkflowState = targetState;
       entity.LastTransitionTime = Clock.Now;
       entity.LastTransitionUser = user.GetUserId();
       
       // 执行状态转换后的业务规则
       await ExecutePostTransitionRules(entity, currentState, targetState, user);
       
       // 触发工作流事件
       await EventBus.PublishAsync(new WorkflowTransitionEvent
       {
         EntityId = entity.Id,
         EntityType = typeof(${workflow.entityType}).Name,
         FromState = currentState,
         ToState = targetState,
         Action = action,
         UserId = user.GetUserId(),
         TransitionTime = Clock.Now
       });
       
       return WorkflowResult.Success(targetState);
     }
     
     ${workflow.states.map(state => this.generateStateHandler(state)).join('\n')}
     
     ${workflow.transitions.map(transition => this.generateTransitionHandler(transition)).join('\n')}
   }`;
