**组合式函数 (useProductForm.ts)**:

```typescript
/**
 * 商品表单状态管理
 * @description 管理商品编辑表单的所有状态和业务逻辑
 * @example
 * ```typescript
 * const { form, handleSubmit } = useProductForm()
 * ```
 */
export function useProductForm() {
  const form = shallowRef({  // 🎯 使用shallowRef优化性能
    basicInfo: {
      name: '',
      categoryId: [],
      price: 0
    },
    detailInfo: {
      description: '',
      images: []
    },
    specifications: []
  })

  const { rules } = useProductValidation()  // 🎯 提取验证逻辑
  const { submitProduct } = useProductApi()  // 🎯 提取API逻辑
  
  const loading = ref(false)
  const formRef = ref<FormInstance>()

  /**
   * 提交表单
   * @description 验证表单并提交数据
   */
  const handleSubmit = async () => {
    if (!formRef.value) return
    
    try {
      loading.value = true
      await formRef.value.validate()
      await submitProduct(form.value)
      ElMessage.success('保存成功')
    } catch (error) {
      ElMessage.error('保存失败')
    } finally {
      loading.value = false
    }
  }

  /**
   * 重置表单  
   * @description 重置所有表单数据到初始状态
   */
  const handleReset = () => {
    formRef.value?.resetFields()
  }

  return {
    form,
    rules,
    loading,
    formRef,
    handleSubmit,
    handleReset
  }
}
```

<template>

    <el-form-item label="商品名称" prop="basicInfo.name">
      <el-input 
        :model-value="modelValue.name"
        @update:model-value="updateField('name', $event)"
        placeholder="请输入商品名称"
        :maxlength="100"
        show-word-limit
      />
    </el-form-item>
    
    <el-form-item label="商品分类" prop="basicInfo.categoryId">
      <CategorySelector
        :model-value="modelValue.categoryId"
        @update:model-value="updateField('categoryId', $event)"
      />
    </el-form-item>
    
    <el-form-item label="商品价格" prop="basicInfo.price">
      <PriceInput
        :model-value="modelValue.price"
        @update:model-value="updateField('price', $event)"
      />
    </el-form-item>
  </div>
</template>

<script setup lang="ts">
interface Props {
  modelValue: {
    name: string
    categoryId: number[]
    price: number
  }
}

interface Emits {
  (e: 'update:modelValue', value: Props['modelValue']): void
}

const props = defineProps<Props>()
const emit = defineEmits<Emits>()

/**
 * 更新字段值
 * @param field 字段名
 * @param value 新值
 */
const updateField = (field: keyof Props['modelValue'], value: any) => {
  emit('update:modelValue', {
    ...props.modelValue,
    [field]: value
  })
}
</script>

```typescript
/**
 * 金融表格数据处理钩子
 * @description 处理金融数据的加载、格式化和缓存
 */
export function useFinancialTableData() {
  const tableData = shallowRef<FinancialData[]>([])
  const tableColumns = computed(() => getFinancialTableColumns())
  
  // 🚀 优化：使用Web Worker处理大数据
  const loadData = async () => {
    const worker = new Worker('/workers/financial-data-processor.js')
    
    return new Promise((resolve, reject) => {
      worker.postMessage({ action: 'PROCESS_FINANCIAL_DATA' })
      
      worker.onmessage = ({ data }) => {
        if (data.success) {
          // 🎯 浅层响应式，避免深度监听大数据
          tableData.value = data.processedData
          resolve(data.processedData)
        } else {
          reject(new Error(data.error))
        }
      }
    })
  }
  
  // 🎯 格式化缓存，避免重复计算
  const formatCache = new Map<string, string>()
  const getCellFormat = (column: TableColumn) => {
    const cacheKey = `${column.key}_${column.type}`
    if (!formatCache.has(cacheKey)) {
      formatCache.set(cacheKey, generateCellFormatter(column))
    }
    return formatCache.get(cacheKey)!
  }
  
  return {
    tableData,
    tableColumns,
    loadData,
    getCellFormat
  }
}
