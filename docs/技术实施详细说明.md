class CustomChart {
  private ctx: CanvasRenderingContext2D
  
  constructor(canvas: HTMLCanvasElement) {
    this.ctx = canvas.getContext('2d')!
  }
  
  drawLineChart(data: number[], labels: string[]) {
    const { width, height } = this.ctx.canvas
    const padding = 40
    
    // 绘制坐标轴
    this.drawAxis(padding, width, height)
    
    // 绘制数据点和连线
    this.drawDataPoints(data, labels, padding, width, height)
    
    // 绘制标签
    this.drawLabels(labels, padding, width, height)
  }
  
  private drawDataPoints(data: number[], labels: string[], padding: number, width: number, height: number) {
    const maxValue = Math.max(...data)
    const stepX = (width - 2 * padding) / (data.length - 1)
    const stepY = (height - 2 * padding) / maxValue
    
    this.ctx.beginPath()
    this.ctx.strokeStyle = '#3b82f6'
    this.ctx.lineWidth = 2
    
    data.forEach((value, index) => {
      const x = padding + index * stepX
      const y = height - padding - value * stepY
      
      if (index === 0) {
        this.ctx.moveTo(x, y)
      } else {
        this.ctx.lineTo(x, y)
      }
      
      // 绘制数据点
      this.ctx.fillStyle = '#3b82f6'
      this.ctx.beginPath()
      this.ctx.arc(x, y, 4, 0, 2 * Math.PI)
      this.ctx.fill()
    })
    
    this.ctx.stroke()
  }
}

// 令牌刷新机制
public async Task<AuthResult> RefreshTokenAsync(string refreshToken)
{
    var storedToken = await _tokenRepository.GetAsync(refreshToken);
    if (storedToken == null || storedToken.IsExpired)
    {
        throw new UnauthorizedAccessException("Invalid refresh token");
    }

    var user = await _userManager.FindByIdAsync(storedToken.UserId);
    var newAccessToken = await GenerateAccessTokenAsync(user);
    var newRefreshToken = await GenerateRefreshTokenAsync(user);

    // 撤销旧的刷新令牌
    await _tokenRepository.RevokeAsync(refreshToken);
    
    return new AuthResult
    {
        AccessToken = newAccessToken,
        RefreshToken = newRefreshToken,
        ExpiresIn = 3600
    };
}

// 审计日志模型
public class AuditLog
{
    public string Id { get; set; } = Guid.NewGuid().ToString();
    public DateTime Timestamp { get; set; } = DateTime.UtcNow;
    public string UserId { get; set; }
    public string UserName { get; set; }
    public string Action { get; set; }
    public string Resource { get; set; }
    public object Parameters { get; set; }
    public string IpAddress { get; set; }
    public string UserAgent { get; set; }
    public bool Success { get; set; }
    public string ErrorMessage { get; set; }
    public Dictionary<string, object> AdditionalData { get; set; }
}

// 自动审计属性
[AttributeUsage(AttributeTargets.Method)]
public class AuditActionAttribute : Attribute, IActionFilter
{
    public string Action { get; }
    public string Resource { get; }

    public AuditActionAttribute(string action, string resource = null)
    {
        Action = action;
        Resource = resource;
    }

    public void OnActionExecuting(ActionExecutingContext context)
    {
        // 记录操作开始
        var auditLog = new AuditLog
        {
            UserId = context.HttpContext.User.FindFirst(ClaimTypes.NameIdentifier)?.Value,
            UserName = context.HttpContext.User.FindFirst(ClaimTypes.Name)?.Value,
            Action = Action,
            Resource = Resource ?? context.ActionDescriptor.DisplayName,
            Parameters = context.ActionArguments,
            IpAddress = context.HttpContext.Connection.RemoteIpAddress?.ToString(),
            UserAgent = context.HttpContext.Request.Headers["User-Agent"].ToString()
        };

        context.HttpContext.Items["AuditLog"] = auditLog;
    }

    public void OnActionExecuted(ActionExecutedContext context)
    {
        var auditLog = (AuditLog)context.HttpContext.Items["AuditLog"];
        if (auditLog != null)
        {
            auditLog.Success = context.Exception == null;
            auditLog.ErrorMessage = context.Exception?.Message;
            
            // 异步写入Elasticsearch
            _ = Task.Run(() => WriteToElasticsearchAsync(auditLog));
        }
    }

    private async Task WriteToElasticsearchAsync(AuditLog auditLog)
    {
        try
        {
            var indexName = $"audit-logs-{DateTime.UtcNow:yyyy-MM}";
            await _elasticClient.IndexAsync(auditLog, idx => idx.Index(indexName));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to write audit log to Elasticsearch");
        }
    }
}

// 性能数据收集器
class PerformanceCollector {
  private metrics: PerformanceMetrics = {}
  private observer: PerformanceObserver

  constructor() {
    this.setupPerformanceObserver()
    this.startMetricsCollection()
  }

  private setupPerformanceObserver() {
    this.observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        switch (entry.entryType) {
          case 'navigation':
            this.handleNavigationEntry(entry as PerformanceNavigationTiming)
            break
          case 'paint':
            this.handlePaintEntry(entry as PerformancePaintTiming)
            break
          case 'largest-contentful-paint':
            this.handleLCPEntry(entry as LargestContentfulPaint)
            break
        }
      }
    })

    this.observer.observe({ entryTypes: ['navigation', 'paint', 'largest-contentful-paint'] })
  }

  private handleNavigationEntry(entry: PerformanceNavigationTiming) {
    this.metrics.pageLoadTime = entry.loadEventEnd - entry.navigationStart
    this.metrics.domContentLoaded = entry.domContentLoadedEventEnd - entry.navigationStart
    this.metrics.firstByte = entry.responseStart - entry.navigationStart
    
    // 发送到监控服务
    this.sendMetrics('navigation', this.metrics)
  }

  private handleLCPEntry(entry: LargestContentfulPaint) {
    this.metrics.largestContentfulPaint = entry.startTime
    
    // LCP超过2.5秒告警
    if (entry.startTime > 2500) {
      this.sendAlert('performance', {
        metric: 'LCP',
        value: entry.startTime,
        threshold: 2500,
        severity: 'warning'
      })
    }
  }

  private sendMetrics(type: string, data: any) {
    fetch('/api/metrics/frontend', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ type, data, timestamp: Date.now() })
    })
  }
}

    public async Task InvokeAsync(HttpContext context)
    {
        var stopwatch = Stopwatch.StartNew();
        var startTime = DateTimeOffset.UtcNow;

        try
        {
            await _next(context);
        }
        finally
        {
            stopwatch.Stop();
            var duration = stopwatch.ElapsedMilliseconds;

            // 记录性能指标
            var metrics = new RequestMetrics
            {
                Path = context.Request.Path,
                Method = context.Request.Method,
                StatusCode = context.Response.StatusCode,
                Duration = duration,
                Timestamp = startTime,
                UserId = context.User.FindFirst(ClaimTypes.NameIdentifier)?.Value
            };

            await _metrics.RecordRequestAsync(metrics);

            // 慢请求告警 (>2秒)
            if (duration > 2000)
            {
                _logger.LogWarning("Slow request detected: {Path} took {Duration}ms", 
                    context.Request.Path, duration);
                
                await _metrics.SendAlertAsync(new PerformanceAlert
                {
                    Type = "SlowRequest",
                    Path = context.Request.Path.Value,
                    Duration = duration,
                    Threshold = 2000
                });
            }
        }
    }
}

// 3. 重构：优化代码结构
function formatNumber(num: number): string {
    const units = ['', 'K', 'M', 'B'];
    const unitIndex = Math.floor(Math.log10(Math.abs(num)) / 3);
    
    if (unitIndex === 0) return num.toString();
    
    const scaledNum = num / Math.pow(1000, unitIndex);
    return scaledNum.toFixed(1) + units[unitIndex];
}

public async Task<List<UserDto>> GetUsersAsync(UserSearchRequest request)
{
    var query = _userRepository.GetQueryable()
        .Where(u => !u.IsDeleted)
        .Include(u => u.Roles)
        .ThenInclude(r => r.Permissions)
        .AsNoTracking(); // 只读查询不需要跟踪

    if (!string.IsNullOrEmpty(request.Search))
    {
        query = query.Where(u => EF.Functions.Like(u.Username, $"%{request.Search}%"));
    }

    return await query
        .OrderBy(u => u.Username)
        .Skip(request.Skip)
        .Take(request.Take)
        .ProjectTo<UserDto>(_mapper.ConfigurationProvider) // 投影避免过度查询
        .ToListAsync();
}
