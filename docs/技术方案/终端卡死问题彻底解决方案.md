## 🚨 问题现状分析

### **卡死现象统计**
根据最近的操作记录，终端卡死问题表现为：

1. **Git操作超时** (频率: 70%)
   - `git push origin main` - 网络握手失败
   - `git pull origin main --rebase` - 长时间无响应
   - `bash scripts/git-safe-sync.sh` - 中途中断

2. **字符编码异常** (频率: 40%)
   - 命令中出现`[201~`、`\E[200~`等控制字符
   - 文件名乱码显示
   - 终端输出格式错乱

3. **复合命令失败** (频率: 60%)
   - 多个命令用`&&`连接时中途失败
   - 管道命令`|`处理异常
   - 长命令行超时

### **方案B: 分步执行策略 (核心解决方案)**

#### **B.1 Git操作分解**
```bash
# 创建智能Git管理脚本
cat > scripts/git-smart-sync.sh << 'EOF'
#!/bin/bash
set -e  # 出错立即退出

echo "=== SmartAbp 智能Git同步工具 ==="
echo "特点: 分步执行、断点续传、自动重试"

# 函数：安全执行Git命令
safe_git_cmd() {
    local cmd="$1"
    local max_retry=3
    local retry=0
    
    while [ $retry -lt $max_retry ]; do
        echo "执行: $cmd (尝试 $((retry+1))/$max_retry)"
        
        if timeout 60 $cmd; then
            echo "✅ 命令执行成功"
            return 0
        else
            retry=$((retry+1))
            if [ $retry -lt $max_retry ]; then
                echo "⚠️ 命令失败，等待5秒后重试..."
                sleep 5
            fi
        fi
    done
    
    echo "❌ 命令最终失败: $cmd"
    return 1
}

# 步骤1: 检查状态
echo "[1/6] 检查Git状态..."
git status --porcelain
if [ $? -ne 0 ]; then
    echo "❌ Git状态检查失败"
    exit 1
fi

# 步骤2: 暂存更改
echo "[2/6] 暂存本地更改..."
git add -A
if [ $? -eq 0 ]; then
    echo "✅ 文件暂存成功"
else
    echo "❌ 文件暂存失败"
    exit 1
fi

# 步骤3: 提交更改（如果有）
if ! git diff --cached --quiet; then
    echo "[3/6] 提交本地更改..."
    git commit -m "docs: 自动同步更新 - $(date '+%Y%m%d_%H%M%S')"
    if [ $? -eq 0 ]; then
        echo "✅ 本地提交成功"
    else
        echo "❌ 本地提交失败"
        exit 1
    fi
else
    echo "[3/6] 无需提交（无更改）"
fi

# 步骤4: 拉取远程更新
echo "[4/6] 拉取远程更新..."
safe_git_cmd "git fetch origin main"
safe_git_cmd "git pull origin main --rebase"

# 步骤5: 推送到远程
echo "[5/6] 推送到远程..."
safe_git_cmd "git push origin main"

# 步骤6: 验证同步
echo "[6/6] 验证同步状态..."
git status
echo "✅ Git智能同步完成"
EOF

chmod +x scripts/git-smart-sync.sh
```

#### **B.2 分步命令执行策略**
```bash
# 创建命令分解执行器
cat > scripts/command-executor.sh << 'EOF'
#!/bin/bash

# 函数：安全执行单个命令
execute_safe() {
    local cmd="$1"
    local timeout_sec="${2:-30}"  # 默认30秒超时
    local description="$3"
    
    echo "🔄 $description"
    echo "   命令: $cmd"
    echo "   超时: ${timeout_sec}秒"
    
    if timeout $timeout_sec bash -c "$cmd"; then
        echo "✅ 执行成功: $description"
        return 0
    else
        echo "❌ 执行失败: $description"
        echo "   错误码: $?"
        return 1
    fi
}

# 使用示例
execute_safe "git status" 10 "检查Git状态"
execute_safe "npm run type-check" 120 "TypeScript类型检查"
execute_safe "git push origin main" 300 "推送到远程仓库"
EOF

chmod +x scripts/command-executor.sh
```

### **方案D: 智能重试机制**

#### **D.1 创建重试装饰器**
```bash
cat > scripts/retry-decorator.sh << 'EOF'
#!/bin/bash

# 智能重试函数
retry_with_backoff() {
    local max_attempts=5
    local timeout=1
    local attempt=0
    local cmd="$*"

    while [ $attempt -lt $max_attempts ]; do
        echo "尝试 $((attempt + 1))/$max_attempts: $cmd"
        
        if $cmd; then
            echo "✅ 命令执行成功"
            return 0
        else
            echo "❌ 命令失败，等待 ${timeout} 秒..."
            sleep $timeout
            attempt=$((attempt + 1))
            timeout=$((timeout * 2))  # 指数退避
        fi
    done

    echo "🚨 命令最终失败: $cmd"
    return 1
}

# 使用示例
retry_with_backoff git push origin main
retry_with_backoff npm run type-check
EOF

chmod +x scripts/retry-decorator.sh
```
