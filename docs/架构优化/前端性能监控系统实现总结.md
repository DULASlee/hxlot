```typescript
class PerformanceMonitor {
  // 🔥 Core Web Vitals 监控
  private observePaintMetrics()              // FP, FCP
  private observeLargestContentfulPaint()    // LCP
  private observeFirstInputDelay()           // FID
  private observeCumulativeLayoutShift()     // CLS
  private observeLongTasks()                 // Long Tasks

  // 🔥 导航时间分析
  private measureNavigationTiming()          // DNS, TCP, TTFB, DOM, Load
  private measureResourceTiming()            // 资源加载分析

  // 🔥 路由性能追踪
  public recordRoutePerformance(path, duration)

  // 🔥 数据获取
  public getMetrics(): PerformanceMetrics
  public getCoreWebVitals(): CoreWebVitals
}

export const performanceMonitor = new PerformanceMonitor()
```

#### CWV评分算法

```typescript
const getScore = (value: number, good: number, needsImprovement: number) => {
  if (value <= good) return 100
  if (value <= needsImprovement) return 70
  return 40
}

const getRating = (lcp: number, fid: number, cls: number): Rating => {
  const isLCPGood = lcp <= 2500
  const isFIDGood = fid <= 100
  const isCLSGood = cls <= 0.1

  if (isLCPGood && isFIDGood && isCLSGood) return 'Good'
  if (lcp <= 4000 && fid <= 300 && cls <= 0.25) return 'Needs Improvement'
  return 'Poor'
}
```

### 2. router/index.ts 集成

```typescript
// src/SmartAbp.Vue/src/router/index.ts

import { performanceMonitor } from "@/utils/performance/monitor"

// 🔥 路由性能追踪
router.afterEach((to, from) => {
  const endTime = performance.now()
  const startTime = (to.meta as any)._startTime || endTime
  const duration = endTime - startTime

  // 记录路由切换性能
  performanceMonitor.recordRoutePerformance(to.path, duration)
  
  console.log(`[Router Performance] ${from.path} → ${to.path}: ${duration.toFixed(2)}ms`)
})
```
