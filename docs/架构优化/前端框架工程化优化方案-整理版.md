export const useResourceManager = () => {
  const resources = new Set<() => void>()
  
  const registerCleanup = (cleanup: () => void) => {
    resources.add(cleanup)
    return () => resources.delete(cleanup)

    resources.clear()
  })
  
  return { registerCleanup }
}
