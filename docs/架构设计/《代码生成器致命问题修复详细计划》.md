// ✅ 我可以独立完成基础检测
public class CircularReferenceDetector
{
    public CircularReferenceResult DetectCircularReferences(List<EnhancedEntityModelDto> entities)
    {
        var result = new CircularReferenceResult();
        var graph = BuildDependencyGraph(entities);
        
        foreach (var entity in entities)
        {
            var visited = new HashSet<string>();
            var stack = new HashSet<string>();
            
            if (HasCycle(entity.Name, graph, visited, stack))
            {
                result.HasCircularReference = true;
                result.CircularPaths.Add(ReconstructCycle(entity.Name, graph, stack));
            }
        }
        
        return result;
    }
    
    private bool HasCycle(string entityName, Dictionary<string, List<string>> graph, 
        HashSet<string> visited, HashSet<string> stack)
    {
        if (stack.Contains(entityName)) return true;
        if (visited.Contains(entityName)) return false;
        
        visited.Add(entityName);
        stack.Add(entityName);
        
        if (graph.ContainsKey(entityName))
        {
            foreach (var dependency in graph[entityName])
            {
                if (HasCycle(dependency, graph, visited, stack))
                    return true;
            }
        }
        
        stack.Remove(entityName);
        return false;
    }
}

public class AtomicFileWriter
{
    public async Task<FileWriteResult> WriteFileAtomicallyAsync(string targetPath, string content)
    {
        var tempFile = Path.GetTempFileName();
        var backupFile = $"{targetPath}.backup.{DateTime.Now:yyyyMMddHHmmss}";
        
        try
        {
            // 1. 写入临时文件
            await File.WriteAllTextAsync(tempFile, content);
            
            // 2. 验证生成的代码语法
            var syntaxValid = await ValidateCodeSyntaxAsync(content, targetPath);
            if (!syntaxValid.IsValid)
            {
                return FileWriteResult.Failed($"生成的代码语法错误: {syntaxValid.Errors}");
            }
            
            // 3. 备份原文件（如果存在）
            if (File.Exists(targetPath))
            {
                File.Copy(targetPath, backupFile);
            }
            
            // 4. 原子移动操作
            File.Move(tempFile, targetPath);
            
            return FileWriteResult.Success(targetPath, backupFile);
        }
        catch (Exception ex)
        {
            // 清理临时文件
            if (File.Exists(tempFile))
                File.Delete(tempFile);
                
            return FileWriteResult.Failed($"文件写入失败: {ex.Message}");
        }
    }
}
