```csharp
    {
        options.SwaggerDoc("v1", new OpenApiInfo
        {
            Title = "SmartAbp API",
            Version = "v1",
            Description = "SmartAbp 低代码引擎 API"
        });

        // 包含XML注释
        var xmlFile = $"{Assembly.GetExecutingAssembly().GetName().Name}.xml";
        var xmlPath = Path.Combine(AppContext.BaseDirectory, xmlFile);
        options.IncludeXmlComments(xmlPath);

        // ABP约定
        options.CustomSchemaIds(type => type.FullName);
        options.DocInclusionPredicate((docName, description) => true);
    });
}

public override void OnApplicationInitialization(ApplicationInitializationContext context)
{
    var app = context.GetApplicationBuilder();

    // 启用Swagger中间件
    app.UseSwagger();
    app.UseSwaggerUI(options =>
    {
        options.SwaggerEndpoint("/swagger/v1/swagger.json", "SmartAbp API v1");
    });
}
```

```csharp
// SmartAbp.CodeGenerator/Core/Templates/HandlebarsTemplateService.cs

using HandlebarsDotNet;

public class HandlebarsTemplateService : ITemplateService
{
    private readonly IHandlebars _handlebars;
    private readonly ILogger<HandlebarsTemplateService> _logger;

    public HandlebarsTemplateService(ILogger<HandlebarsTemplateService> logger)
    {
        _logger = logger;
        _handlebars = Handlebars.Create(new HandlebarsConfiguration
        {
            ThrowOnUnresolvedBindingExpression = true
        });

        RegisterHelpers();
    }

    private void RegisterHelpers()
    {
        // 注册自定义Helper
        _handlebars.RegisterHelper("camelCase", (writer, context, parameters) =>
        {
            var value = parameters[0]?.ToString();
            if (!string.IsNullOrEmpty(value))
            {
                writer.WriteSafeString(char.ToLowerInvariant(value[0]) + value.Substring(1));
            }
        });

        _handlebars.RegisterHelper("pascalCase", (writer, context, parameters) =>
        {
            var value = parameters[0]?.ToString();
            if (!string.IsNullOrEmpty(value))
            {
                writer.WriteSafeString(char.ToUpperInvariant(value[0]) + value.Substring(1));
            }
        });
    }

    public string Render(string templateContent, object model)
    {
        try
        {
            var template = _handlebars.Compile(templateContent);
            return template(model);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "模板渲染失败");
            throw;
        }
    }
}
```

### 4. uni-app适配器

#### 4.1 适配器实现

```typescript
// packages/lowcode-core/src/adapters/uniapp-adapter.ts

import Handlebars from 'handlebars'
import type { UnifiedModuleMetadata, UnifiedEntityDefinition } from '@smartabp/lowcode-shared'

export class UniAppAdapter {
  private templates: Map<string, HandlebarsTemplateDelegate> = new Map()

  constructor() {
    this.registerTemplates()
  }

  private registerTemplates() {
    // 列表页模板
    const listPageTemplate = `
<template>
  <view class="page">
    <uni-search-bar v-model="keyword" @confirm="handleSearch" />
    <scroll-view scroll-y class="list">
      <view v-for="item in list" :key="item.id" @click="handleDetail(item.id)">
        {{#each fields}}
        <text>{{displayName}}: \{{ item.{{name}} }}</text>
        {{/each}}
      </view>
    </scroll-view>
  </view>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { {{entityName}}Api } from '@/api/{{moduleName}}'

const list = ref([])
const keyword = ref('')

const loadData = async () => {
  const api = new {{entityName}}Api()
  const result = await api.getList({ keyword: keyword.value })
  list.value = result.items
}

const handleDetail = (id: string) => {
  uni.navigateTo({ url: \`/pages/{{moduleName}}/detail?id=\${id}\` })
}
</script>
    `

    this.templates.set('listPage', Handlebars.compile(listPageTemplate))
  }

  public generatePages(metadata: UnifiedModuleMetadata) {
    return metadata.entities.map(entity => ({
      entity: entity.name,
      files: {
        listPage: this.templates.get('listPage')!({
          entityName: entity.name,
          moduleName: metadata.name.toLowerCase(),
          fields: entity.properties.filter(p => p.listVisible).slice(0, 5)
        })
      }
    }))
  }
}
```
