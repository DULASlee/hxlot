## 🧪 测试体系

### 1. 单元测试（Jest）

```typescript
// packages/lowcode-core/src/adapters/__tests__/uniapp-adapter.spec.ts

import { describe, it, expect } from '@jest/globals'
import { UniAppAdapter } from '../uniapp-adapter'
import type { UnifiedModuleMetadata } from '@smartabp/lowcode-shared'

describe('UniAppAdapter', () => {
  it('应该正确生成uni-app列表页', () => {
    const adapter = new UniAppAdapter()
    const metadata: UnifiedModuleMetadata = {
      id: '1',
      name: 'Project',
      displayName: '项目管理',
      entities: [{
        id: '1',
        name: 'Project',
        displayName: '项目',
        properties: [
          { name: 'name', displayName: '项目名称', type: 'string', listVisible: true },
          { name: 'code', displayName: '项目编号', type: 'string', listVisible: true }
        ]
      }]
    }

    const result = adapter.generatePages(metadata)

    expect(result).toHaveLength(1)
    expect(result[0].entity).toBe('Project')
    expect(result[0].files.listPage).toContain('<template>')
    expect(result[0].files.listPage).toContain('ProjectApi')
  })

  it('应该正确映射字段类型', () => {
    const adapter = new UniAppAdapter()

    expect(adapter.mapFieldType('string')).toBe('uni-easyinput')
    expect(adapter.mapFieldType('int')).toBe('uni-number-box')
    expect(adapter.mapFieldType('datetime')).toBe('uni-datetime-picker')
  })
})
```

### 3. E2E测试（Playwright）

// e2e/code-generation.spec.ts

import { test, expect } from '@playwright/test'

test('完整的代码生成流程', async ({ page }) => {
  // 1. 打开极简代码生成器
  await page.goto('http://localhost:9001/CodeGen/ultra-simple')

  // 2. 填写基本信息
  await page.fill('[placeholder="请输入系统名称"]', 'TestSystem')
  await page.fill('[placeholder="请输入模块名称"]', 'TestModule')

  // 3. 选择数据库表
  await page.click('el-select:has-text("选择数据库表")')
  await page.click('el-option:has-text("Users")')

  // 4. 点击生成按钮
  await page.click('button:has-text("一键生成全栈代码")')

  // 5. 等待生成完成
  await page.waitForSelector('.generation-success', { timeout: 30000 })

  // 6. 验证生成的代码

  expect(codePreview).toContain('<template>')
})

// scripts/performance-test.ts

async function performanceTest() {
  console.log('开始性能测试...')

  // 测试1: 单实体生成
  const start1 = Date.now()
  await generateSingleEntity()
  const time1 = Date.now() - start1
  console.log(`单实体生成: ${time1}ms ${time1 < 2000 ? '✅' : '❌'}`)

  // 测试2: 10实体批量生成
  const start2 = Date.now()
  await generate10Entities()
  const time2 = Date.now() - start2
  console.log(`10实体生成: ${time2}ms ${time2 < 10000 ? '✅' : '❌'}`)

  // 测试3: 并发测试
  const start3 = Date.now()
  await Promise.all(Array(10).fill(0).map(() => generateSingleEntity()))
  const time3 = Date.now() - start3
  console.log(`10并发生成: ${time3}ms ${time3 < 30000 ? '✅' : '❌'}`)
