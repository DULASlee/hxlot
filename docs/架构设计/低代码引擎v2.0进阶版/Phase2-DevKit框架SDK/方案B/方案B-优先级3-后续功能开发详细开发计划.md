## 第1章：项目概述

### 1.1 功能范围

**Layer1 - UltraSimpleStudio增强**：
- 极简实体配置表单（5个字段）
- 一键生成CRUD代码
- 零前端配置（使用默认模板）

**Layer2 - SmartStudio Lite**：
- 可视化字段配置表格
- 字段类型选择（10种常用类型）
- 字段验证规则配置
- 前端控件类型选择

**Layer3 - Studio Pro**：
- 完整实体设计器
- 关系建模（1对多、多对多）
- 高级验证规则
- 自定义模板选择

### 1.2 开发优先级

**Phase 1（Week 1-2）**: Layer1增强
**Phase 2（Week 3-4）**: Layer2框架 + 字段配置表格
**Phase 3（Week 5-6）**: Layer3完整功能

```vue
<template>
  <div class="ultra-simple-studio">
    <el-card>
      <template #header>
        <div class="card-header">
          <h2>🎯 Layer1: UltraSimple Studio</h2>
          <el-tag type="success">零代码配置</el-tag>
        </div>
      </template>

      <!-- 配置表单 -->
      <el-form
        ref="formRef"
        :model="form"
        :rules="rules"
        label-width="120px"
        size="large"
      >
        <el-form-item label="实体名称" prop="name" required>

<script setup lang="ts">
import { ref, computed } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { View } from '@element-plus/icons-vue'
import { codeGenerationApi } from '@/api/lowcode/code-generation'

const router = useRouter()
const formRef = ref()
const form = ref({
  name: '',
  displayName: '',
  tablePrefix: 'App',
  tableName: '',
  description: ''
})

const rules = {
  name: [
    { required: true, message: '请输入实体名称', trigger: 'blur' },
    { pattern: /^[A-Z][a-zA-Z0-9]*$/, message: '必须以大写字母开头', trigger: 'blur' }
  ],
  displayName: [
    { required: true, message: '请输入显示名称', trigger: 'blur' }
  ]
}

const showPreview = ref(false)
const generating = ref(false)
const showProgress = ref(false)
const progress = ref(0)
const progressStatus = ref<'success' | 'exception' | ''>('')
const progressMessage = ref('')
const progressSteps = ref<Array<{ name: string; status: string; time: string }>>([])

// 计算表名
const computedTableName = computed(() => {
  const prefix = form.value.tablePrefix || 'App'
  const name = form.value.name
  return name ? `${prefix}${name}s` : ''
})

// 自动填充字段
const autoFillFields = () => {
  if (form.value.name && !form.value.displayName) {
    // 可以添加自动翻译逻辑
  }
  form.value.tableName = computedTableName.value
}

// 表单验证状态
const isFormValid = computed(() => {
  return form.value.name && form.value.displayName
})

// 预览文件列表
const previewFiles = computed(() => {
  const name = form.value.name
  if (!name) return []

  return [
    `Domain/${name}.cs`,
    `Application/${name}AppService.cs`,
    `Application.Contracts/Dtos/${name}Dto.cs`,
    `HttpApi/Controllers/${name}Controller.cs`,
    `Vue/views/${name.toLowerCase()}/${name}List.vue`,
    `Vue/views/${name.toLowerCase()}/${name}Form.vue`,
    `数据库迁移文件`
  ]
})

// 预览生成
const handlePreview = async () => {
  const valid = await formRef.value.validate()
  if (!valid) return

  showPreview.value = true
  ElMessage.success('预览生成成功，请确认后点击"确认生成"')
}

// 确认生成
const handleGenerate = async () => {
  generating.value = true
  showProgress.value = true
  progress.value = 0
  progressSteps.value = []

  try {
    // 调用异步生成API
    const result = await codeGenerationApi.generateAsync({
      entityName: form.value.name,
      displayName: form.value.displayName,
      tablePrefix: form.value.tablePrefix,
      description: form.value.description
    })

    // 监听SignalR进度推送
    listenProgress(result.taskId)
  } catch (error) {
    ElMessage.error('生成失败')
    generating.value = false
    showProgress.value = false
  }
}

// 监听进度（SignalR）
const listenProgress = (taskId: string) => {
  // 模拟进度更新（实际应使用SignalR）
  const interval = setInterval(() => {
    progress.value += 10
    progressMessage.value = `正在生成代码... ${progress.value}%`

    if (progress.value >= 100) {
      clearInterval(interval)
      progressStatus.value = 'success'
      progressMessage.value = '代码生成完成！'

      setTimeout(() => {
        showProgress.value = false
        generating.value = false
        router.push('/lowcode/modules')
      }, 2000)
    }
  }, 500)
}

// 重置表单
const resetForm = () => {
  formRef.value.resetFields()
  showPreview.value = false
}
</script>
```
