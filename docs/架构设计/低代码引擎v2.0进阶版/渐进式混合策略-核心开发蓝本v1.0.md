## 🔍 CI/CD自动检查

### 检查脚本：check-ai-constraints.sh

```bash
#!/bin/bash
set -e

echo "🔍 检查AI约束规则..."

# 检查1: types.ts是否被手动修改
if git diff HEAD src/SmartAbp.Vue/src/api/generated/types.ts 2>/dev/null | grep -v "^+++" | grep "^+"; then
  echo "❌ 错误: types.ts被手动修改（NSwag生成，只读）"
  echo "正确做法: 修改后端C# DTO → 运行NSwag重新生成"
  exit 1
fi

# 检查2: 禁止手动定义DTO类型
if find src/SmartAbp.Vue/src -name "*.ts" -not -path "*/generated/*" -exec grep -l "export interface.*Dto" {} \; | grep -v node_modules; then
  echo "❌ 错误: 发现手动定义的DTO类型"
  echo "正确做法: import type { ...Dto } from '@/api/generated/types'"
  exit 1
fi

# 检查3: unified-schema.ts必须re-export
if ! grep -q "export \* from '@/api/generated/types'" src/SmartAbp.Vue/packages/lowcode-shared/src/types/unified-schema.ts 2>/dev/null; then
  echo "❌ 错误: unified-schema.ts必须re-export types.ts"
  exit 1
fi

# 检查4: 禁止其他文件re-export NSwag类型
if find src/SmartAbp.Vue/src -name "*.ts" -not -path "*/unified-schema.ts" -not -path "*/generated/*" -exec grep -l "export \* from '@/api/generated/types'" {} \; | grep -v node_modules; then
  echo "❌ 错误: 只有unified-schema.ts可以re-export NSwag类型"
  exit 1
fi

echo "✅ AI约束检查全部通过"
EOF

chmod +x .cursor/rules/check-ai-constraints.sh
```
