const saveToDatabase = async () => {
  const config = (config as any).value as MetadataConfig;
  const selectedTableData = availableTables.value.find(t => t.name === selectedTable.value);

  // 1. 保存模块
  const module = await lowCodeApi.createModule({
    systemName: config.systemName,
    moduleName: config.moduleName,
    displayName: config.displayName,
    namespace: derivedNamespace.value,
    version: '1.0.0',
    architectureConfig: {
      pattern: config.architecturePattern,
      databaseProvider: config.databaseProvider,
      connectionString: 'Default',
      schema: selectedTableData?.schema || 'dbo'
    },
    frontendConfig: {
      routePrefix: derivedRoutePrefix.value,
      parentMenuId: config.parentMenuId,
      menuIcon: config.menuIcon,
      menuOrder: 10
    },
    codeGenOptions: {
      generateBackend: true,
      generateFrontend: true,
      generateDatabase: true,
      generateTests: false,
      useAutoMapper: true,
      generateSwagger: true
    }
  });

  // 2. 保存实体
  const entity = await lowCodeApi.createEntity({
    moduleId: module.id,
    name: selectedTable.value,
    displayName: selectedTable.value,
    tableName: selectedTable.value,
    schema: selectedTableData?.schema || 'dbo',
    entityConfig: {
      isAggregateRoot: true,
      baseClass: 'AuditedAggregateRoot',
      interfaces: ['ISoftDelete'],
      isAudited: true,
      isSoftDelete: true,
      isMultiTenant: false
    },
    uiConfig: {
      icon: config.menuIcon,
      color: '#409EFF',
      listPageSize: 20,
      enableExport: true,
      enableImport: true,
      enableBatchDelete: true
    }
  });

  // 3. 保存属性（含UIConfig）
  const columns = selectedTableData?.schema?.columns || [];
  for (const col of columns) {
    await lowCodeApi.createProperty({
      entityId: entity.id,
      name: col.name,
      displayName: col.name,
      type: col.dataType || 'string',
      isNullable: col.isNullable,
      columnName: col.name,
      columnType: col.dataType || 'NVARCHAR(100)',
      isKey: col.isPrimaryKey,
      isRequired: !col.isNullable,
      maxLength: col.maxLength,
      // ⭐ 核心：UIConfig JSON
      uiConfig: {
        listVisible: true,
        formVisible: true,
        detailVisible: true,
        searchable: true,
        sortable: true,
        filterable: true,
        controlType: getControlType(col.dataType),
        controlProps: {
          placeholder: `请输入${col.name}`,
          clearable: true
        },
        list: {
          width: 150,
          align: 'left'
        },
        form: {
          col: 12,
          required: !col.isNullable
        }
      },
      // ⭐ 验证规则
      validationRules: !col.isNullable ? [{
        type: 'required',
        message: `${col.name}不能为空`
      }] : []
    });
  }

  // 4. 保存页面配置（含PageConfig）
  await lowCodeApi.createPageConfig({
    entityId: entity.id,
    name: `${selectedTable.value}Form`,
    displayName: `${selectedTable.value}表单`,
    pageType: 'form',
    // ⭐⭐⭐ 核心：PageConfig JSON（form-create完整规则）
    pageConfig: {
      form: {
        rules: columns.map(col => ({
          type: getFormControlType(col.dataType),
          field: col.name,
          title: col.name,
          value: '',
          props: {
            placeholder: `请输入${col.name}`,
            clearable: true
          },
          validate: !col.isNullable ? [{
            type: 'required',
            message: `${col.name}不能为空`
          }] : [],
          col: { span: 12 }
        })),
        config: {
          size: 'default',
          labelPosition: 'right',
          labelWidth: 100,
          submitButtonText: '提交',
          resetButtonText: '重置'
        },
        effects: []
      },
      list: {
        columns: columns.map(col => ({
          prop: col.name,
          label: col.name,
          width: 150,
          sortable: true,
          filterable: true,
          searchable: true
        })),
        pagination: {
          pageSize: 20,
          pageSizes: [10, 20, 50, 100]
        },
        actions: [
          { type: 'create', label: '新增', icon: 'el-icon-plus' },
          { type: 'edit', label: '编辑', icon: 'el-icon-edit' },
          { type: 'delete', label: '删除', icon: 'el-icon-delete' }
        ]
      },
      detail: {
        layout: 'vertical',
        sections: [{
          title: '基本信息',
          fields: columns.map(col => col.name)
        }]
      },
      events: {},
      layout: {
        type: 'grid',
        columns: 24,
        gutter: 20
      }
    }
  });

  return { moduleId: module.id, entityId: entity.id };

// 辅助函数：根据数据库类型推断控件类型
const getControlType = (dataType: string): string => {
  const lowerType = dataType?.toLowerCase() || 'string';
  if (lowerType.includes('int') || lowerType.includes('decimal')) return 'number';
  if (lowerType.includes('date') || lowerType.includes('time')) return 'date';
  if (lowerType.includes('bool') || lowerType.includes('bit')) return 'switch';
  return 'input';
}

const getFormControlType = (dataType: string): string => {
  const lowerType = dataType?.toLowerCase() || 'string';
  if (lowerType.includes('int') || lowerType.includes('decimal')) return 'inputNumber';
  if (lowerType.includes('date')) return 'datePicker';
  if (lowerType.includes('time')) return 'timePicker';
  if (lowerType.includes('bool') || lowerType.includes('bit')) return 'switch';
  if (lowerType.includes('text')) return 'textarea';
  return 'input';
}

export const lowCodeApi = {
  // 模块管理

  // 实体管理
  createEntity: (data: CreateEntityDto) =>
    request.post('/api/app/low-code-entity', data),

  // 属性管理
  createProperty: (data: CreatePropertyDto) =>
    request.post('/api/app/low-code-property', data),

  // 页面配置管理
  createPageConfig: (data: CreatePageConfigDto) =>
    request.post('/api/app/low-code-page-config', data),

  // 从数据库读取配置
  getEntity: (id: string) =>
    request.get(`/api/app/low-code-entity/${id}`),

  getProperties: (entityId: string) =>
    request.get(`/api/app/low-code-property/by-entity/${entityId}`),

  getPageConfig: (entityId: string, pageType: string) =>
    request.get(`/api/app/low-code-page-config/by-entity/${entityId}/${pageType}`)
