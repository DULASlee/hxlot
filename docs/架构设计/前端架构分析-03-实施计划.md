#### 4.1 热更新支持（Week 3）

**代码实现**:

```typescript
// packages/lowcode-shared/src/components/ComponentRegistry.ts

export class ComponentRegistry {
  private componentStates = new Map<string, any>()
  
  /**
   * 热更新组件
   */
  async hotReload(componentName: string): Promise<void> {
    console.log(`🔄 Hot reloading component: ${componentName}`)
    
    // 1. 保存组件状态
    const state = this.saveComponentState(componentName)
    
    // 2. 卸载旧组件
    await this.unload(componentName)
    
    // 3. 清除加载缓存
    this.loadingPromises.delete(componentName)
    
    // 4. 重新加载组件
    await this.load(componentName)
    
    // 5. 恢复状态
    if (state) {
      this.restoreComponentState(componentName, state)
    }
    
    console.log(`✅ Component ${componentName} hot reloaded`)
  }
  
  /**
   * 保存组件状态
   */
  private saveComponentState(componentName: string): any {
    const instance = this.loadedComponents.get(componentName)
    if (!instance) return null
    
    // 提取可序列化的状态
    const state = {
      usageCount: instance.metadata.usageCount,
      lastUsed: instance.metadata.lastUsed,
      active: instance.active
    }
    
    this.componentStates.set(componentName, state)
    return state
  }
  
  /**
   * 恢复组件状态
   */
  private restoreComponentState(componentName: string, state: any): void {
    const instance = this.loadedComponents.get(componentName)
    if (!instance) return
    
    instance.metadata.usageCount = state.usageCount
    instance.metadata.lastUsed = state.lastUsed
    instance.active = state.active
  }
  
  /**
   * 启用HMR支持（仅开发模式）
   */
  enableHMR(): void {
    if (!import.meta.hot) return
    
    import.meta.hot.on('component-update', async (data: { componentName: string }) => {
      await this.hotReload(data.componentName)
    })
    
    console.log('🔥 HMR enabled for ComponentRegistry')
  }
}
```

#### 4.2 权限控制（Week 4）

**代码实现**:

```typescript
// packages/lowcode-shared/src/components/ComponentPermission.ts

export interface User {
  id: string
  roles: string[]
  permissions: string[]
}

export interface ComponentPermission {
  roles?: string[]
  permissions?: string[]
  custom?: (user: User) => boolean
}

export class ComponentRegistry {
  private permissions = new Map<string, ComponentPermission>()
  
  /**
   * 设置组件权限
   */
  setPermission(componentName: string, permission: ComponentPermission): void {
    this.permissions.set(componentName, permission)
  }
  
  /**
   * 检查用户权限
   */
  checkPermission(componentName: string, user: User): boolean {
    const permission = this.permissions.get(componentName)
    if (!permission) return true // 无限制
    
    // 自定义权限函数
    if (permission.custom) {
      return permission.custom(user)
    }
    
    // 角色检查
    if (permission.roles && permission.roles.length > 0) {
      if (!permission.roles.some(role => user.roles.includes(role))) {
        return false
      }
    }
    
    // 权限检查
    if (permission.permissions && permission.permissions.length > 0) {
      if (!permission.permissions.some(perm => user.permissions.includes(perm))) {
        return false
      }
    }
    
    return true
  }
  
  /**
   * 获取用户可用组件列表
   */
  getAvailableComponents(user: User, category?: ComponentCategory): ComponentMetadata[] {
    let components = category 
      ? this.getCategoryComponents(category)
      : Array.from(this.components.values())
    
    return components.filter(component => 
      this.checkPermission(component.name, user)
    )
  }
}
```

**使用示例**:

```typescript

    // 自定义逻辑
```

```typescript
// packages/lowcode-shared/src/components/ComponentVersion.ts

import semver from 'semver'

export class ComponentRegistry {
  /**
   * 检查版本冲突
   */
  checkVersionConflict(componentName: string, version: string): boolean {
    const existing = this.components.get(componentName)
    if (!existing) return false
    
    // 使用semver比较版本
    const existingVersion = existing.version
    const isMajorConflict = semver.major(existingVersion) !== semver.major(version)
    
    if (isMajorConflict) {
      console.warn(
        `⚠️ Version conflict: ${componentName}@${existingVersion} vs @${version}`
      )
    }
    
    return isMajorConflict
  }
  
  /**
   * 注册带版本的组件
   */
  registerVersioned(metadata: ComponentMetadata): void {
    const { name, version } = metadata
    
    // 检查冲突
    if (this.checkVersionConflict(name, version)) {
      throw new Error(
        `Major version conflict for ${name}: existing ${this.components.get(name)?.version}, new ${version}`
      )
    }
    
    // 标记旧版本为过时
    const existing = this.components.get(name)
    if (existing && semver.lt(existing.version, version)) {
      existing.deprecated = true
      existing.deprecationMessage = `Please upgrade to ${version}`
    }
    
    // 注册新版本
    this.register(metadata)
  }
  
  /**
   * 获取组件版本历史
   */
  getVersionHistory(componentName: string): ComponentMetadata[] {
    // 这需要额外的版本历史记录机制
    // 可以在未来扩展
    return []
  }
}
```

```typescript
// src/performance/webVitals.ts

import { onCLS, onFID, onLCP, onFCP, onTTFB } from 'web-vitals'

export interface WebVitalsMetric {
  name: string
  value: number
  rating: 'good' | 'needs-improvement' | 'poor'
  delta: number
  id: string
}

export class WebVitalsMonitor {
  private metrics: WebVitalsMetric[] = []
  
  /**
   * 开始监控
   */
  start(): void {
    onCLS(this.handleMetric.bind(this))
    onFID(this.handleMetric.bind(this))
    onLCP(this.handleMetric.bind(this))
    onFCP(this.handleMetric.bind(this))
    onTTFB(this.handleMetric.bind(this))
    
    console.log('📊 Web Vitals监控已启动')
  }
  
  /**
   * 处理指标
   */
  private handleMetric(metric: any): void {
    this.metrics.push(metric)
    
    // 打印指标
    console.log(`📊 ${metric.name}:`, {
      value: metric.value,
      rating: metric.rating,
      delta: metric.delta
    })
    
    // 上报到服务器
    this.reportMetric(metric)
    
    // 性能警告
    if (metric.rating === 'poor') {
      console.warn(`⚠️ Poor ${metric.name}: ${metric.value}`)
    }
  }
  
  /**
   * 上报指标
   */
  private async reportMetric(metric: WebVitalsMetric): Promise<void> {
    try {
      await fetch('/api/performance/metrics', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(metric)
      })
    } catch (error) {
      console.error('Failed to report metric:', error)
    }
  }
  
  /**
   * 获取所有指标
   */
  getMetrics(): WebVitalsMetric[] {
    return this.metrics
  }
  
  /**
   * 获取性能评分
   */
  getScore(): number {
    if (this.metrics.length === 0) return 0
    
    const goodCount = this.metrics.filter(m => m.rating === 'good').length
    return (goodCount / this.metrics.length) * 100
  }
}
```

```typescript
// src/plugins/sentry.ts

import * as Sentry from '@sentry/vue'
import type { App } from 'vue'

export function setupSentry(app: App): void {
  if (!import.meta.env.VITE_SENTRY_DSN) {
    console.warn('⚠️ Sentry DSN not configured')
    return
  }
  
  Sentry.init({
    app,
    dsn: import.meta.env.VITE_SENTRY_DSN,
    environment: import.meta.env.MODE,
    release: import.meta.env.VITE_APP_VERSION,
    
    // 采样率
    tracesSampleRate: import.meta.env.PROD ? 0.1 : 1.0,
    
    // 集成
    integrations: [
      new Sentry.BrowserTracing({
        routingInstrumentation: Sentry.vueRouterInstrumentation(router)
      }),
      new Sentry.Replay({
        maskAllText: false,
        blockAllMedia: false
      })
    ],
    
    // 过滤
    beforeSend(event, hint) {
      // 过滤开发环境错误
      if (import.meta.env.DEV) {
        return null
      }
      
      // 过滤特定错误
      if (event.exception) {
        const error = hint.originalException as Error
        if (error.message.includes('ResizeObserver loop')) {
          return null // 忽略已知的无害错误
        }
      }
      
      return event
    }
  })
  
  console.log('✅ Sentry initialized')
}
```
