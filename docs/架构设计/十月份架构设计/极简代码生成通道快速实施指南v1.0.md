#### Step 4：实现批量生成逻辑（1.5小时）

```typescript
const startBatchGeneration = async () => {
  batchGenerating.value = true
  batchProgress.value = 0
  
  for (let i = 0; i < batchEntities.value.length; i++) {
    const entity = batchEntities.value[i]
    
    try {
      entity.logs.push({ message: `开始生成 ${entity.displayName}...`, type: 'info' })
      
      // 构建输入参数
      const input: SimplifiedModuleCreationDto = {
        systemName: config.systemName,
        moduleName: config.moduleName,
        entityName: entity.name,
        entityDisplayName: entity.displayName,
        displayName: entity.displayName,
        description: `${entity.displayName}模块`,
        fields: entity.fields
      }
      
      // 调用后端API
      const result = await SmartStudioLiteService.postApiLowcodeSmartStudioLiteCreateModule({ 
        requestBody: input 
      })
      
      if (result.success) {
        entity.generated = true
        entity.fileCount = result.generatedFiles?.length || 0
        entity.logs.push({ 
          message: `✅ 生成成功！文件数：${entity.fileCount}`, 
          type: 'success' 
        })
      } else {
        entity.logs.push({ 
          message: `❌ 生成失败：${result.message}`, 
          type: 'error' 
        })
      }
    } catch (error) {
      entity.logs.push({ 
        message: `❌ 生成异常：${(error as Error).message}`, 
        type: 'error' 
      })
    }
    
    batchProgress.value = i + 1
    
    // 等待1秒避免冲突
    await new Promise(resolve => setTimeout(resolve, 1000))
  }
  
  batchGenerating.value = false
  
  const successCount = batchEntities.value.filter(e => e.generated).length
  if (successCount === batchEntities.value.length) {
    ElMessage.success(`🎉 全部生成成功！共 ${successCount} 个模块`)
  } else {
    ElMessage.warning(`部分生成失败：${successCount}/${batchEntities.value.length}`)
  }
}
```

创建文件：`src/SmartAbp.Vue/src/router/dynamicRouteLoader.ts`

```typescript
import type { RouteRecordRaw, Router } from 'vue-router'
import { menuApi } from '@/api/permission-management/menu-api'
import type { MenuDto } from '@/types/permission-management/menu.types'

export class DynamicRouteLoader {
  async loadMenusFromDatabase(): Promise<MenuDto[]> {
    const result = await menuApi.getList({
      maxResultCount: 1000,
      sorting: 'Order asc'
    })
    return result.items || []
  }

  convertMenuToRoute(menu: MenuDto): RouteRecordRaw | null {
    if (!menu.isEnabled) return null
    
    return {
      path: menu.path || `/${menu.name.toLowerCase()}`,
      name: menu.name,
      component: () => import(`@/views/${menu.component}.vue`),
      meta: {
        title: menu.displayName,
        icon: menu.icon,
        hidden: !menu.isVisible
      }
    }
  }

  buildRouteTree(menus: MenuDto[]): RouteRecordRaw[] {
    const rootMenus = menus.filter(m => !m.parentId)
    
    const buildChildren = (parent: MenuDto): RouteRecordRaw[] => {
      return menus
        .filter(m => m.parentId === parent.id)
        .map(child => {
          const route = this.convertMenuToRoute(child)
          if (route) {
            const grandChildren = buildChildren(child)
            if (grandChildren.length > 0) {
              route.children = grandChildren
            }
          }
          return route
        })
        .filter(r => r !== null) as RouteRecordRaw[]
    }
    
    return rootMenus.map(root => {
      const route = this.convertMenuToRoute(root)
      if (route) {
        route.children = buildChildren(root)
      }
      return route
    }).filter(r => r !== null) as RouteRecordRaw[]
  }

  async registerDynamicRoutes(router: Router): Promise<void> {
    const menus = await this.loadMenusFromDatabase()
    const routes = this.buildRouteTree(menus)
    
    routes.forEach(route => router.addRoute(route))
    
    console.log(`✅ 动态路由注册完成：${routes.length} 个路由`)
  }
}
```

  { path: '/', redirect: '/dashboard' },
  { path: '/login', component: () => import('@/views/auth/Login.vue') },
  { path: '/dashboard', component: () => import('@/views/dashboard/Dashboard.vue') },
  { path: '/:pathMatch(.*)*', component: () => import('@/views/error/NotFound.vue') }
]

const router = createRouter({
  history: createWebHistory(),
  routes: staticRoutes
})

// 初始化时加载动态路由
let loaded = false
router.beforeEach(async (to, from, next) => {
  if (!loaded) {
    await new DynamicRouteLoader().registerDynamicRoutes(router)
    loaded = true
    next({ ...to, replace: true })
  } else {
    next()
  }
})

export default router

3. F12 控制台查看错误信息
4. 确认 `dynamicRouteLoader.ts` 正确导入

### Q3: 生成的代码找不到导航属性

2. 检查后端是否接收到关系定义
3. 检查模板文件是否正确修改
4. 重新生成代码
