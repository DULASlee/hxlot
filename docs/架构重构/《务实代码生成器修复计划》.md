public class ReliableTemplatePathResolver
{
    public string GetTemplatePath(string templateName)
    {
        // 优先级1: 配置路径（生产环境）
        var configPath = _configuration["CodeGeneration:TemplateRootPath"];
        if (!string.IsNullOrEmpty(configPath))
        {
            var fullPath = Path.Combine(configPath, templateName);
            if (File.Exists(fullPath)) return fullPath;
        }
        
        // 优先级2: 相对路径（开发环境）
        var relativePath = Path.Combine("templates", templateName);
        if (File.Exists(relativePath)) return relativePath;
        
        // 优先级3: 解决方案根目录查找
        var solutionRoot = FindSolutionRoot();
        if (solutionRoot != null)
        {
            var solutionTemplatePath = Path.Combine(solutionRoot, "templates", templateName);
            if (File.Exists(solutionTemplatePath)) return solutionTemplatePath;
        }
        
        // 优先级4: 内嵌资源（容器环境兜底）
        return ExtractEmbeddedTemplate(templateName);
    }
}

// ✅ 务实解决方案：简单但可靠
public class SimpleVariableReplacer
{
    public string ReplaceVariables(string template, ModuleMetadataDto metadata, EntityModelDto? entity = null)
    {
        var result = template;
        
        // 模块级变量
        result = result.Replace("{{ModuleName}}", metadata.Name);
        result = result.Replace("{{ModuleNamespace}}", metadata.Namespace);
        result = result.Replace("{{ModuleDescription}}", metadata.Description ?? "");
        
        // 实体级变量（如果有）
        if (entity != null)
        {
            result = result.Replace("{{EntityName}}", entity.Name);
            result = result.Replace("{{EntityNamePlural}}", GetPluralName(entity.Name));
            result = result.Replace("{{EntityNameCamel}}", ToCamelCase(entity.Name));
            result = result.Replace("{{EntityDescription}}", entity.Description ?? "");
            
            // 属性相关变量
            result = ReplacePropertyVariables(result, entity.Properties);
        }
        
        // 验证是否还有未替换的变量
        var unresolved = Regex.Matches(result, @"\{\{([^}]+)\}\}");
        if (unresolved.Count > 0)
        {
            var unresolvedList = string.Join(", ", unresolved.Cast<Match>().Select(m => m.Value));
            throw new TemplateException($"模板包含未解析的变量: {unresolvedList}");
        }
        
        return result;
    }
    
    private string ReplacePropertyVariables(string template, List<PropertyModelDto> properties)
    {
        // 生成属性列表
        var propertyDeclarations = string.Join("\n    ", properties.Select(p => 
            $"public {GetCSharpType(p.Type)} {p.Name} {{ get; set; }}"));
        
        template = template.Replace("{{PropertyDeclarations}}", propertyDeclarations);
        
        // 生成构造函数参数
        var constructorParams = string.Join(", ", properties.Where(p => p.IsRequired).Select(p => 
            $"{GetCSharpType(p.Type)} {ToCamelCase(p.Name)}"));
        
        template = template.Replace("{{ConstructorParameters}}", constructorParams);
        
        return template;
    }
}

        ["int[]"] = new("int[]", "Array.Empty<int>()", true),
        ["byte[]"] = new("byte[]", "Array.Empty<byte>()", true),
        
        // === Dictionary类型 ===
        ["dictionary<string,string>"] = new("Dictionary<string, string>", "new Dictionary<string, string>()", true),

    public static string GetCSharpType(string inputType)
    {
        if (string.IsNullOrEmpty(inputType))
            return "string";
            
        var normalizedType = inputType.ToLowerInvariant().Trim();
        
        if (TypeMappings.TryGetValue(normalizedType, out var typeInfo))
        {
            return typeInfo.CSharpType;
        }
        
        // 处理泛型类型（简单解析）
        if (normalizedType.StartsWith("list<") && normalizedType.EndsWith(">"))
        {
            var innerType = normalizedType.Substring(5, normalizedType.Length - 6);
            var mappedInnerType = GetCSharpType(innerType);
            return $"List<{mappedInnerType}>";
        }
        
        // 未知类型提示但不中断
        _logger.LogWarning("未知类型 '{Type}'，将使用 string 类型替代", inputType);
        return "string";
    }
    
    public record TypeInfo(string CSharpType, string DefaultValue, bool IsReferenceType);

// ✅ 务实解决方案：简单检测+友好提示
public class SimpleCircularReferenceDetector
{
    public CircularReferenceResult CheckCircularReferences(List<EntityModelDto> entities)
    {
        var result = new CircularReferenceResult();
        var entityGraph = BuildEntityGraph(entities);
        
        foreach (var entity in entities)
        {
            var visited = new HashSet<string>();
            if (HasCircularPath(entity.Name, entityGraph, visited, new List<string>()))
            {
                result.HasCircularReference = true;
                result.ProblematicEntities.Add(entity.Name);
            }
        }
        
        if (result.HasCircularReference)
        {
            result.RecommendedAction = "检测到循环引用。建议：1) 使用导航属性而非直接引用，2) 考虑将一个方向设为可选";
        }
        
        return result;
    }
    
    private bool HasCircularPath(string entityName, Dictionary<string, List<string>> graph, 
        HashSet<string> visited, List<string> currentPath)
    {
        if (currentPath.Contains(entityName))
        {
            return true; // 发现循环
        }
        
        if (visited.Contains(entityName))
        {
            return false; // 已访问过，无循环
        }
        
        visited.Add(entityName);
        currentPath.Add(entityName);
        
        if (graph.ContainsKey(entityName))
        {
            foreach (var referencedEntity in graph[entityName])
            {
                if (HasCircularPath(referencedEntity, graph, visited, currentPath))
                    return true;
            }
        }
        
        currentPath.Remove(entityName);
        return false;
    }
}

// ✅ 务实解决方案：备份+验证+清理
public class SafeFileWriter
{
    public async Task<FileWriteResult> WriteFilesSafelyAsync(Dictionary<string, string> filesToWrite)
    {
        var result = new FileWriteResult();
        var backups = new List<string>();
        
        try
        {
            // 阶段1: 创建备份
            foreach (var file in filesToWrite)
            {
                if (File.Exists(file.Key))
                {
                    var backupPath = $"{file.Key}.backup.{DateTime.Now:yyyyMMddHHmmss}";
                    File.Copy(file.Key, backupPath);
                    backups.Add(backupPath);
                    result.BackupFiles.Add(file.Key, backupPath);
                }
            }
            
            // 阶段2: 验证生成的代码
            foreach (var file in filesToWrite)
            {
                if (Path.GetExtension(file.Key) == ".cs")
                {
                    var syntaxValid = await ValidateCSharpSyntaxAsync(file.Value);
                    if (!syntaxValid.IsValid)
                    {
                        throw new CodeGenerationException($"生成的C#代码语法错误 ({file.Key}): {syntaxValid.Error}");
                    }
                }
            }
            
            // 阶段3: 写入文件
            foreach (var file in filesToWrite)
            {
                await File.WriteAllTextAsync(file.Key, file.Value, Encoding.UTF8);
                result.WrittenFiles.Add(file.Key);
            }
            
            result.Success = true;
            return result;
        }
        catch (Exception ex)
        {
            // 失败时恢复备份
            await RestoreBackupsAsync(result.BackupFiles);
            result.Success = false;
            result.Error = ex.Message;
            return result;
        }
    }
}

                .MapExtraProperties();
                
            CreateMap<Create{entity.Name}Dto, {entity.Name}>()
                .ConstructUsing(src => new {entity.Name}(GuidGenerator.Create(){GetRequiredProperties(entity)}))
                .MapExtraProperties();
                
            CreateMap<Update{entity.Name}Dto, {entity.Name}>()
                .ForMember(dest => dest.Id, opt => opt.Ignore())
                .ForMember(dest => dest.CreationTime, opt => opt.Ignore())
