### 步骤1.2: 创建前端类型生成脚本

**目标**: 自动从后端JSON Schema生成TypeScript类型定义

**新增文件**: `src/SmartAbp.Vue/scripts/generate-frontend-types.js`

```javascript
#!/usr/bin/env node
/**
 * 前端TypeScript类型自动生成脚本
 * 从后端JSON Schema生成TypeScript接口和Zod验证器
 */

const { execSync } = require('child_process')
const path = require('path')
const fs = require('fs')

// 配置
const BACKEND_SCHEMA_DIR = '../../src/SmartAbp.CodeGenerator/Core/Schema'
const FRONTEND_OUTPUT_DIR = './packages/lowcode-shared/src/types/generated'
const VALIDATOR_OUTPUT_DIR = './packages/lowcode-shared/src/validators'

// Schema映射配置
const SCHEMAS = [
  {
    input: 'EntityMetadataSchema.json',
    output: 'entity-metadata.ts',
    validator: 'entity-validator.ts',
    typeName: 'EntityMetadata'
  },
  {
    input: 'ModuleMetadataSchema.json',
    output: 'module-metadata.ts',
    validator: 'module-validator.ts',
    typeName: 'ModuleMetadata'
  },
  {
    input: 'AspireSolutionSchema.json',
    output: 'aspire-solution-metadata.ts',
    validator: 'aspire-validator.ts',
    typeName: 'AspireSolutionMetadata'
  }
]

console.log('🚀 开始生成前端TypeScript类型...\n')

// 1. 检查后端Schema文件是否存在
console.log('📋 步骤1: 检查后端Schema文件')
SCHEMAS.forEach(schema => {
  const schemaPath = path.join(__dirname, BACKEND_SCHEMA_DIR, schema.input)
  if (!fs.existsSync(schemaPath)) {
    console.error(`❌ Schema文件不存在: ${schemaPath}`)
    process.exit(1)
  }
  console.log(`✅ 发现Schema: ${schema.input}`)
})

// 2. 确保输出目录存在
console.log('\n📁 步骤2: 创建输出目录')
;[FRONTEND_OUTPUT_DIR, VALIDATOR_OUTPUT_DIR].forEach(dir => {
  const fullPath = path.join(__dirname, dir)
  if (!fs.existsSync(fullPath)) {
    fs.mkdirSync(fullPath, { recursive: true })
    console.log(`✅ 创建目录: ${dir}`)
  }
})

// 3. 使用quicktype生成TypeScript类型
console.log('\n🔧 步骤3: 生成TypeScript接口')
SCHEMAS.forEach(schema => {
  const schemaPath = path.join(__dirname, BACKEND_SCHEMA_DIR, schema.input)
  const outputPath = path.join(__dirname, FRONTEND_OUTPUT_DIR, schema.output)
  
  try {
    // 使用quicktype从JSON Schema生成TypeScript
    const cmd = `npx quicktype ` +
      `--src "${schemaPath}" ` +
      `--lang typescript ` +
      `--out "${outputPath}" ` +
      `--just-types ` +
      `--prefer-unions ` +
      `--nice-property-names`
    
    execSync(cmd, { stdio: 'inherit' })
    console.log(`✅ 生成类型: ${schema.output}`)
  } catch (error) {
    console.error(`❌ 生成失败: ${schema.output}`)
    throw error
  }
})

// 4. 生成统一导出文件
console.log('\n📦 步骤4: 生成统一导出')
const indexContent = SCHEMAS.map(schema => 
  `export * from './${schema.output.replace('.ts', '')}'`
).join('\n')

fs.writeFileSync(
  path.join(__dirname, FRONTEND_OUTPUT_DIR, 'index.ts'),
  indexContent
)
console.log('✅ 生成index.ts')

// 5. 生成Zod验证器模板
console.log('\n🛡️ 步骤5: 生成Zod验证器模板')
SCHEMAS.forEach(schema => {
  const validatorPath = path.join(__dirname, VALIDATOR_OUTPUT_DIR, schema.validator)
  
  // 如果验证器文件不存在，创建模板
  if (!fs.existsSync(validatorPath)) {
    const validatorTemplate = `import { z } from 'zod'
import type { ${schema.typeName} } from '../types/generated'

/**
 * ${schema.typeName} Zod Schema
 * 自动生成的验证器，可根据需求手动调整
 */
export const ${schema.typeName}Schema = z.object({
  // TODO: 根据${schema.input}手动定义Zod Schema
  // 或使用json-schema-to-zod自动转换
})

/**
 * 验证${schema.typeName}
 */
export const validate${schema.typeName} = (data: unknown): ${schema.typeName} => {
  return ${schema.typeName}Schema.parse(data)
}

/**
 * 安全验证${schema.typeName}（返回结果）
 */
export const safeValidate${schema.typeName} = (data: unknown) => {
  return ${schema.typeName}Schema.safeParse(data)
}
`
    
    fs.writeFileSync(validatorPath, validatorTemplate)
    console.log(`✅ 创建验证器模板: ${schema.validator}`)
  } else {
    console.log(`⏭️  验证器已存在: ${schema.validator}`)
  }
})

// 6. 生成验证器统一导出
const validatorIndexContent = SCHEMAS.map(schema => 
  `export * from './${schema.validator.replace('.ts', '')}'`
).join('\n')

fs.writeFileSync(
  path.join(__dirname, VALIDATOR_OUTPUT_DIR, 'index.ts'),
  validatorIndexContent
)
console.log('✅ 生成validators/index.ts')

console.log('\n✅ 前端TypeScript类型生成完成！')
console.log('\n📋 下一步操作:')
console.log('1. 检查生成的类型文件: packages/lowcode-shared/src/types/generated/')
console.log('2. 完善Zod验证器: packages/lowcode-shared/src/validators/')
console.log('3. 运行type-check: cd packages/lowcode-shared && npm run type-check')
```

**执行命令**:
```bash
cd src/SmartAbp.Vue
node scripts/generate-frontend-types.js
```

**验证标准**:
- ✅ 生成的TypeScript类型无语法错误
- ✅ 类型定义与后端C# DTO一致
- ✅ Zod验证器模板创建成功

/**
 * Schema版本管理器
 * 负责Schema版本控制、兼容性检查和迁移
 */

export interface SchemaVersion {
  version: string
  releaseDate: string
  status: 'current' | 'deprecated' | 'archived'
  schemas: {
    entity: string
    module: string
    aspire: string
  }
  compatibleWith: string[]
  breakingChanges: boolean
}

export interface SchemaRegistry {
  schemaVersions: SchemaVersion[]
}

export class SchemaVersionManager {
  private registry: SchemaRegistry | null = null
  
  /**
   * 加载Schema版本注册表
   */
  async loadRegistry(): Promise<SchemaRegistry> {
    if (this.registry) {
      return this.registry
    }
    
    // 从后端或本地加载注册表
    const response = await fetch('/api/schema/registry')
    this.registry = await response.json()
    return this.registry
  }
  
  /**
   * 获取当前Schema版本
   */
  async getCurrentVersion(): Promise<SchemaVersion> {
    const registry = await this.loadRegistry()
    const current = registry.schemaVersions.find(v => v.status === 'current')
    
    if (!current) {
      throw new Error('未找到当前Schema版本')
    }
    
    return current
  }
  
  /**
   * 检查Schema版本兼容性
   */
  async isCompatible(fromVersion: string, toVersion: string): Promise<boolean> {
    const registry = await this.loadRegistry()
    const targetVersion = registry.schemaVersions.find(v => v.version === toVersion)
    
    if (!targetVersion) {
      return false
    }
    
    // 检查兼容性列表
    return targetVersion.compatibleWith.includes(fromVersion)
  }
  
  /**
   * 迁移元数据到新版本（前端简化版）
   */
  async migrateMetadata<T>(
    metadata: unknown,
    fromVersion: string,
    toVersion: string
  ): Promise<T> {
    // 实际项目中，这里应该实现完整的迁移逻辑
    // 前端可以简化处理，主要由后端负责迁移
    
    const compatible = await this.isCompatible(fromVersion, toVersion)
    
    if (!compatible) {
      throw new Error(`Schema版本${fromVersion}不兼容${toVersion}`)
    }
    
    // 简单返回（实际应实现字段映射）
    return metadata as T
  }
  
  /**
   * 验证元数据版本
   */
  validateVersion(metadata: { schemaVersion?: string }): boolean {
    if (!metadata.schemaVersion) {
      console.warn('元数据缺少schemaVersion字段')
      return false
    }
    
    // 检查版本格式
    const versionRegex = /^\d+\.\d+\.\d+$/
    return versionRegex.test(metadata.schemaVersion)
  }
}
