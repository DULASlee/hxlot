#### 6.1.1 重构entityStore（1.5天）

**目标**: 迁移实体状态管理到统一元数据模型

**修改文件**: `src/SmartAbp.Vue/packages/lowcode-core/src/stores/entityStore.ts`

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import {
  type EntityMetadata,
  type PropertyMetadata,
  validateEntityMetadata,
  safeValidateEntityMetadata
} from '@smartabp/lowcode-shared'  // ⬅️ 统一类型
import { codeGeneratorApi } from '@smartabp/lowcode-api'

export const useEntityStore = defineStore('entity', () => {
  // ========== 状态 ==========
  const entities = ref<Map<string, EntityMetadata>>(new Map())  // ⬅️ 使用统一类型
  const currentEntity = ref<EntityMetadata | null>(null)
  const isLoading = ref(false)
  const errors = ref<string[]>([])
  
  // ========== 计算属性 ==========
  const entityList = computed(() => Array.from(entities.value.values()))
  
  const hasValidationErrors = computed(() => {
    if (!currentEntity.value) return false
    const result = safeValidateEntityMetadata(currentEntity.value)
    return !result.success
  })
  
  const validationErrors = computed(() => {
    if (!currentEntity.value) return []
    const result = safeValidateEntityMetadata(currentEntity.value)
    if (result.success) return []
    return result.error.errors.map(e => `${e.path.join('.')}: ${e.message}`)
  })
  
  // ========== 操作 ==========
  
  /**
   * 创建新实体（使用统一元数据）
   */
  function createEntity(module: string): EntityMetadata {
    const newEntity: EntityMetadata = {
      name: '',
      module,
      keyType: 'Guid',
      isAggregateRoot: true,
      isMultiTenant: true,
      isSoftDelete: true,
      hasExtraProperties: true,
      properties: [],
      navigationProperties: []
    }
    
    currentEntity.value = newEntity
    return newEntity
  }
  
  /**
   * 加载实体
   */
  async function loadEntity(entityName: string): Promise<EntityMetadata | null> {
    isLoading.value = true
    errors.value = []
    
    try {
      // 调用API加载实体元数据
      const metadata = await codeGeneratorApi.loadEntityMetadata(entityName)
      
      // 验证元数据
      const validatedMetadata = validateEntityMetadata(metadata)
      
      entities.value.set(entityName, validatedMetadata)
      currentEntity.value = validatedMetadata
      
      return validatedMetadata
    } catch (error: any) {
      errors.value = [error.message || '加载实体失败']
      return null
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 保存实体
   */
  async function saveEntity(entity: EntityMetadata): Promise<boolean> {
    isLoading.value = true
    errors.value = []
    
    try {
      // 验证元数据
      validateEntityMetadata(entity)
      
      // 调用API保存
      await codeGeneratorApi.saveEntityMetadata(entity)
      
      // 更新缓存
      entities.value.set(entity.name, entity)
      
      return true
    } catch (error: any) {
      errors.value = [error.message || '保存实体失败']
      return false
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 删除实体
   */
  async function deleteEntity(entityName: string): Promise<boolean> {
    isLoading.value = true
    errors.value = []
    
    try {
      await codeGeneratorApi.deleteEntityMetadata(entityName)
      entities.value.delete(entityName)
      
      if (currentEntity.value?.name === entityName) {
        currentEntity.value = null
      }
      
      return true
    } catch (error: any) {
      errors.value = [error.message || '删除实体失败']
      return false
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 添加属性
   */
  function addProperty(property: PropertyMetadata) {
    if (!currentEntity.value) {
      throw new Error('当前无实体')
    }
    
    currentEntity.value.properties.push(property)
  }
  
  /**
   * 更新属性
   */
  function updateProperty(index: number, property: PropertyMetadata) {
    if (!currentEntity.value) {
      throw new Error('当前无实体')
    }
    
    currentEntity.value.properties[index] = property
  }
  
  /**
   * 删除属性
   */
  function removeProperty(index: number) {
    if (!currentEntity.value) {
      throw new Error('当前无实体')
    }
    
    currentEntity.value.properties.splice(index, 1)
  }
  
  /**
   * 重置当前实体
   */
  function resetCurrent() {
    currentEntity.value = null
    errors.value = []
  }
  
  /**
   * 批量加载实体
   */
  async function loadEntitiesByModule(module: string): Promise<EntityMetadata[]> {
    isLoading.value = true
    errors.value = []
    
    try {
      const metadataList = await codeGeneratorApi.loadEntitiesByModule(module)
      
      // 验证并缓存
      metadataList.forEach(metadata => {
        const validated = validateEntityMetadata(metadata)
        entities.value.set(validated.name, validated)
      })
      
      return metadataList
    } catch (error: any) {
      errors.value = [error.message || '加载模块实体失败']
      return []
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 清空缓存
   */
  function clearCache() {
    entities.value.clear()
    currentEntity.value = null
    errors.value = []
  }
  
  // ========== 返回 ==========
  return {
    // 状态
    entities,
    currentEntity,
    isLoading,
    errors,
    
    // 计算属性
    entityList,
    hasValidationErrors,
    validationErrors,
    
    // 操作
    createEntity,
    loadEntity,
    saveEntity,
    deleteEntity,
    addProperty,
    updateProperty,
    removeProperty,
    resetCurrent,
    loadEntitiesByModule,
    clearCache
  }
})
```

#### 6.1.2 重构moduleStore（1.5天）

**目标**: 迁移模块状态管理到统一元数据模型

**修改文件**: `src/SmartAbp.Vue/packages/lowcode-core/src/stores/moduleStore.ts`

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import {
  type ModuleMetadata,
  type RouteMetadata,
  type StoreMetadata,
  validateModuleMetadata,
  safeValidateModuleMetadata,
  schemaVersionManager
} from '@smartabp/lowcode-shared'  // ⬅️ 统一类型
import { codeGeneratorApi } from '@smartabp/lowcode-api'

export const useModuleStore = defineStore('module', () => {
  // ========== 状态 ==========
  const modules = ref<Map<string, ModuleMetadata>>(new Map())
  const currentModule = ref<ModuleMetadata | null>(null)
  const isLoading = ref(false)
  const errors = ref<string[]>([])
  const currentSchemaVersion = ref<string>('1.0.0')
  
  // ========== 计算属性 ==========
  const moduleList = computed(() => Array.from(modules.value.values()))
  
  const hasValidationErrors = computed(() => {
    if (!currentModule.value) return false
    const result = safeValidateModuleMetadata(currentModule.value)
    return !result.success
  })
  
  const validationErrors = computed(() => {
    if (!currentModule.value) return []
    const result = safeValidateModuleMetadata(currentModule.value)
    if (result.success) return []
    return result.error.errors.map(e => `${e.path.join('.')}: ${e.message}`)
  })
  
  // ========== 初始化 ==========
  async function init() {
    try {
      // 加载当前Schema版本
      const version = await codeGeneratorApi.getCurrentSchemaVersion()
      currentSchemaVersion.value = version.version
    } catch (error) {
      console.error('加载Schema版本失败', error)
    }
  }
  
  // ========== 操作 ==========
  
  /**
   * 创建新模块
   */
  function createModule(): ModuleMetadata {
    const newModule: ModuleMetadata = {
      name: '',
      displayName: '',
      version: '1.0.0',
      abpStyle: true,
      order: 100,
      dependsOn: [],
      routes: [],
      stores: [],
      policies: [],
      lifecycle: {},
      features: {
        enableAuth: true,
        enableCache: true,
        enableI18n: true
      }
    }
    
    currentModule.value = newModule
    return newModule
  }
  
  /**
   * 加载模块
   */
  async function loadModule(moduleName: string): Promise<ModuleMetadata | null> {
    isLoading.value = true
    errors.value = []
    
    try {
      const metadata = await codeGeneratorApi.loadModuleMetadata(moduleName)
      
      // 检查Schema版本兼容性
      if (metadata.schemaVersion) {
        const compatible = await schemaVersionManager.isCompatible(
          metadata.schemaVersion,
          currentSchemaVersion.value
        )
        
        if (!compatible) {
          throw new Error(`模块Schema版本${metadata.schemaVersion}不兼容当前版本${currentSchemaVersion.value}`)
        }
      }
      
      // 验证元数据
      const validatedMetadata = validateModuleMetadata(metadata)
      
      modules.value.set(moduleName, validatedMetadata)
      currentModule.value = validatedMetadata
      
      return validatedMetadata
    } catch (error: any) {
      errors.value = [error.message || '加载模块失败']
      return null
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 保存模块
   */
  async function saveModule(module: ModuleMetadata): Promise<boolean> {
    isLoading.value = true
    errors.value = []
    
    try {
      // 验证元数据
      validateModuleMetadata(module)
      
      // 添加Schema版本
      const moduleWithVersion = {
        ...module,
        schemaVersion: currentSchemaVersion.value
      }
      
      // 调用API保存
      await codeGeneratorApi.saveModuleMetadata(moduleWithVersion)
      
      // 更新缓存
      modules.value.set(module.name, module)
      
      return true
    } catch (error: any) {
      errors.value = [error.message || '保存模块失败']
      return false
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 删除模块
   */
  async function deleteModule(moduleName: string): Promise<boolean> {
    isLoading.value = true
    errors.value = []
    
    try {
      await codeGeneratorApi.deleteModuleMetadata(moduleName)
      modules.value.delete(moduleName)
      
      if (currentModule.value?.name === moduleName) {
        currentModule.value = null
      }
      
      return true
    } catch (error: any) {
      errors.value = [error.message || '删除模块失败']
      return false
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 添加路由
   */
  function addRoute(route: RouteMetadata) {
    if (!currentModule.value) {
      throw new Error('当前无模块')
    }
    
    currentModule.value.routes.push(route)
  }
  
  /**
   * 更新路由
   */
  function updateRoute(index: number, route: RouteMetadata) {
    if (!currentModule.value) {
      throw new Error('当前无模块')
    }
    
    currentModule.value.routes[index] = route
  }
  
  /**
   * 删除路由
   */
  function removeRoute(index: number) {
    if (!currentModule.value) {
      throw new Error('当前无模块')
    }
    
    currentModule.value.routes.splice(index, 1)
  }
  
  /**
   * 添加Store
   */
  function addStore(store: StoreMetadata) {
    if (!currentModule.value) {
      throw new Error('当前无模块')
    }
    
    currentModule.value.stores.push(store)
  }
  
  /**
   * 批量加载模块
   */
  async function loadAllModules(): Promise<ModuleMetadata[]> {
    isLoading.value = true
    errors.value = []
    
    try {
      const metadataList = await codeGeneratorApi.loadAllModules()
      
      // 验证并缓存
      metadataList.forEach(metadata => {
        const validated = validateModuleMetadata(metadata)
        modules.value.set(validated.name, validated)
      })
      
      return metadataList
    } catch (error: any) {
      errors.value = [error.message || '加载模块列表失败']
      return []
    } finally {
      isLoading.value = false
    }
  }
  
  /**
   * 重置当前模块
   */
  function resetCurrent() {
    currentModule.value = null
    errors.value = []
  }
  
  /**
   * 清空缓存
   */
  function clearCache() {
    modules.value.clear()
    currentModule.value = null
    errors.value = []
  }
  
  // ========== 返回 ==========
  return {
    // 状态
    modules,
    currentModule,
    isLoading,
    errors,
    currentSchemaVersion,
    
    // 计算属性
    moduleList,
    hasValidationErrors,
    validationErrors,
    
    // 操作
    init,
    createModule,
    loadModule,
    saveModule,
    deleteModule,
    addRoute,
    updateRoute,
    removeRoute,
    addStore,
    loadAllModules,
    resetCurrent,
    clearCache
  }
})
```

```typescript
export function migrateManifestToModuleMetadata(manifest: any): ModuleMetadata {
  return {
    name: manifest.name,
    displayName: manifest.displayName,
    version: manifest.version || '1.0.0',
    description: manifest.description,
    author: manifest.author,
    abpStyle: manifest.abpStyle ?? true,
    order: manifest.order ?? 100,
    dependsOn: manifest.dependsOn || [],
    routes: manifest.routes || [],
    stores: manifest.stores || [],
    policies: manifest.policies || [],
    lifecycle: manifest.lifecycle || {},
    features: manifest.features || {
      enableAuth: true,
      enableCache: true,
      enableI18n: true
    },
    menuConfig: manifest.menuConfig
  }
}

/**
 * 批量迁移旧Manifest文件
 */
export async function migrateAllManifests(oldManifests: any[]): Promise<ModuleMetadata[]> {
  return oldManifests.map(manifest => {
    try {
      const metadata = migrateManifestToModuleMetadata(manifest)
      
      // 验证迁移结果
      validateModuleMetadata(metadata)
      
      return metadata
    } catch (error) {
      console.error(`迁移Manifest失败: ${manifest.name}`, error)
      throw error
    }
  })
}
```

#### 6.2.1 重构LowCodeEngineView（1.5天）

**目标**: 低代码引擎主视图适配统一元数据

**修改文件**: `src/SmartAbp.Vue/packages/lowcode-designer/src/views/LowCodeEngineView.vue`

**关键改动**:

```vue
<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
import { useEntityStore } from '@smartabp/lowcode-core'
import {
  type EntityMetadata,
  type PropertyMetadata,
  validateEntityMetadata,
  safeValidateEntityMetadata
} from '@smartabp/lowcode-shared'  // ⬅️ 统一类型
import { codeGeneratorApi } from '@smartabp/lowcode-api'
import { ElMessage } from 'element-plus'

// ========== Store ==========
const entityStore = useEntityStore()

// ========== 状态 ==========
const entity = ref<EntityMetadata | null>(null)
const isGenerating = ref(false)
const generatedFiles = ref<any[]>([])

// ========== 生命周期 ==========
onMounted(async () => {
  // 如果有现有实体，加载它
  if (entityStore.currentEntity) {
    entity.value = entityStore.currentEntity
  } else {
    // 创建新实体
    entity.value = entityStore.createEntity('DefaultModule')
  }
})

// ========== 实时验证 ==========
watch(entity, (value) => {
  if (value) {
    const result = safeValidateEntityMetadata(value)
    
    if (!result.success) {
      console.warn('实体验证失败', result.error.errors)
    }
  }
}, { deep: true })

// ========== 操作 ==========
const handleGenerate = async () => {
  if (!entity.value) {
    ElMessage.error('请先创建实体')
    return
  }
  
  // 最终验证
  try {
    validateEntityMetadata(entity.value)
  } catch (error: any) {
    ElMessage.error(`元数据验证失败：${error.message}`)
    return
  }
  
  isGenerating.value = true
  
  try {
    const result = await codeGeneratorApi.generateFromUnifiedMetadata({
      metadataType: 'Entity',
      schemaVersion: '1.0.0',
      metadata: entity.value
    })
    
    if (result.success) {
      generatedFiles.value = result.generatedFiles
      ElMessage.success('代码生成成功！')
      
      // 保存实体元数据
      await entityStore.saveEntity(entity.value)
    } else {
      ElMessage.error(`生成失败：${result.errors.join(', ')}`)
    }
  } catch (error: any) {
    ElMessage.error(`生成失败：${error.message}`)
  } finally {
    isGenerating.value = false
  }
}

const handleSave = async () => {
  if (!entity.value) {
    ElMessage.error('无实体可保存')
    return
  }
  
  const success = await entityStore.saveEntity(entity.value)
  
  if (success) {
    ElMessage.success('实体保存成功')
  } else {
    ElMessage.error(`保存失败：${entityStore.errors.join(', ')}`)
  }
}

const handleAddProperty = () => {
  const newProperty: PropertyMetadata = {
    name: '',
    type: 'string',
    isRequired: false,
    isReadOnly: false,
    isUnique: false,
    validationRules: []
  }
  
  entityStore.addProperty(newProperty)
}
</script>

<template>
  <div class="lowcode-engine-view">
    <!-- 组件UI -->
    <!-- 使用统一元数据类型 -->
  </div>
</template>
```

#### 6.3.2 重构moduleLoader

**目标**: 模块加载器适配统一元数据

**修改文件**: `src/SmartAbp.Vue/src/tools/moduleLoader.ts`

```typescript
import {
  type ModuleMetadata,
  validateModuleMetadata,
  schemaVersionManager
} from '@smartabp/lowcode-shared'
import { codeGeneratorApi } from '@smartabp/lowcode-api'

/**
 * 模块加载器（统一元数据版本）
 */
export class ModuleLoader {
  private loadedModules: Map<string, ModuleMetadata> = new Map()
  
  /**
   * 加载单个模块
   */
  async loadModule(moduleName: string): Promise<ModuleMetadata> {
    // 检查缓存
    if (this.loadedModules.has(moduleName)) {
      return this.loadedModules.get(moduleName)!
    }
    
    // 从API加载
    const metadata = await codeGeneratorApi.loadModuleMetadata(moduleName)
    
    // 验证Schema版本
    if (metadata.schemaVersion) {
      const currentVersion = await codeGeneratorApi.getCurrentSchemaVersion()
      const compatible = await schemaVersionManager.isCompatible(
        metadata.schemaVersion,
        currentVersion.version
      )
      
      if (!compatible) {
        throw new Error(`模块Schema版本不兼容`)
      }
    }
    
    // 验证元数据
    const validatedMetadata = validateModuleMetadata(metadata)
    
    // 缓存
    this.loadedModules.set(moduleName, validatedMetadata)
    
    return validatedMetadata
  }
  
  /**
   * 批量加载模块
   */
  async loadModules(moduleNames: string[]): Promise<ModuleMetadata[]> {
    return Promise.all(moduleNames.map(name => this.loadModule(name)))
  }
  
  /**
   * 清空缓存
   */
  clearCache() {
    this.loadedModules.clear()
  }
  
  /**
   * 向后兼容：加载旧Manifest（自动迁移）
   */
  async loadLegacyManifest(manifestPath: string): Promise<ModuleMetadata> {
    console.warn('[Deprecated] 请使用loadModule代替')
    
    // 加载旧格式
    const response = await fetch(manifestPath)
    const oldManifest = await response.json()
    
    // 迁移到新格式
    const metadata = manifestToModuleMetadata(oldManifest)
    
    // 验证
    validateModuleMetadata(metadata)
    
    return metadata
  }
}

// 导出单例
export const moduleLoader = new ModuleLoader()
```
