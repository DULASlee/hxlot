/// <summary>
/// 数据库Schema元数据读取器 - 替代复杂的可视化实体设计
/// </summary>
public class DatabaseSchemaReader : ITransientDependency
{
    private readonly IConnectionStringResolver _connectionStringResolver;
    private readonly ILogger<DatabaseSchemaReader> _logger;
    
    /// <summary>
    /// 🚀 根据表名读取完整表结构 - 用户只需输入表名
    /// </summary>
    public async Task<DatabaseTableSchema> ReadTableSchemaAsync(string tableName, string? connectionName = null)
    {
        var connectionString = _connectionStringResolver.Resolve(connectionName);
        using var connection = new SqlConnection(connectionString);
        
        _logger.LogInformation("🔍 开始读取数据库表结构: {TableName}", tableName);
        
        // 1. 读取表基本信息
        var tableInfo = await ReadTableInfoAsync(connection, tableName);
        
        // 2. 读取列信息 (字段定义)
        var columns = await ReadColumnsAsync(connection, tableName);
        
        // 3. 读取主键和外键关系
        var primaryKeys = await ReadPrimaryKeysAsync(connection, tableName);
        var foreignKeys = await ReadForeignKeysAsync(connection, tableName);
        
        // 🎯 构建完整表结构模型
        return new DatabaseTableSchema
        {
            TableName = tableName,
            Columns = columns,
            PrimaryKeys = primaryKeys,
            ForeignKeys = foreignKeys,
            Description = tableInfo.Description
        };
    }

    /// <summary>
    /// 🔍 获取数据库所有表列表 - 支持表名自动补全
    /// </summary>
    public async Task<List<string>> GetAllTableNamesAsync(string? connectionName = null)
    {
        var connectionString = _connectionStringResolver.Resolve(connectionName);
        using var connection = new SqlConnection(connectionString);
        
        var query = "SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_TYPE = 'BASE TABLE'";
        var tableNames = await connection.QueryAsync<string>(query);
        
        return tableNames.ToList();
    }
}

// src/SmartAbp.CodeGenerator/Core/Database/DatabaseTypeMapper.cs
public class DatabaseTypeMapper : ITransientDependency
{
    /// <summary>
    /// 🔧 数据库类型到C#类型映射
    /// </summary>
    public string MapDatabaseTypeToCSharp(string dbType, bool isNullable)
    {
        var csharpType = dbType.ToLowerInvariant() switch
        {
            "int" or "integer" => "int",
            "bigint" => "long",
            "varchar" or "nvarchar" or "text" or "ntext" => "string",
            "bit" => "bool",
            "datetime" or "datetime2" => "DateTime",
            "decimal" or "numeric" or "money" => "decimal",
            "uniqueidentifier" => "Guid",
            "float" or "real" => "double",
            _ => "string" // 默认为string
        };
        
        // 处理可空类型 (除了string)
        if (isNullable && csharpType != "string")
        {
            csharpType += "?";
        }
        
        return csharpType;
    }
    
    /// <summary>
    /// 🔧 数据库类型到TypeScript类型映射
    /// </summary>
    public string MapDatabaseTypeToTypeScript(string dbType)
    {
        return dbType.ToLowerInvariant() switch
        {
            "int" or "bigint" or "decimal" or "float" => "number",
            "bit" => "boolean",
            "datetime" or "datetime2" => "Date",
            _ => "string"
        };
    }
}

// src/SmartAbp.CodeGenerator/Core/Database/SchemaToEntityConverter.cs
public class SchemaToEntityConverter : ITransientDependency
{
    private readonly DatabaseTypeMapper _typeMapper;
    
    /// <summary>
    /// 🎯 数据库Schema转换为实体模型 - 核心转换逻辑
    /// </summary>
    public EnhancedEntityModelDto ConvertSchemaToEntity(DatabaseTableSchema schema)
    {
        var entity = new EnhancedEntityModelDto
        {
            Name = ConvertToPascalCase(schema.TableName),
            DisplayName = GenerateDisplayName(schema.TableName),
            TableName = schema.TableName,
            Description = schema.Description,
            Properties = new List<EntityPropertyDto>()
        };
        
        // 转换列为属性
        foreach (var column in schema.Columns)
        {
            var property = new EntityPropertyDto
            {
                Name = ConvertToPascalCase(column.ColumnName),
                Type = _typeMapper.MapDatabaseTypeToCSharp(column.DataType, column.IsNullable == "YES"),
                IsRequired = column.IsNullable == "NO",
                IsKey = column.IsPrimaryKey,
                MaxLength = column.MaxLength,
                Description = column.Description,
                DefaultValue = column.DefaultValue
            };
            
            entity.Properties.Add(property);
        }
        
        return entity;
    }
    
    private string ConvertToPascalCase(string input)
    {
        return string.Join("", input.Split('_')
            .Select(word => char.ToUpper(word[0]) + word[1..].ToLower()));
    }
    
    private string GenerateDisplayName(string tableName)
    {
        // Users -> 用户, Products -> 产品
        return tableName switch
        {
            "Users" => "用户",
            "Products" => "产品", 
            "Orders" => "订单",
            _ => ConvertToPascalCase(tableName)
        };
    }
}

// 添加Redis缓存支持
public class CachedDatabaseSchemaReader : ITransientDependency
{
    private readonly DatabaseSchemaReader _reader;
    private readonly IDistributedCache _cache;
    
    public async Task<DatabaseTableSchema> ReadTableSchemaAsync(string tableName, string? connectionName = null)
    {
        var cacheKey = $"table_schema:{connectionName ?? "default"}:{tableName}";
        
        // 尝试从缓存获取
        var cachedSchema = await _cache.GetStringAsync(cacheKey);
        if (cachedSchema != null)
        {
            return JsonSerializer.Deserialize<DatabaseTableSchema>(cachedSchema)!;
        }
        
        // 从数据库读取
        var schema = await _reader.ReadTableSchemaAsync(tableName, connectionName);
        
        // 缓存30分钟
        await _cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(schema), 
            new DistributedCacheEntryOptions { SlidingExpiration = TimeSpan.FromMinutes(30) });
            
        return schema;
    }
}

// 支持MySQL, PostgreSQL, SQLite
public class MultiDatabaseSchemaReader : ITransientDependency
{
    public async Task<DatabaseTableSchema> ReadTableSchemaAsync(string tableName, DatabaseType dbType)
    {
        return dbType switch
        {
            DatabaseType.SqlServer => await ReadSqlServerSchemaAsync(tableName),
            DatabaseType.MySql => await ReadMySqlSchemaAsync(tableName),
            DatabaseType.PostgreSql => await ReadPostgreSqlSchemaAsync(tableName),
            DatabaseType.SQLite => await ReadSQLiteSchemaAsync(tableName),
            _ => throw new NotSupportedException($"数据库类型 {dbType} 暂不支持")
        };
    }
}
