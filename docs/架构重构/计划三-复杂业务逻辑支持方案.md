// src/SmartAbp.CodeGenerator/Core/BusinessRules/BusinessRuleEngine.cs
public class BusinessRuleEngine : ITransientDependency
{
    /// <summary>
    /// 🧠 业务规则定义和执行引擎
    /// </summary>
    public async Task<BusinessRuleResult> ExecuteRulesAsync(
        string entityName, 
        string operation, 
        object data, 
        List<BusinessRule> rules)
    {
        var result = new BusinessRuleResult();
        var context = new BusinessRuleContext
        {
            EntityName = entityName,
            Operation = operation,
            Data = data,
            ExecutionTime = DateTime.UtcNow
        };

        foreach (var rule in rules.Where(r => r.IsActive))
        {
            try
            {
                var ruleResult = await ExecuteSingleRuleAsync(rule, context);
                result.RuleResults.Add(ruleResult);
                
                if (!ruleResult.Success && rule.IsBlocking)
                {
                    result.Success = false;
                    result.BlockingError = ruleResult.ErrorMessage;
                    break;
                }
            }
            catch (Exception ex)
            {
                result.RuleResults.Add(new SingleRuleResult
                {
                    RuleId = rule.Id,
                    Success = false,
                    ErrorMessage = ex.Message
                });
            }
        }

        return result;
    }

    /// <summary>
    /// 🔧 单个规则执行
    /// </summary>
    private async Task<SingleRuleResult> ExecuteSingleRuleAsync(BusinessRule rule, BusinessRuleContext context)
    {
        // 规则类型分发
        return rule.RuleType switch
        {
            BusinessRuleType.Validation => await ExecuteValidationRule(rule, context),
            BusinessRuleType.Calculation => await ExecuteCalculationRule(rule, context),
            BusinessRuleType.Workflow => await ExecuteWorkflowRule(rule, context),
            BusinessRuleType.Permission => await ExecutePermissionRule(rule, context),
            BusinessRuleType.Audit => await ExecuteAuditRule(rule, context),
            _ => throw new NotSupportedException($"不支持的规则类型: {rule.RuleType}")
        };
    }
}

// 业务规则定义
public class BusinessRule
{
    public string Id { get; set; } = string.Empty;
    public string Name { get; set; } = string.Empty;
    public BusinessRuleType RuleType { get; set; }
    public string Expression { get; set; } = string.Empty; // 规则表达式
    public bool IsActive { get; set; } = true;
    public bool IsBlocking { get; set; } = true; // 是否阻塞操作
    public int Priority { get; set; } = 0; // 执行优先级
    public Dictionary<string, object> Parameters { get; set; } = new();
}

public enum BusinessRuleType
{
    Validation,    // 数据验证规则
    Calculation,   // 计算规则
    Workflow,      // 工作流规则
    Permission,    // 权限规则
    Audit,         // 审计规则
    Notification   // 通知规则
}

            (Pattern: @"^\d+\.\d+$", Type: "Decimal", Confidence: 0.80),

// src/SmartAbp.Vue/packages/lowcode-core/src/events/UnifiedEventSystem.ts
export class UnifiedEventSystem {
  private eventHandlers = new Map<string, EventHandler[]>()
  private globalMiddlewares: EventMiddleware[] = []
  
  /**
   * ⚡ 注册事件处理器
   */
  on<T = any>(eventType: string, handler: EventHandler<T>): void {
    if (!this.eventHandlers.has(eventType)) {
      this.eventHandlers.set(eventType, [])
    }
    this.eventHandlers.get(eventType)!.push(handler)
  }
  
  /**
   * 🚀 触发事件
   */
  async emit<T = any>(eventType: string, data: T, context?: EventContext): Promise<EventResult> {
    const handlers = this.eventHandlers.get(eventType) || []
    const event: EventPayload<T> = {
      type: eventType,
      data,
      context: context || {},
      timestamp: Date.now(),
      id: generateId()
    }
    
    // 🔧 中间件处理
    for (const middleware of this.globalMiddlewares) {
      const result = await middleware.process(event)
      if (result.preventDefault) {
        return { success: false, message: result.message }
      }
    }
    
    // 🎯 事件处理器执行
    const results = await Promise.allSettled(
      handlers.map(handler => handler.handle(event))
    )
    
    const failures = results.filter(r => r.status === 'rejected')
    
    return {
      success: failures.length === 0,
      processedCount: handlers.length,
      failureCount: failures.length,
      message: failures.length > 0 ? `${failures.length} 个处理器执行失败` : '所有处理器执行成功'
    }
  }
}

  'before-${entity.name.toLowerCase()}-update': async (event: EventPayload<{old: ${entity.Name}, new: ${entity.Name}}>) => {
    const { old: oldData, new: newData } = event.data
    
    // 🔍 变更检测
    const changes = detectChanges(oldData, newData)
    console.log('${entity.DisplayName}字段变更:', changes)
    
    // 🔒 权限检查
    await checkUpdatePermissions(oldData, newData, event.context.userId)
  }
}
  `
}
