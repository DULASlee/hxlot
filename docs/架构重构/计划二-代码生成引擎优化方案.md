// src/SmartAbp.CodeGenerator/Core/Templates/AdvancedTemplateCacheService.cs
public class AdvancedTemplateCacheService : ITransientDependency
{
    private readonly ILogger<AdvancedTemplateCacheService> _logger;
    private readonly IMemoryCache _memoryCache;
    private readonly IEmbeddedTemplateExtractor _embeddedExtractor;
    private readonly ConcurrentDictionary<string, TemplateMetadata> _templateIndex;
    private readonly SemaphoreSlim _initializationSemaphore;
    
    public AdvancedTemplateCacheService(
        ILogger<AdvancedTemplateCacheService> logger,
        IMemoryCache memoryCache, 
        IEmbeddedTemplateExtractor embeddedExtractor)
    {
        _logger = logger;
        _memoryCache = memoryCache;
        _embeddedExtractor = embeddedExtractor;
        _templateIndex = new ConcurrentDictionary<string, TemplateMetadata>();
        _initializationSemaphore = new SemaphoreSlim(1, 1);
    }

    /// <summary>
    /// 🚀 模板预加载 - 启动时一次性加载所有模板到内存
    /// 性能目标: 从800ms优化到200ms (75%性能提升)
    /// </summary>
    public async Task<TemplatePreloadResult> PreloadAllTemplatesAsync()
    {
        await _initializationSemaphore.WaitAsync();
        
        try
        {
            var stopwatch = Stopwatch.StartNew();
            _logger.LogInformation("🚀 开始模板预加载...");

            // 1. 获取所有可用模板 (100ms)
            var templatePaths = await _embeddedExtractor.GetAvailableTemplatesAsync();
            
            // 2. 并发加载模板内容 (预计优化到80ms)
            var loadTasks = templatePaths.Select(async templatePath => {
                try
                {
                    var content = await LoadTemplateContentAsync(templatePath);
                    var metadata = ParseTemplateMetadata(content, templatePath);
                    
                    // 缓存模板内容 (24小时过期)
                    var cacheKey = $"template_content:{templatePath}";
                    _memoryCache.Set(cacheKey, content, TimeSpan.FromHours(24));
                    
                    // 缓存模板元数据 (永不过期)
                    var metadataKey = $"template_metadata:{templatePath}";
                    _memoryCache.Set(metadataKey, metadata, TimeSpan.FromDays(7));
                    
                    _templateIndex[templatePath] = metadata;
                    
                    return new TemplateLoadResult { Path = templatePath, Success = true };
                }
                catch (Exception ex)
                {
                    _logger.LogWarning(ex, "模板加载失败: {TemplatePath}", templatePath);
                    return new TemplateLoadResult { Path = templatePath, Success = false, Error = ex.Message };
                }
            });

            var results = await Task.WhenAll(loadTasks);
            
            stopwatch.Stop();
            
            var successCount = results.Count(r => r.Success);
            var failureCount = results.Length - successCount;
            
            _logger.LogInformation("✅ 模板预加载完成: {SuccessCount}/{TotalCount}, 耗时: {ElapsedMs}ms", 
                successCount, results.Length, stopwatch.ElapsedMilliseconds);

            return new TemplatePreloadResult
            {
                TotalTemplates = results.Length,
                SuccessCount = successCount, 
                FailureCount = failureCount,
                ElapsedTime = stopwatch.Elapsed,
                FailedTemplates = results.Where(r => !r.Success).ToList()
            };
        }
        finally
        {
            _initializationSemaphore.Release();
        }
    }

    /// <summary>
    /// 🔧 O(1)模板内容获取 - 从内存缓存获取
    /// </summary>
    public string? GetTemplateContent(string templatePath)
    {
        var cacheKey = $"template_content:{templatePath}";
        var content = _memoryCache.Get<string>(cacheKey);
        
        if (content == null)
        {
            _logger.LogWarning("模板缓存未命中: {TemplatePath}", templatePath);
        }
        
        return content;
    }

    /// <summary>
    /// 🔧 模板元数据快速获取
    /// </summary>
    public TemplateMetadata? GetTemplateMetadata(string templatePath)
    {
        return _templateIndex.TryGetValue(templatePath, out var metadata) ? metadata : null;
    }

    /// <summary>
    /// 🔍 模板依赖分析
    /// </summary>
    private TemplateMetadata ParseTemplateMetadata(string content, string templatePath)
    {
        var metadata = new TemplateMetadata
        {
            Path = templatePath,
            Size = content.Length,
            LastModified = DateTime.UtcNow,
            Variables = ExtractTemplateVariables(content),
            Dependencies = ExtractTemplateDependencies(content),
            Category = InferTemplateCategory(templatePath),
            Complexity = CalculateTemplateComplexity(content)
        };
        
        return metadata;
    }

    /// <summary>
    /// 🔧 变量提取 - 预编译正则表达式
    /// </summary>
    private static readonly Regex VariablePattern = new(@"{{(\w+)}}", RegexOptions.Compiled);
    
    private List<string> ExtractTemplateVariables(string content)
    {
        var matches = VariablePattern.Matches(content);
        return matches.Select(m => m.Groups[1].Value).Distinct().ToList();
    }
}

// src/SmartAbp.CodeGenerator/Core/Templates/OptimizedVariableReplacer.cs
public class OptimizedVariableReplacer : ITransientDependency
{
    private readonly ILogger<OptimizedVariableReplacer> _logger;
    private readonly ConcurrentDictionary<string, Func<object, string>> _variableProcessors;
    
    // 🔥 预编译正则表达式 - 避免运行时编译开销
    private static readonly Regex VariablePattern = new(@"{{(\w+)}}", RegexOptions.Compiled);
    
    public OptimizedVariableReplacer(ILogger<OptimizedVariableReplacer> logger)
    {
        _logger = logger;
        _variableProcessors = new ConcurrentDictionary<string, Func<object, string>>();
        InitializeStandardProcessors();
    }

    /// <summary>
    /// 🚀 高性能变量替换 - O(n)复杂度算法
    /// 性能目标: 从1200ms优化到300ms (75%性能提升)
    /// </summary>
    public string ReplaceVariablesOptimized(string template, ModuleMetadataDto metadata, EnhancedEntityModelDto? entity = null)
    {
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            // 1. 构建变量字典 (50ms)
            var variables = BuildVariableDictionary(metadata, entity);
            
            // 2. 单次扫描替换 - O(n)复杂度 (200ms)
            var result = VariablePattern.Replace(template, match => {
                var variableName = match.Groups[1].Value;
                
                if (variables.TryGetValue(variableName, out var value))
                {
                    return ProcessVariableValue(variableName, value);
                }
                
                _logger.LogWarning("未找到变量: {VariableName}", variableName);
                return match.Value; // 保留原始占位符
            });
            
            stopwatch.Stop();
            _logger.LogDebug("变量替换完成: {ElapsedMs}ms, 模板长度: {Length}", 
                stopwatch.ElapsedMilliseconds, template.Length);
            
            return result;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "变量替换失败");
            throw;
        }
    }

    /// <summary>
    /// 🚀 批量模板并发替换
    /// </summary>
    public async Task<Dictionary<string, string>> ReplaceVariablesBatchAsync(
        Dictionary<string, string> templates, 
        ModuleMetadataDto metadata,
        EnhancedEntityModelDto? entity = null)
    {
        var results = new ConcurrentDictionary<string, string>();
        var variables = BuildVariableDictionary(metadata, entity);
        
        // 🔥 并发处理多个模板 - 5个并发任务
        var semaphore = new SemaphoreSlim(5);
        var tasks = templates.Select(async kvp => {
            await semaphore.WaitAsync();
            try
            {
                var result = ReplaceVariablesOptimized(kvp.Value, metadata, entity);
                results[kvp.Key] = result;
            }
            finally
            {
                semaphore.Release();
            }
        });
        
        await Task.WhenAll(tasks);
        
        return results.ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
    }

    /// <summary>
    /// 🔧 高性能变量字典构建
    /// </summary>
    private Dictionary<string, object> BuildVariableDictionary(ModuleMetadataDto metadata, EnhancedEntityModelDto? entity)
    {
        var variables = new Dictionary<string, object>(StringComparer.OrdinalIgnoreCase)
        {
            // 模块级变量
            ["ModuleName"] = metadata.Name,
            ["ModuleDisplayName"] = metadata.DisplayName,
            ["ModuleNamespace"] = metadata.Namespace,
            ["ModuleDescription"] = metadata.Description,
            ["Author"] = metadata.Author,
            
            // 系统变量
            ["SystemName"] = metadata.SystemName,
            ["GeneratedDate"] = DateTime.UtcNow.ToString("yyyy-MM-dd HH:mm:ss"),
            ["GeneratorVersion"] = "2.0.0"
        };

        // 实体级变量
        if (entity != null)
        {
            variables["EntityName"] = entity.Name;
            variables["EntityDisplayName"] = entity.DisplayName;
            variables["entityName"] = entity.Name.ToCamelCase();
            variables["entity_name"] = entity.Name.ToSnakeCase();
            variables["ENTITY_NAME"] = entity.Name.ToUpperInvariant();
            variables["EntityDescription"] = entity.Description;
            variables["TableName"] = entity.TableName ?? $"{entity.Name}s";
            
            // 字段相关变量
            if (entity.Properties?.Any() == true)
            {
                variables["PrimaryKeyType"] = entity.Properties.FirstOrDefault(p => p.IsKey)?.Type ?? "Guid";
                variables["PropertyCount"] = entity.Properties.Count;
                variables["RequiredPropertyCount"] = entity.Properties.Count(p => p.IsRequired);
            }
        }

        return variables;
    }

    /// <summary>
    /// 🔧 变量值处理器 - 支持复杂变量处理
    /// </summary>
    private string ProcessVariableValue(string variableName, object value)
    {
        // 使用预注册的处理器
        if (_variableProcessors.TryGetValue(variableName, out var processor))
        {
            return processor(value);
        }
        
        // 默认字符串转换
        return value?.ToString() ?? string.Empty;
    }

    /// <summary>
    /// 🔧 标准变量处理器初始化
    /// </summary>
    private void InitializeStandardProcessors()
    {
        // 命名转换处理器
        _variableProcessors["entityName"] = value => value.ToString()?.ToCamelCase() ?? "";
        _variableProcessors["entity_name"] = value => value.ToString()?.ToSnakeCase() ?? "";
        _variableProcessors["ENTITY_NAME"] = value => value.ToString()?.ToUpperInvariant() ?? "";
        
        // 路径转换处理器
        _variableProcessors["EntityPath"] = value => $"/{value.ToString()?.ToLowerInvariant()}";
        _variableProcessors["EntityRoute"] = value => value.ToString()?.ToLowerInvariant()?.Replace(" ", "-") ?? "";
        
        // 权限转换处理器
        _variableProcessors["EntityPermission"] = value => $"SmartAbp.{value}.Default";
        _variableProcessors["CreatePermission"] = value => $"SmartAbp.{value}.Create";
        _variableProcessors["EditPermission"] = value => $"SmartAbp.{value}.Edit";
        _variableProcessors["DeletePermission"] = value => $"SmartAbp.{value}.Delete";
    }
}

// 支持扩展方法
public static class StringExtensions
{
    public static string ToCamelCase(this string str)
    {
        if (string.IsNullOrEmpty(str)) return str;
        return char.ToLowerInvariant(str[0]) + str[1..];
    }
    
    public static string ToSnakeCase(this string str)
    {
        if (string.IsNullOrEmpty(str)) return str;
        return Regex.Replace(str, "([a-z0-9])([A-Z])", "$1_$2").ToLowerInvariant();
    }
}

// src/SmartAbp.CodeGenerator/Core/FileOperations/ConcurrentFileWriteEngine.cs
public class ConcurrentFileWriteEngine : ITransientDependency
{
    private readonly ILogger<ConcurrentFileWriteEngine> _logger;
    private readonly AtomicFileWriter _atomicWriter;
    private readonly SemaphoreSlim _concurrencyLimiter;
    private readonly ConcurrentQueue<FileWriteTask> _writeQueue;
    
    public ConcurrentFileWriteEngine(
        ILogger<ConcurrentFileWriteEngine> logger,
        AtomicFileWriter atomicWriter)
    {
        _logger = logger;
        _atomicWriter = atomicWriter;
        _concurrencyLimiter = new SemaphoreSlim(5); // 最多5个并发写入
        _writeQueue = new ConcurrentQueue<FileWriteTask>();
    }

    /// <summary>
    /// 🚀 并发文件写入 - 性能目标: 900ms → 150ms
    /// </summary>
    public async Task<FileWriteResult> WriteConcurrentlyAsync(Dictionary<string, string> filesToWrite)
    {
        var stopwatch = Stopwatch.StartNew();
        var result = new FileWriteResult();
        
        try
        {
            _logger.LogInformation("🚀 开始并发文件写入: {FileCount}个文件", filesToWrite.Count);

            // 1. 并发冲突检测 (100ms)
            var conflictTasks = filesToWrite.Select(async kvp => {
                return new {
                    Path = kvp.Key,
                    Content = kvp.Value,
                    ConflictType = await _atomicWriter.DetectConflictAsync(kvp.Key, kvp.Value)
                };
            });
            
            var conflictResults = await Task.WhenAll(conflictTasks);
            
            // 2. 分组处理不同冲突类型 (50ms)
            var noConflictFiles = conflictResults.Where(r => r.ConflictType == ConflictType.NoConflict).ToList();
            var conflictFiles = conflictResults.Where(r => r.ConflictType != ConflictType.NoConflict).ToList();
            
            // 3. 并发写入无冲突文件 (预计100ms)
            var writeTasks = noConflictFiles.Select(async file => {
                await _concurrencyLimiter.WaitAsync();
                try
                {
                    await WriteFileSafelyAsync(file.Path, file.Content);
                    return new FileWriteResult.FileResult
                    {
                        Path = file.Path,
                        Success = true,
                        Size = file.Content.Length
                    };
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "文件写入失败: {FilePath}", file.Path);
                    return new FileWriteResult.FileResult
                    {
                        Path = file.Path,
                        Success = false,
                        Error = ex.Message
                    };
                }
                finally
                {
                    _concurrencyLimiter.Release();
                }
            });

            var writeResults = await Task.WhenAll(writeTasks);
            
            // 4. 处理冲突文件 (根据策略串行处理)
            foreach (var conflictFile in conflictFiles)
            {
                var conflictResult = await _atomicWriter.ResolveConflictAsync(
                    conflictFile.Path, 
                    conflictFile.Content, 
                    ConflictResolutionStrategy.Backup);
                    
                result.ConflictResults.Add(conflictResult);
            }
            
            stopwatch.Stop();
            
            result.TotalFiles = filesToWrite.Count;
            result.SuccessCount = writeResults.Count(r => r.Success);
            result.FailureCount = writeResults.Length - result.SuccessCount;
            result.ConflictCount = conflictFiles.Count;
            result.ElapsedTime = stopwatch.Elapsed;
            result.FileResults = writeResults.ToList();
            
            _logger.LogInformation("✅ 并发文件写入完成: {SuccessCount}/{TotalCount}, 耗时: {ElapsedMs}ms, 冲突: {ConflictCount}个",
                result.SuccessCount, result.TotalFiles, stopwatch.ElapsedMilliseconds, result.ConflictCount);
                
            return result;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "并发文件写入异常");
            throw;
        }
    }

    /// <summary>
    /// 🔒 原子性安全文件写入
    /// </summary>
    private async Task WriteFileSafelyAsync(string filePath, string content)
    {
        // 确保目录存在
        var directory = Path.GetDirectoryName(filePath);
        if (!string.IsNullOrEmpty(directory) && !Directory.Exists(directory))
        {
            Directory.CreateDirectory(directory);
        }
        
        // 原子性写入 (临时文件 → 重命名)
        var tempPath = $"{filePath}.tmp{Guid.NewGuid():N}";
        
        try
        {
            await File.WriteAllTextAsync(tempPath, content, Encoding.UTF8);
            File.Move(tempPath, filePath); // 原子性操作
        }
        finally
        {
            // 清理临时文件
            if (File.Exists(tempPath))
            {
                try { File.Delete(tempPath); } catch { /* 忽略清理错误 */ }
            }

public class OptimizedFileConflictDetector : ITransientDependency
{
    private readonly ILogger<OptimizedFileConflictDetector> _logger;
    
    /// <summary>
    /// 🚀 快速文件指纹计算 - 基于SHA256哈希
    /// </summary>
    public string CalculateFileFingerprint(string content)
    {
        using var sha256 = SHA256.Create();
        var hash = sha256.ComputeHash(Encoding.UTF8.GetBytes(content));
        return Convert.ToBase64String(hash)[..16]; // 取前16位，平衡性能和唯一性
    }
    
    /// <summary>
    /// 🚀 批量冲突检测 - 并发处理
    /// </summary>
    public async Task<Dictionary<string, ConflictDetectionResult>> DetectConflictsBatchAsync(
        Dictionary<string, string> filesToCheck)
    {
        var results = new ConcurrentDictionary<string, ConflictDetectionResult>();
        
        // 🔥 并发检测 - 最多5个并发任务
        var semaphore = new SemaphoreSlim(5);
        var tasks = filesToCheck.Select(async kvp => {
            await semaphore.WaitAsync();
            try
            {
                var result = await DetectSingleFileConflictAsync(kvp.Key, kvp.Value);
                results[kvp.Key] = result;
            }
            finally
            {
                semaphore.Release();
            }
        });
        
        await Task.WhenAll(tasks);
        
        return results.ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
    }
    
    /// <summary>
    /// 🔧 单文件冲突检测
    /// </summary>
    private async Task<ConflictDetectionResult> DetectSingleFileConflictAsync(string filePath, string newContent)
    {
        if (!File.Exists(filePath))
        {
            return new ConflictDetectionResult
            {
                ConflictType = ConflictType.NoConflict,
                Reason = "文件不存在，可以直接创建"
            };
        }
        
        var existingContent = await File.ReadAllTextAsync(filePath, Encoding.UTF8);
        var existingFingerprint = CalculateFileFingerprint(existingContent);
        var newFingerprint = CalculateFileFingerprint(newContent);
        
        if (existingFingerprint == newFingerprint)
        {
            return new ConflictDetectionResult
            {
                ConflictType = ConflictType.NoConflict,
                Reason = "文件内容相同，无需更新"
            };
        }
        
        // 🔍 智能冲突分析
        var conflictAnalysis = AnalyzeContentConflict(existingContent, newContent);
        
        return new ConflictDetectionResult
        {
            ConflictType = ConflictType.ContentDifferent,
            Reason = "文件内容不同",
            ConflictAnalysis = conflictAnalysis,
            RecommendedStrategy = DetermineRecommendedStrategy(conflictAnalysis)
        };
    }
    
    /// <summary>
    /// 🧠 内容冲突智能分析
    /// </summary>
    private ConflictAnalysis AnalyzeContentConflict(string existingContent, string newContent)
    {
        var analysis = new ConflictAnalysis();
        
        // 检查是否为生成代码更新
        if (existingContent.Contains("// AUTO-GENERATED FILE") && 
            newContent.Contains("// AUTO-GENERATED FILE"))
        {
            analysis.IsGeneratedCodeUpdate = true;
            analysis.RecommendedAction = "自动覆盖生成代码";
        }
        
        // 检查是否包含手动修改
        if (existingContent.Contains("// CUSTOM:") || 
            existingContent.Contains("// TODO:") ||
            existingContent.Contains("// MODIFIED:"))
        {
            analysis.HasManualModifications = true;
            analysis.RecommendedAction = "备份后合并";
        }
        
        // 计算内容差异程度
        var diffLines = CalculateLineDifferences(existingContent, newContent);
        analysis.DifferencePercentage = diffLines.DifferencePercentage;
        analysis.AddedLines = diffLines.AddedLines;
        analysis.DeletedLines = diffLines.DeletedLines;
        analysis.ModifiedLines = diffLines.ModifiedLines;
        
        return analysis;
    }

        sb.AppendLine($"    // 🚀 高性能分页查询 - 先计数后查询，避免重复计算");
        sb.AppendLine($"    ");
        sb.AppendLine($"    // 1. 应用筛选条件");
        
        // 生成筛选条件
        var searchableProperties = entity.Properties.Where(p => 
            p.Type == "string" || p.Type == "int" || p.Type == "DateTime").ToList();
            
        foreach (var prop in searchableProperties)
        {
            if (prop.Type == "string")
            {
                sb.AppendLine($"    query = query.WhereIf(!string.IsNullOrEmpty(input.{prop.Name}), x => x.{prop.Name}.Contains(input.{prop.Name}));");
            }
            else if (prop.Type == "int")
            {
                sb.AppendLine($"    query = query.WhereIf(input.{prop.Name}.HasValue, x => x.{prop.Name} == input.{prop.Name}.Value);");
            }
            else if (prop.Type == "DateTime")
            {
                sb.AppendLine($"    query = query.WhereIf(input.{prop.Name}Start.HasValue, x => x.{prop.Name} >= input.{prop.Name}Start.Value);");
                sb.AppendLine($"    query = query.WhereIf(input.{prop.Name}End.HasValue, x => x.{prop.Name} <= input.{prop.Name}End.Value);");
