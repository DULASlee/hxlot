export class DynamicRouteLoader {
  async loadUserRoutes(): Promise<RouteRecordRaw[]> {
    // 1. 调用后端API获取用户菜单
    const userMenus = await menuApi.getUserMenus()
    
    // 2. 转换为Vue Router格式
    const routes = this.convertToVueRoutes(userMenus)
    
    // 3. 动态注册到路由器
    routes.forEach(route => router.addRoute(route))
    
    return routes
  }
}

### 菜单权限算法

```typescript
class MenuPermissionChecker {
  async checkMenuAccess(menuId: string, userPermissions: string[]): Promise<boolean> {
    const menu = await menuRepository.findById(menuId)
    
    // 1. 检查菜单必需权限
    if (menu.requiredPermissions?.length > 0) {
      const hasPermissions = menu.permissionMode === 'All' 
        ? menu.requiredPermissions.every(p => userPermissions.includes(p))
        : menu.requiredPermissions.some(p => userPermissions.includes(p))
      
      if (!hasPermissions) return false
    }
    
    // 2. 检查父菜单权限（递归）
    if (menu.parentId) {
      return await this.checkMenuAccess(menu.parentId, userPermissions)
    }
    
    return true
  }
}
```

### 缓存策略

```typescript
class MenuCacheManager {
  private cacheKey = (userId: string) => `user_menus_${userId}`
  
  async getCachedUserMenus(userId: string): Promise<UserMenuDto[]> {
    // 1. Redis缓存（5分钟TTL）
    const cached = await redis.get(this.cacheKey(userId))
    if (cached) return JSON.parse(cached)
    
    // 2. 数据库查询
    const menus = await menuService.getUserMenusFromDb(userId)
    
    // 3. 写入缓存
    await redis.setex(this.cacheKey(userId), 300, JSON.stringify(menus))
    
    return menus
  }
}
```
