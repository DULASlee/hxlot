## 🎯 执行摘要

**核心目标**: 建立基于`@smartabp/metadata-core`的统一类型注册管理系统，强制约束所有前端组件使用统一的元数据定义，消除类型定义分散、重复、冲突等架构债务。

**预期收益**:
- ✅ 类型定义重复率从40% → 0%（-100%）
- ✅ 类型错误数量从53个 → 0个（-100%）
- ✅ 维护成本降低70%
- ✅ 开发效率提升50%
- ✅ 代码质量评分从75分 → 95分（+27%）

**实施策略**: 渐进式重构，分8周4个阶段完成，零停机、零业务影响。

## 🔬 二、第一性原理分析

### 2.1 基本真理

**真理1: 低代码平台 = 元数据驱动**
```
低代码平台本质 = 将业务逻辑抽象为元数据 + 运行时引擎解释执行元数据

因此：
  - UI组件 = 元数据描述
  - 表单字段 = 元数据定义
  - 配置项 = 元数据管理
  - 业务规则 = 元数据表达
```

**真理2: 分散定义 = 架构债务**
```
分散类型定义 → 重复维护 → 不一致 → BUG → 维护成本指数级增长

数学证明:
  设类型数量为 n，定义位置数为 m
  维护成本 = O(n × m)
  
  当前：n=50, m=8 → 成本 = 400单位
  统一后：n=50, m=1 → 成本 = 50单位
  降低: 87.5%
```

**真理3: 统一管理 = 能力复用**
```
统一类型注册中心 → 自动获得:
  ✅ 验证能力（@metadata-core/MetadataValidator）
  ✅ 版本管理（@metadata-core/SchemaVersionManager）
  ✅ 兼容性检查（@metadata-core/CompatibilityChecker）
  ✅ 国际化（@metadata-core/i18n）
  ✅ 格式转换（@metadata-core/SchemaConverter）
```

### 2.2 推导结论

**结论1**: **必须建立统一类型注册中心**

**结论2**: **必须强制所有组件使用统一类型**（通过ESLint规则）

**结论3**: **必须复用@metadata-core已有能力**（避免重复造轮子）

### 4.2 核心组件设计

#### 4.2.1 TypeRegistry（类型注册中心）

**文件位置**: `packages/metadata-core/src/registry/TypeRegistry.ts`

**核心功能**:
```typescript
/**
 * 统一类型注册中心
 * 
 * @description
 * 参考业界最佳实践：
 * - Salesforce Metadata API: 强制注册机制
 * - OutSystems Type System: 中央类型仓库
 * - Mendix Domain Model: 统一数据模型
 * - Power Platform Dataverse: 版本控制
 * 
 * @example
 * ```typescript
 * // 注册类型
 * const registry = TypeRegistry.getInstance()
 * registry.registerType(SmartFormMetadata)
 * 
 * // 获取类型
 * const metadata = registry.getType('SmartForm')
 * 
 * // 创建实例（自动验证）
 * const formField = registry.createInstance<FormField>('SmartForm.FormField', data)
 * ```
 */
export class TypeRegistry {
  private static instance: TypeRegistry
  private types: Map<string, ModuleMetadata> = new Map()
  private validator: MetadataValidator
  private versionManager: SchemaVersionManager
  private compatibilityChecker: CompatibilityChecker
  private i18nManager: I18nManager
  
  private constructor() {
    this.validator = new MetadataValidator()
    this.versionManager = new SchemaVersionManager()
    this.compatibilityChecker = new CompatibilityChecker()
    this.i18nManager = new I18nManager()
  }
  
  /**
   * 获取单例实例
   */
  static getInstance(): TypeRegistry {
    if (!TypeRegistry.instance) {
      TypeRegistry.instance = new TypeRegistry()
    }
    return TypeRegistry.instance
  }
  
  /**
   * 注册类型（强制）
   * 
   * @param metadata - 模块元数据
   * @throws {TypeError} 验证失败时抛出
   * 
   * @example
   * ```typescript
   * registry.registerType({
   *   moduleName: 'SmartForm',
   *   version: '2.0.0',
   *   entities: [...]
   * })
   * ```
   */
  registerType(metadata: ModuleMetadata): void {
    // 1. 验证元数据格式
    const validationResult = this.validator.validate(metadata)
    if (!validationResult.isValid) {
      throw new TypeError(
        `Type validation failed for '${metadata.moduleName}': ${validationResult.errors.join(', ')}`
      )
    }
    
    // 2. 版本兼容性检查
    if (this.types.has(metadata.moduleName)) {
      const existing = this.types.get(metadata.moduleName)!
      const compatResult = this.compatibilityChecker.check(existing, metadata)
      
      if (!compatResult.isCompatible) {
        console.warn(
          `⚠️ Breaking changes detected in '${metadata.moduleName}':`,
          compatResult.breakingChanges
        )
      }
      
      // 3. 版本管理
      this.versionManager.recordVersion(metadata)
    }
    
    // 4. 注册到中央仓库
    this.types.set(metadata.moduleName, metadata)
    
    console.info(
      `✅ Type registered: ${metadata.moduleName} v${metadata.version}`
    )
  }
  
  /**
   * 获取类型定义
   * 
   * @param moduleName - 模块名称
   * @returns 模块元数据
   * @throws {Error} 类型未注册时抛出
   * 
   * @example
   * ```typescript
   * const formMetadata = registry.getType('SmartForm')
   * ```
   */
  getType(moduleName: string): ModuleMetadata {
    const type = this.types.get(moduleName)
    if (!type) {
      throw new Error(
        `Type '${moduleName}' is not registered. ` +
        `Please register it first using TypeRegistry.getInstance().registerType()`
      )
    }
    return type
  }
  
  /**
   * 创建类型实例（自动验证）
   * 
   * @param typePath - 类型路径，格式: 'ModuleName' 或 'ModuleName.EntityName'
   * @param data - 实例数据
   * @returns 验证后的实例
   * @throws {TypeError} 数据验证失败时抛出
   * 
   * @example
   * ```typescript
   * const formField = registry.createInstance<FormField>(
   *   'SmartForm.FormField',
   *   { field: 'name', title: '姓名', type: 'input' }
   * )
   * ```
   */
  createInstance<T>(typePath: string, data: Partial<T>): T {
    const [moduleName, entityName] = typePath.split('.')
    const metadata = this.getType(moduleName)
    
    // 查找实体定义
    const entity = entityName 
      ? metadata.entities.find(e => e.entityName === entityName)
      : metadata.entities[0]
    
    if (!entity) {
      throw new Error(
        `Entity '${entityName}' not found in module '${moduleName}'`
      )
    }
    
    // 自动验证数据
    const validationResult = this.validator.validateData(entity, data)
    if (!validationResult.isValid) {
      const errors = validationResult.errors.map(err =>
        this.i18nManager.translate(err.message)
      )
      throw new TypeError(
        `Data validation failed for '${typePath}': ${errors.join(', ')}`
      )
    }
    
    return data as T
  }
  
  /**
   * 列出所有已注册类型
   * 
   * @returns 类型名称列表
   */
  listTypes(): string[] {
    return Array.from(this.types.keys())
  }
  
  /**
   * 检查类型是否已注册
   * 
   * @param moduleName - 模块名称
   * @returns 是否已注册
   */
  isRegistered(moduleName: string): boolean {
    return this.types.has(moduleName)
  }
  
  /**
   * 获取类型版本历史
   * 
   * @param moduleName - 模块名称
   * @returns 版本历史
   */
  getVersionHistory(moduleName: string): string[] {
    return this.versionManager.getHistory(moduleName)
  }
  
  /**
   * 生成类型文档
   * 
   * @param moduleName - 模块名称
   * @returns Markdown格式文档
   */
  generateDocs(moduleName: string): string {
    const metadata = this.getType(moduleName)
    return this.generateMarkdownDocs(metadata)
  }
  
  private generateMarkdownDocs(metadata: ModuleMetadata): string {
    // 自动生成类型文档
    return `
# ${metadata.moduleName} v${metadata.version}

## 实体定义

${metadata.entities.map(entity => `
### ${entity.entityName}

| 属性 | 类型 | 必填 | 描述 |
|------|------|------|------|
${entity.properties.map(prop => 
  `| ${prop.name} | ${prop.type} | ${prop.required ? '✅' : '❌'} | ${prop.description || '-'} |`

/**
 * ESLint规则：强制使用TypeRegistry
 * 
 * @description
 * 确保所有组件类型都通过TypeRegistry注册
 * 禁止直接定义interface/type而不注册
 */

import { ESLintUtils } from '@typescript-eslint/utils'
import { TypeRegistry } from '@smartabp/metadata-core'

const createRule = ESLintUtils.RuleCreator(
  name => `https://smartabp.com/eslint-rules/${name}`
)

export const enforceTypeRegistry = createRule({
  name: 'enforce-type-registry',
  meta: {
    type: 'problem',
    docs: {
      description: '强制所有组件类型使用TypeRegistry注册',
      recommended: 'error'
    },
    messages: {
      unregisteredType: 
        "Type '{{name}}' must be registered in TypeRegistry. " +
        "Please call TypeRegistry.getInstance().registerType() in type-registry.ts",
      directTypeDefinition:
        "Direct type definition is not allowed for component types. " +
        "Please use TypeRegistry to manage types."
    },
    schema: []
  },
  defaultOptions: [],
  
  create(context) {
    const registry = TypeRegistry.getInstance()
    
    return {
      // 检测interface定义
      TSInterfaceDeclaration(node) {
        const interfaceName = node.id.name
        
        // 允许的例外（内部类型）
        if (
          interfaceName.startsWith('_') ||
          interfaceName.endsWith('Props') ||
          interfaceName.endsWith('State')
        ) {
          return
        }
        
        // 检查是否已注册
        if (!registry.isRegistered(interfaceName)) {
          context.report({
            node,
            messageId: 'unregisteredType',
            data: { name: interfaceName }
          })
        }
      },
      
      // 检测type定义
      TSTypeAliasDeclaration(node) {
        const typeName = node.id.name
        
        // 允许的例外
        if (typeName.startsWith('_') || typeName.endsWith('Type')) {
          return
        }
        
        if (!registry.isRegistered(typeName)) {
          context.report({
            node,
            messageId: 'unregisteredType',
            data: { name: typeName }
          })
        }
      }
    }
  }
})

#### 4.2.4 自动迁移工具

**文件位置**: `scripts/type-migration/migrate-to-registry.ts`

```typescript
/**
 * 自动迁移工具：将现有类型迁移到TypeRegistry
 * 
 * @description
 * 扫描项目中的所有类型定义，自动生成TypeRegistry注册代码
 * 
 * @example
 * ```bash
 * npm run migrate-types
 * ```
 */

import * as ts from 'typescript'
import { glob } from 'glob'
import { readFile, writeFile } from 'fs/promises'
import path from 'path'

interface ExtractedType {
  name: string
  filePath: string
  properties: Array<{
    name: string
    type: string
    required: boolean
    description?: string
  }>
}

/**
 * 提取TypeScript接口定义
 */
function extractInterfaces(sourceCode: string, filePath: string): ExtractedType[] {
  const sourceFile = ts.createSourceFile(
    filePath,
    sourceCode,
    ts.ScriptTarget.Latest,
    true
  )
  
  const types: ExtractedType[] = []
  
  function visit(node: ts.Node) {
    if (ts.isInterfaceDeclaration(node)) {
      const interfaceName = node.name.text
      const properties: ExtractedType['properties'] = []
      
      node.members.forEach(member => {
        if (ts.isPropertySignature(member) && member.name) {
          const propName = member.name.getText(sourceFile)
          const propType = member.type?.getText(sourceFile) || 'any'
          const isRequired = !member.questionToken
          
          // 提取JSDoc注释
          const jsDocComment = ts.getJSDocCommentsAndTags(member)
          const description = jsDocComment.length > 0
            ? jsDocComment[0].comment?.toString()
            : undefined
          
          properties.push({
            name: propName,
            type: propType,
            required: isRequired,
            description
          })
        }
      })
      
      types.push({
        name: interfaceName,
        filePath,
        properties
      })
    }
    
    ts.forEachChild(node, visit)
  }
  
  visit(sourceFile)
  return types
}

/**
 * 转换为ModuleMetadata格式
 */
function convertToMetadata(types: ExtractedType[]): string {
  const metadataDefinitions = types.map(type => `
export const ${type.name}Metadata: ModuleMetadata = {
  moduleName: '${type.name}',
  version: '1.0.0',
  entities: [{
    entityName: '${type.name}',
    properties: [
${type.properties.map(prop => `      {
        name: '${prop.name}',
        type: '${prop.type}',
        required: ${prop.required}${prop.description ? `,\n        description: '${prop.description}'` : ''}
      }`).join(',\n')}
    ]
  }]
}

// 注册${type.name}类型
TypeRegistry.getInstance().registerType(${type.name}Metadata)
`).join('\n')
  
  return `
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 自动生成的类型注册文件
// 生成时间: ${new Date().toISOString()}
// 警告: 请勿手动编辑此文件
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

import { TypeRegistry } from '@smartabp/metadata-core'
import type { ModuleMetadata } from '@smartabp/metadata-core'

${metadataDefinitions}
  `.trim()
}

/**
 * 主迁移函数
 */
async function migrateToTypeRegistry() {
  console.log('🔄 开始类型迁移...\n')
  
  // 1. 扫描所有.ts/.vue文件
  const files = await glob('src/**/*.{ts,vue}', {
    ignore: ['**/node_modules/**', '**/dist/**']
  })
  
  console.log(`📁 发现 ${files.length} 个文件`)
  
  // 2. 提取接口定义
  const allTypes: ExtractedType[] = []
  
  for (const file of files) {
    const content = await readFile(file, 'utf-8')
    const types = extractInterfaces(content, file)
    
    if (types.length > 0) {
      console.log(`   ✅ ${file}: 发现 ${types.length} 个类型`)
      allTypes.push(...types)
    }
  }
  
  console.log(`\n📊 共发现 ${allTypes.length} 个类型定义`)
  
  // 3. 转换为ModuleMetadata
  const registrationCode = convertToMetadata(allTypes)
  
  // 4. 写入type-registry.ts
  const outputPath = path.join(
    process.cwd(),
    'packages/lowcode-core/src/types/type-registry-generated.ts'
  )
  
  await writeFile(outputPath, registrationCode, 'utf-8')
  
  console.log(`\n✅ 迁移完成！`)
  console.log(`📝 注册代码已写入: ${outputPath}`)
  console.log(`\n📋 下一步：`)
  console.log(`   1. 检查生成的代码`)
  console.log(`   2. 手动调整验证规则`)
  console.log(`   3. 在main.ts中导入并注册`)
}

// 执行迁移
migrateToTypeRegistry().catch(console.error)
```

// packages/lowcode-core/src/generators/aspire/ConfigurationCenterEnhancer.ts

import { TypeRegistry } from '@smartabp/metadata-core'

export class ConfigurationCenterEnhancer {
  private registry = TypeRegistry.getInstance()
  
  async setConfiguration(key: string, value: string, environment: string) {
    // ✅ 使用TypeRegistry自动验证
    const configItem = this.registry.createInstance<ConfigurationItem>(
      'ConfigurationCenter.ConfigurationItem',
      {
        key,
        value,
        environment,
        dataType: this.detectDataType(value),
        sensitive: this.isSensitive(key),
        hotReload: true
      }
    )
    
    // 保存到配置中心
    await this.save(configItem)
  }
  
  private detectDataType(value: string): string {
    // 自动检测数据类型
    if (!isNaN(Number(value))) return 'number'
    if (value === 'true' || value === 'false') return 'boolean'
    if (value.startsWith('{') || value.startsWith('[')) return 'json'
    return 'string'
  }
  
  private isSensitive(key: string): boolean {
    // 检测是否为敏感信息
    const sensitiveKeywords = ['password', 'secret', 'token', 'key', 'apikey']
    return sensitiveKeywords.some(keyword => 
      key.toLowerCase().includes(keyword)
    )
  }
}
