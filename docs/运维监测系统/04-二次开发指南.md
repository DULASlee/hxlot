### 2.3 实现应用服务

在 `Application/Services/` 创建应用服务：

```csharp
using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using Volo.Abp.Application.Services;

namespace SmartAbp.OpsManagement.Services
{
    public class NetworkAppService : ApplicationService
    {
        private readonly INetworkMetricRepository _repository;
        
        public NetworkAppService(INetworkMetricRepository repository)
        {
            _repository = repository;
        }
        
        public async Task<NetworkMetricDto> GetRealtimeMetricAsync(string serviceName)
        {
            var metric = await _repository.GetLatestAsync(serviceName);
            return ObjectMapper.Map<NetworkMetric, NetworkMetricDto>(metric);
        }
        
        public async Task<List<NetworkMetricDto>> GetHistoryAsync(
            NetworkMetricSearchRequest request)
        {
            var metrics = await _repository.GetListAsync(
                request.ServiceName,
                request.StartTime,
                request.EndTime
            );
            return ObjectMapper.Map<List<NetworkMetric>, List<NetworkMetricDto>>(metrics);
        }
    }
}
```

### 4.2 实现数据源集成

在 `Infrastructure/CustomMetric/` 实现集成：

```csharp
using System;
using System.Net.Http;
using System.Threading.Tasks;

namespace SmartAbp.OpsManagement.Infrastructure.CustomMetric
{
    public class CustomMetricService : ICustomMetricService
    {
        private readonly HttpClient _httpClient;
        private readonly ILogger<CustomMetricService> _logger;
        
        public CustomMetricService(
            HttpClient httpClient,
            ILogger<CustomMetricService> logger)
        {
            _httpClient = httpClient;
            _logger = logger;
        }
        
        public async Task<Dictionary<string, object>> GetMetricsAsync(
            string serviceName,
            DateTime startTime,
            DateTime endTime)
        {
            try
            {
                var response = await _httpClient.GetAsync(
                    $"/api/metrics?service={serviceName}&start={startTime}&end={endTime}"
                );
                response.EnsureSuccessStatusCode();
                
                var data = await response.Content.ReadFromJsonAsync<
                    Dictionary<string, object>>();
                return data ?? new Dictionary<string, object>();
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to get custom metrics");
                throw;
            }
        }
        
        public async Task<bool> PushMetricAsync(string metricName, double value)
        {
            try
            {
                var content = JsonContent.Create(new { metricName, value });
                var response = await _httpClient.PostAsync("/api/metrics", content);
                return response.IsSuccessStatusCode;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to push custom metric");
                return false;
            }
        }
    }
}
```

### 5.1 添加自定义告警规则类型

```csharp
public class AlertRule : AggregateRoot<Guid>
{
    // 现有属性...
    
    /// <summary>
    /// 自定义规则表达式（支持复杂条件）
    /// 例如: "CpuUsage > 80 AND MemoryUsage > 70"
    /// </summary>
    public string? CustomExpression { get; set; }
    
    /// <summary>
    /// 评估自定义表达式
    /// </summary>
    public bool EvaluateCustomExpression(Dictionary<string, double> metrics)
    {
        if (string.IsNullOrEmpty(CustomExpression))
            return ShouldTrigger(metrics[MetricType]);
        
        // 使用表达式引擎评估（如DynamicExpresso）
        var interpreter = new Interpreter();
        foreach (var metric in metrics)
        {
            interpreter.SetVariable(metric.Key, metric.Value);
        }
        
        return interpreter.Eval<bool>(CustomExpression);
    }
}
```

### 6.1 数据库查询优化

**分页查询**
```csharp
public async Task<PagedResultDto<LogEntryDto>> GetPagedLogsAsync(
    LogSearchRequest request)
{
    var query = await _repository.GetQueryableAsync();
    
    // 应用筛选条件
    query = ApplyFilters(query, request);
    
    // 获取总数
    var totalCount = await AsyncExecuter.CountAsync(query);
    
    // 分页
    var items = await AsyncExecuter.ToListAsync(
        query.OrderByDescending(x => x.Timestamp)
             .Skip(request.SkipCount)
             .Take(request.MaxResultCount)
    );
    
    return new PagedResultDto<LogEntryDto>(
        totalCount,
        ObjectMapper.Map<List<LogEntry>, List<LogEntryDto>>(items)
    );
}
```

### 6.2 缓存策略

**分布式缓存**
```csharp
public class CachedMetricsAppService : MetricsAppService
{
    private readonly IDistributedCache<MetricData> _cache;
    
    public override async Task<MetricData> GetRealtimeMetricAsync(string serviceName)
    {
        var cacheKey = $"metrics:realtime:{serviceName}";
        
        // 尝试从缓存读取
        var cached = await _cache.GetAsync(cacheKey);
        if (cached != null)
            return cached;
        
        // 从数据源获取
        var data = await base.GetRealtimeMetricAsync(serviceName);
        
        // 写入缓存（10秒过期）
        await _cache.SetAsync(cacheKey, data, new DistributedCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromSeconds(10)
        });
        
        return data;
    }
}
```

### 6.3 异步处理

**后台任务**
```csharp
public class LogIndexBackgroundWorker : AsyncPeriodicBackgroundWorkerBase
{
    private readonly ILogEntryRepository _repository;
    private readonly IElasticsearchService _esService;
    
    public LogIndexBackgroundWorker(
        AbpAsyncTimer timer,
        IServiceScopeFactory serviceScopeFactory,
        ILogEntryRepository repository,
        IElasticsearchService esService) 
        : base(timer, serviceScopeFactory)
    {
        _repository = repository;
        _esService = esService;
        Timer.Period = 60000; // 每分钟执行一次
    }
    
    protected override async Task DoWorkAsync(PeriodicBackgroundWorkerContext workerContext)
    {
        // 获取未索引的日志
        var unindexedLogs = await _repository.GetUnindexedLogsAsync(1000);
        
        if (unindexedLogs.Count == 0)
            return;
        
        // 批量索引
        var documents = unindexedLogs.Select(log => MapToDocument(log)).ToList();
        var success = await _esService.BulkIndexLogsAsync(documents);
        
        // 更新索引状态
        if (success)
        {
            foreach (var log in unindexedLogs)
            {
                log.MarkAsIndexed(log.Id.ToString());
            }
            await _repository.UpdateManyAsync(unindexedLogs, autoSave: true);
        }
    }
}
```
