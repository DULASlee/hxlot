#### 任务3.2: 生成历史记录
**时间**: 3小时

**需求**:
- LocalStorage存储历史记录
- 显示最近10次生成记录
- 支持重新生成（复用配置）
- 支持删除历史
- 显示生成时间、模块名称、状态

**数据结构**:
```typescript
interface GenerationHistory {
  id: string
  timestamp: number
  moduleName: string
  systemName: string
  tableName: string
  status: 'success' | 'failed' | 'pending'
  sessionId: string
  metadata: ModuleMetadata
}
```

**实现**:
```vue
<script setup lang="ts">
import { useLocalStorage } from '@vueuse/core'

const generationHistory = useLocalStorage<GenerationHistory[]>(
  'smartabp-codegen-history',
  []
)

const addToHistory = (metadata: ModuleMetadata, sessionId: string, status: string) => {
  generationHistory.value.unshift({
    id: crypto.randomUUID(),
    timestamp: Date.now(),
    moduleName: metadata.moduleName,
    systemName: metadata.systemName,
    tableName: metadata.tableName,
    status,
    sessionId,
    metadata
  })
  
  // 只保留最近10条
  if (generationHistory.value.length > 10) {
    generationHistory.value = generationHistory.value.slice(0, 10)
  }
}

const loadFromHistory = (history: GenerationHistory) => {
  formData.value = { ...history.metadata }
  ElMessage.success(t('ultraSimple.history.loaded'))
}
</script>
```
