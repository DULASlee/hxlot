    ### 🚀 快速开始

    #### 1. 安装依赖

    <div className="code-block">
      <pre>
{`
# 使用 npm
npm install @smartabp/advanced-ui-components

# 使用 yarn
yarn add @smartabp/advanced-ui-components

# 使用 pnpm
pnpm add @smartabp/advanced-ui-components
`}
      </pre>
    </div>

    #### 2. 全局引入

    <div className="code-block">
      <pre>
{`
// main.ts
import { createApp } from 'vue'
import ElementPlus from 'element-plus'
import AdvancedUI from '@smartabp/advanced-ui-components'
import 'element-plus/dist/index.css'
import '@smartabp/advanced-ui-components/dist/style.css'

const app = createApp(App)
app.use(ElementPlus)
app.use(AdvancedUI)
app.mount('#app')
`}
      </pre>
    </div>

    #### 3. 按需引入

    <div className="code-block">
      <pre>
{`
// 组件中使用
import { AdvancedTable, AdvancedForm } from '@smartabp/advanced-ui-components'

export default {
  components: {
    AdvancedTable,
    AdvancedForm
  }
}
`}
      </pre>
    </div>

    #### 4. 基本使用

    <div className="code-block">
      <pre>
{`
<template>
  <advanced-table
    :data="tableData"
    :columns="columns"
    virtual-scroll
    selectable
  />
</template>

<script setup lang="ts">
const tableData = ref([
  { id: 1, name: '张三', email: 'zhangsan@example.com' },
  { id: 2, name: '李四', email: 'lisi@example.com' }
])

const columns = [
  { key: 'id', label: 'ID', width: 80 },
  { key: 'name', label: '姓名', width: 120 },
  { key: 'email', label: '邮箱', width: 200 }
]
</script>
`}
      </pre>
    </div>
