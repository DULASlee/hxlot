## 📋 枚举类型使用

### 导入方式

```typescript
import {
  DatabaseType,
  RelationType,
  LayoutType,
  FormControlType,
  SortDirection,
  ValidationSeverity,
  PermissionAction,
  HttpMethod,
  MicroserviceType
} from '@smartabp/lowcode-shared'
```

### 使用示例

#### 1. 数据库类型选择

```typescript
// 在EntityDesigner中
const databaseType: DatabaseType = DatabaseType.PostgreSQL

// 在UI中使用enumToOptions
import { enumToOptions, DatabaseType } from '@smartabp/lowcode-shared'

const dbOptions = enumToOptions(DatabaseType, {
  'PostgreSQL': 'PostgreSQL数据库',
  'MySQL': 'MySQL数据库',
  'SQLServer': 'SQL Server数据库'
})

// 在Element Plus中
<el-select v-model="entityConfig.databaseType">
  <el-option
    v-for="opt in dbOptions"
    :key="opt.value"
    :label="opt.label"
    :value="opt.value"
  />
</el-select>
```

#### 2. 实体关系类型


## 🌐 国际化错误消息使用

### 导入方式

```typescript
import {
  setCurrentLocale,
  formatValidationError,
  ErrorMessageContext,
  getErrorMessage
} from '@smartabp/lowcode-shared'
```

### 使用示例

#### 1. 设置语言

```typescript
// 在应用初始化时
import { setCurrentLocale } from '@smartabp/lowcode-shared'

// 根据用户语言设置
const userLanguage = localStorage.getItem('language') || 'zh-CN'
setCurrentLocale(userLanguage as 'zh-CN' | 'en-US')
```

#### 2. 格式化验证错误

```typescript
// 在表单验证中
import { formatValidationError } from '@smartabp/lowcode-shared'

// 必填验证
if (!entityData.name) {
  const errorMsg = formatValidationError('name', 'required')
  // zh-CN: "名称是必填项"
  // en-US: "Name is required"
  ElMessage.error(errorMsg)
  return false
}

// 邮箱验证
if (!isValidEmail(entityData.email)) {
  const errorMsg = formatValidationError('email', 'invalid_email')
  // zh-CN: "邮箱不是有效的邮箱地址"
  // en-US: "Email is not a valid email address"
  ElMessage.error(errorMsg)
  return false
}

// 长度验证
if (entityData.name.length < 3) {
  const errorMsg = formatValidationError('name', 'too_small', {
    minimum: '3'
  })
  // zh-CN: "名称长度不足，最小需要3"
  // en-US: "Name is too short, minimum length is 3"
  ElMessage.error(errorMsg)
  return false
}
```

#### 3. 批量格式化错误

```typescript
// 在表单批量验证中
import { ErrorMessageContext } from '@smartabp/lowcode-shared'

const validateEntity = (entity: UnifiedEntityDefinition) => {
  const errors: Array<{field: string; key: ErrorMessageKey; params?: Record<string, string>}> = []
  
  if (!entity.name) {
    errors.push({ field: 'name', key: 'required' })
  }
  
  if (!entity.module) {
    errors.push({ field: 'module', key: 'required' })
  }
  
  if (entity.fields.length === 0) {
    errors.push({ field: 'fields', key: 'too_small', params: { minimum: '1' } })
  }
  
  if (errors.length > 0) {
    const context = new ErrorMessageContext('zh-CN')
    const errorMessages = context.formatBatch(errors)
    
    // 显示所有错误
    errorMessages.forEach(msg => ElMessage.error(msg))
    return false
  }
  
  return true
}
```

#### 4. 在代码生成器中使用

```typescript
// 在CodeGenerator中验证模板配置
import { formatValidationError } from '@smartabp/lowcode-shared'

const validateTemplateConfig = (config: TemplateConfig) => {
  if (!config.name) {
    throw new Error(formatValidationError('name', 'required'))
  }
  
  if (!config.version) {
    throw new Error(formatValidationError('version', 'required'))
  }
  
  if (config.fields.length === 0) {
    throw new Error(formatValidationError('fields', 'too_small', { minimum: '1' }))
  }
}
```

## 📝 模板更新指南

### 1. Entity模板更新

**Before**:
```typescript
// 硬编码字符串
const dbType = 'PostgreSQL'  // ❌ 不推荐
```

**After**:
```typescript
// 使用枚举
import { DatabaseType } from '@smartabp/lowcode-shared'
const dbType: DatabaseType = DatabaseType.PostgreSQL  // ✅ 推荐
```

### 2. RelationshipDesigner模板更新

**Before**:
```typescript
// 硬编码关系类型
if (relationshipType === 'OneToMany') { ... }  // ❌ 不推荐
```

**After**:
```typescript
// 使用枚举
import { RelationType } from '@smartabp/lowcode-shared'
if (relationshipType === RelationType.OneToMany) { ... }  // ✅ 推荐
```

### 3. FormDesigner模板更新

**Before**:
```typescript
// 硬编码错误消息
if (!fieldName) {
  ElMessage.error('字段名称是必填项')  // ❌ 不推荐
}
```

**After**:
```typescript
// 使用国际化消息
import { formatValidationError } from '@smartabp/lowcode-shared'
if (!fieldName) {
  ElMessage.error(formatValidationError('name', 'required'))  // ✅ 推荐
}
```

## 🎯 最佳实践

### 1. 枚举使用

```yaml
DO ✅:
  - 使用DatabaseType等枚举常量
  - 使用enumToOptions转换为UI选项
  - 使用isValidEnumValue验证

DON'T ❌:
  - 硬编码字符串 'PostgreSQL'
  - 手动创建选项数组
  - 字符串比较 if (type === 'PostgreSQL')
```

### 2. 错误消息

```yaml
DO ✅:
  - 使用formatValidationError格式化消息
  - 使用ErrorMessageContext批量处理
  - 设置用户语言setCurrentLocale

DON'T ❌:
  - 硬编码错误消息
  - 混合中英文消息
  - 不提供字段名称本地化
```

### 3. 类型安全

```yaml
DO ✅:
  - 明确类型注解: const type: DatabaseType
  - 使用枚举值: DatabaseType.PostgreSQL
  - 类型导入: import { DatabaseType }

DON'T ❌:
  - 使用any: const type: any
  - 字符串字面量: const type = 'PostgreSQL'
  - 类型断言: type as DatabaseType
```
